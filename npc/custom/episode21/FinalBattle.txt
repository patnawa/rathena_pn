jor_raise1,132,323,4	script	Nyar#ep21_fb_daily	22006,{
	questinfo QTYPE_DAILYQUEST,QMARK_YELLOW,"EP21_MainComplete() && EP21_FinalBattle_Daily != EP21_DailyKey()";
	mes "[Nyar]";
	if (!callfunc("EP21_MainComplete")) {
		mes "This memory cannot be repeated until the Episode 21 main story is complete.";
		close;
	}
	EP21_Main_Complete = 1;
	.@key = callfunc("EP21_DailyKey");
	if (EP21_FinalBattle_Daily == .@key) {
		mes "You already claimed today's Final Battle reward. The memory resets at 04:00 server time.";
		close;
	}
	mes "Choose the memory's difficulty. Normal requires the completed story; Hard also requires 1,000 reputation with all seven families.";
	next;
	.@choice = select("Easy.:Normal.:Hard.:Review rewards.:Leave.");
	if (.@choice == 4) {
		mes "[Final Battle Rewards]";
		mes "Easy: 10 Wigner vouchers.";
		mes "Normal: 10 vouchers and Antiquity of Giant Snake (Devouring Tan).";
		mes "Hard: 50 vouchers and Antiquity of Giant Snake (Encroached Tan Hard).";
		mes "Boss and Giant Serpent Crystal loot is separate.";
		close;
	}
	if (.@choice == 5)
		close;
	if (.@choice == 3 && !callfunc("EP21_AllFamilyReputation")) {
		mes "[Nyar]";
		mes "Hard mode remains sealed. Raise Gaebolg, Nerius, Heine, Lugenburg, Walter, Wigner, and Richard reputation to 1,000.";
		close;
	}
	setarray .@quest[1],12649,12650,12651;
	setarray .@name$[1],"Final Battle","Final Battle(Normal)","Final Battle(Hard)";
	if (isbegin_quest(.@quest[.@choice]))
		erasequest .@quest[.@choice];
	setquest .@quest[.@choice];
	callfunc "EP21_EnterInstance",.@name$[.@choice];
}

1@ep21b,1,1,0	script	#EP21_FinalBattle_Control	HIDDEN_NPC,{
	end;

OnInstanceInit:
	'fb_stage = 0;
	'fb_kind = 0;
	'fb_channels = 0;
	'fb_wave = 0;
	'fb_name$ = instance_live_info(ILI_NAME,instance_id());
	if ('fb_name$ == "Final Battle")
		enablenpc instance_npcname("Heine's Tablet#ep21_fb_start");
	else
		enablenpc instance_npcname("Nyar#ep21_fb_start");
	mapannounce instance_mapname("1@ep21b"),"Final Battle: speak with the guide at the entry point.",bc_map,0x77AAFF;
	end;

OnStoryBossDead:
	if ('fb_stage != 3) end;
	'fb_stage = 4;
	enablenpc instance_npcname("Nyar#ep21_fb_story_end");
	mapannounce instance_mapname("1@ep21b"),"The serpent and Tan fall together. Speak with Nyar.",bc_map,0xFFDD88;
	end;

OnRepeatStart:
	if ('fb_stage != 10) end;
	.@map$ = instance_mapname("1@ep21b");
	.@event$ = instance_npcname(strnpcinfo(0)) + "::OnRepeatBossDead";
	.@mob = ('fb_kind == 3) ? 22410 : 22373;
	monster .@map$,168,170,"Encroached Tan",.@mob,1,.@event$;
	'fb_boss = $@mobid[0];
	setunitdata 'fb_boss,UMOB_DMGIMMUNE,true;
	'fb_wave = 1;
	donpcevent instance_npcname(strnpcinfo(0)) + "::OnSpawnShadows";
	end;

OnSpawnShadows:
	if ('fb_stage != 10 || !unitexists('fb_boss)) end;
	.@map$ = instance_mapname("1@ep21b");
	.@event$ = instance_npcname(strnpcinfo(0)) + "::OnShadowDead";
	areamonster .@map$,158,160,178,180,"Tan's Shadow",22353,4,.@event$;
	mapannounce .@map$,"Destroy Tan's Shadows to force open a damage window.",bc_map,0xCC88FF;
	end;

OnShadowDead:
	.@event$ = instance_npcname(strnpcinfo(0)) + "::OnShadowDead";
	if ('fb_stage != 10 || mobcount(instance_mapname("1@ep21b"),.@event$) > 0) end;
	if (!unitexists('fb_boss)) end;
	setunitdata 'fb_boss,UMOB_DMGIMMUNE,false;
	if ('fb_wave >= 3) {
		mapannounce instance_mapname("1@ep21b"),"Tan is fully immobilized. Finish the battle!",bc_map,0x55FFAA;
		end;
	}
	mapannounce instance_mapname("1@ep21b"),"Tan is immobilized for 12 seconds!",bc_map,0x55FFAA;
	sleep 12000;
	if ('fb_stage != 10 || !unitexists('fb_boss)) end;
	setunitdata 'fb_boss,UMOB_DMGIMMUNE,true;
	'fb_wave++;
	donpcevent instance_npcname(strnpcinfo(0)) + "::OnSpawnShadows";
	end;

OnRepeatBossDead:
	if ('fb_stage != 10) end;
	'fb_stage = 20;
	if ('fb_kind == 3)
		enablenpc instance_npcname("Giant Serpent Crystal#ep21_fb_hard");
	else
		enablenpc instance_npcname("Giant Serpent Crystal#ep21_fb");
	enablenpc instance_npcname("Nyar#ep21_fb_reward");
	mapannounce instance_mapname("1@ep21b"),"Encroached Tan is defeated. Claim the crystal loot and speak with Nyar.",bc_map,0xFFDD88;
	end;
}

1@ep21b,172,170,4	script(DISABLED)	Nyar#ep21_fb_reward	22006,{
	if ('fb_stage != 20) end;
	mes "[Nyar]";
	.@key = callfunc("EP21_DailyKey");
	if (EP21_FinalBattle_Daily == .@key) {
		mes "You already claimed today's completion reward.";
		close;
	}
	if ('fb_kind == 3) {
		if (!checkweight(1001618,50) || !checkweight(103537,1)) {
			mes "Make room for fifty vouchers and the Hard-mode antiquity.";
			close;
		}
		getitem 1001618,50;
		getitem 103537,1;
		.@qid = 12651;
		.@reward$ = "50 Wigner vouchers and 1 Encroached Tan Hard antiquity";
	} else if ('fb_kind == 2) {
		if (!checkweight(1001618,10) || !checkweight(102948,1)) {
			mes "Make room for ten vouchers and the Normal-mode antiquity.";
			close;
		}
		getitem 1001618,10;
		getitem 102948,1;
		.@qid = 12650;
		.@reward$ = "10 Wigner vouchers and 1 Devouring Tan antiquity";
	} else {
		if (!checkweight(1001618,10)) {
			mes "Make room for ten Wigner vouchers.";
			close;
		}
		getitem 1001618,10;
		.@qid = 12649;
		.@reward$ = "10 Wigner vouchers";
	}
	if (isbegin_quest(.@qid) == 1)
		completequest .@qid;
	EP21_FinalBattle_Daily = .@key;
	mes "Memory complete. Reward: ^3366FF" + .@reward$ + "^000000.";
	close2;
	enablenpc instance_npcname("#EP21_FB_DailyExit");
	end;
}
