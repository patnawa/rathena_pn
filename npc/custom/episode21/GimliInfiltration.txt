function	script	EP21_GimliComplete	{
	return EP21_Gimli_Complete || isbegin_quest(17769) == 2;
}

function	script	EP21_EnterGimli	{
	.@name$ = "Gimli Infiltration";
	.@party_id = getcharid(1);
	if (.@party_id) {
		.@iid = instance_id(IM_PARTY);
		if (.@iid && instance_live_info(ILI_NAME,.@iid) != .@name$) {
			mes "[Gimli Infiltration]";
			mes "Your party already owns another active instance.";
			close;
		}
		if (!.@iid) {
			if (!is_party_leader()) {
				mes "[Gimli Infiltration]";
				mes "Your party leader must create the entrance first.";
				close;
			}
			.@iid = instance_create(.@name$,IM_PARTY,.@party_id);
		}
	} else {
		.@iid = instance_id(IM_CHAR);
		if (.@iid && instance_live_info(ILI_NAME,.@iid) != .@name$) {
			mes "[Gimli Infiltration]";
			mes "You already own another active solo instance.";
			close;
		}
		if (!.@iid)
			.@iid = instance_create(.@name$,IM_CHAR,getcharid(0));
	}
	if (.@iid < 1) {
		mes "[Gimli Infiltration]";
		mes "The disguised unit could not assemble. Try again shortly.";
		close;
	}
	if (instance_enter(.@name$,-1,-1,getcharid(0),.@iid) != IE_OK) {
		mes "[Gimli Infiltration]";
		mes "The temple route is not ready.";
		close;
	}
	end;
}

// Quest 17764 is the official final infiltration state immediately before
// this memorial dungeon. Episode 21 characters may enter from Base Level 230.
jor_tmple1,238,268,4	script	Disguised Temple Guard#ep21gimli	22366,{
	questinfo QTYPE_QUEST,QMARK_YELLOW,"BaseLevel >= 230 && !EP21_GimliComplete() && isbegin_quest(17764) == 0";
	mes "[Disguised Temple Guard]";
	if (BaseLevel < 230) {
		mes "Only experienced operatives may enter the restricted prayer rooms. Return at Base Level 230.";
		close;
	}
	if (callfunc("EP21_GimliComplete")) {
		EP21_Gimli_Complete = 1;
		mes "The infiltration report is complete. Maristella is preparing the Ghost Ship expedition in Fishing Village Alberta.";
		close;
	}
	if (isbegin_quest(17764) == 2) {
		erasequest 17764;
		setquest 17765;
		mes "Your unit returned safely. Report to Nadoyo through the northern safe spot at ^3366FF(jor_raise1 322, 69)^000000.";
		close;
	}
	mes "Nadoyo registered your group under false names. Carmigand's escape plan was exposed, so every prisoner must be found before the cult moves them.";
	next;
	.@choice = select("Begin Gimli Infiltration.:Review the mission.:Leave.");
	if (.@choice == 2) {
		mes "[Disguised Temple Guard]";
		mes "Follow Nadoyo, Tan and Wilhelm. Search the prayer rooms, protect the believers, and eliminate every guard that discovers the disguise.";
		close;
	}
	if (.@choice == 3)
		close;
	if (!isbegin_quest(17764))
		setquest 17764;
	callfunc "EP21_EnterGimli";
}
