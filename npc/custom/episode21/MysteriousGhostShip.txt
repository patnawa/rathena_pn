function	script	EP21_DailyKey	{
	// Daily period starts at 04:00 in the map-server's local timezone.
	.@now = gettimetick(2);
	.@reset = .@now - (gettime(DT_HOUR) * 3600 + gettime(DT_MINUTE) * 60 + gettime(DT_SECOND)) + 14400;
	if (.@now < .@reset)
		.@reset -= 86400;
	return .@reset;
}

function	script	EP21_GhostShipUnlocked	{
	// The quest checks migrate characters that completed this section before
	// the custom unlock variable was introduced.
	return EP21_GhostShip_Access || isbegin_quest(16818) || isbegin_quest(16819) == 2;
}

function	script	EP21_AddReputation	{
	// Reputation type 13 is Gaebolg Resistance in db/import/reputation.yml.
	add_reputation_points 13,getarg(0);
	// Mirror the value for compatibility with early custom Episode 21 scripts.
	EP21_Reputation = get_reputation_points(13);
	return EP21_Reputation;
}

function	script	EP21_EnterGhostShip	{
	.@name$ = "Mysterious Ghost Ship";
	.@party_id = getcharid(1);

	if (.@party_id) {
		.@iid = instance_id(IM_PARTY);
		if (.@iid && instance_live_info(ILI_NAME,.@iid) != .@name$) {
			mes "[Ghost Ship Expedition]";
			mes "Your party already owns another active instance. Close it from the instance window first.";
			close;
		}
		if (!.@iid) {
			if (!is_party_leader()) {
				mes "[Ghost Ship Expedition]";
				mes "Your party leader must create the entrance first.";
				close;
			}
			.@iid = instance_create(.@name$,IM_PARTY,.@party_id);
		}
	} else {
		// Party play is recommended, but a solo owner is supported so content
		// never becomes inaccessible on a low-population server.
		.@iid = instance_id(IM_CHAR);
		if (.@iid && instance_live_info(ILI_NAME,.@iid) != .@name$) {
			mes "[Ghost Ship Expedition]";
			mes "You already own another active solo instance. Close it from the instance window first.";
			close;
		}
		if (!.@iid)
			.@iid = instance_create(.@name$,IM_CHAR,getcharid(0));
	}

	if (.@iid < 1) {
		mes "[Ghost Ship Expedition]";
		mes "The ship could not be reached. Please try again shortly.";
		close;
	}
	if (instance_enter(.@name$,-1,-1,getcharid(0),.@iid) != IE_OK) {
		mes "[Ghost Ship Expedition]";
		mes "The entrance is not ready yet.";
		close;
	}
	end;
}

jor_albe,190,186,4	script	Maristella Walter#ep21gs	22306,{
	questinfo QTYPE_QUEST,QMARK_YELLOW,"BaseLevel >= 230 && EP21_GimliComplete() && isbegin_quest(16816) == 0 && !EP21_GhostShipUnlocked()";
	mes "[Maristella Walter]";
	if (BaseLevel < 230) {
		mes "The Ghost Ship is too dangerous. Return after reaching Base Level 230.";
		close;
	}
	if (isbegin_quest(16817) == 1) {
		mes "That was my father's ship. Megas has received our report, but a Wigner executive is waiting nearby.";
		changequest 16817,16819;
		close;
	}
	if (isbegin_quest(16819) == 1) {
		mes "Please speak with the Wigner Executive beside the harbor.";
		close;
	}
	if (callfunc("EP21_GhostShipUnlocked")) {
		EP21_GhostShip_Access = 1;
		mes "Nillem now manages repeat expeditions from the lower harbor at ^3366FF(195, 179)^000000.";
		close;
	}
	if (!callfunc("EP21_GimliComplete") && isbegin_quest(16816) == 0) {
		mes "We still need the cult's shipping records. Complete Gimli Infiltration through the guard at ^3366FF(jor_tmple1 238, 268)^000000 first.";
		close;
	}
	if (isbegin_quest(16816) == 2) {
		erasequest 16816;
		setquest 16817;
		mes "Megas needs our report. Speak to me once more.";
		close;
	}
	mes "We found the ship that may have belonged to my father, but its crew no longer seems human.";
	mes "We must board it, examine the Walter family's traces, and learn what happened.";
	next;
	if (select("Board the Mysterious Ghost Ship.:Not yet.") == 2)
		close;
	if (!isbegin_quest(16816))
		setquest 16816;
	callfunc "EP21_EnterGhostShip";
}

// Daily entrance and report NPC. Published daily access unlocks after the
// story reward and resets at 04:00 server time.
jor_albe,195,179,4	script	Nillem#ep21gs_daily	4_M_SAGE_C,{
	questinfo QTYPE_DAILYQUEST,QMARK_YELLOW,"EP21_GhostShipUnlocked() && EP21_GhostShip_Daily != EP21_DailyKey()";
	mes "[Nillem]";
	if (!callfunc("EP21_GhostShipUnlocked")) {
		mes "Maristella's first expedition has not identified the Ghost Ship yet.";
		close;
	}
	EP21_GhostShip_Access = 1;
	.@key = callfunc("EP21_DailyKey");

	if (isbegin_quest(16822) == 1) {
		if (!checkweight(1001618,10) || !checkweight(102947,1)) {
			mes "Make room for ten vouchers and the Ghost Ship antiquity before I file the report.";
			close;
		}
		changequest 16822,16823;
		EP21_GhostShip_Daily = .@key;
		callfunc "EP21_AddReputation",5;
		getitem 1001618,10;
		getitem 102947,1;
		getexp 91298960,59995050;
		mes "The surviving memories confirm your report.";
		mes "Reward: ^3366FF10 Wigner Merchant Guild Vouchers^000000, ^3366FF1 Ghost Ship Antiquity^000000 and ^33AA335 Gaebolg Command reputation^000000.";
		close;
	}
	if (EP21_GhostShip_Daily == .@key) {
		mes "You already completed today's Ghost Ship investigation.";
		mes "The expedition resets at 04:00 server time.";
		close;
	}
	if (isbegin_quest(16823))
		erasequest 16823;
	if (isbegin_quest(16821) == 2)
		erasequest 16821;

	mes "The Ghost Ship has returned. We still need fragments of old Walter's memories.";
	mes "Defeat its captain and report back to me. The published route is unchanged, but this daily version is exceptionally dangerous.";
	next;
	.@choice = select("Begin today's expedition.:Check reputation.:Not now.");
	if (.@choice == 2) {
		mes "[Nillem]";
		mes "Gaebolg Command reputation: ^33AA33" + get_reputation_points(13) + "^000000";
		close;
	}
	if (.@choice == 3)
		close;
	if (!isbegin_quest(16821))
		setquest 16821;
	callfunc "EP21_EnterGhostShip";
}
