1@ep21a,1,1,0	script	#EP21_SecretAltar_Control	HIDDEN_NPC,{
	end;
OnInstanceInit:
	'sa_stage = 0;
	enablenpc instance_npcname("Lehar#ep21_sa_start");
	mapannounce instance_mapname("1@ep21a"),"Secret Altar: speak with Lehar at the corridor entrance.",bc_map,0x77AAFF;
	end;
OnCorridorDead:
	if ('sa_stage != 1 || mobcount(instance_mapname("1@ep21a"),instance_npcname(strnpcinfo(0))+"::OnCorridorDead") > 0) end;
	'sa_stage = 2;
	enablenpc instance_npcname("Door#ep21_sa_1");
	mapannounce instance_mapname("1@ep21a"),"The first altar door is open.",bc_map,0x55FFAA;
	end;
OnPriestsDead:
	if ('sa_stage != 3 || mobcount(instance_mapname("1@ep21a"),instance_npcname(strnpcinfo(0))+"::OnPriestsDead") > 0) end;
	'sa_stage = 4;
	enablenpc instance_npcname("Jormungandr Shaman#ep21_sa");
	end;
OnShamanDead:
	if ('sa_stage != 5) end;
	'sa_stage = 6;
	enablenpc instance_npcname("Door#ep21_sa_2");
	end;
OnSerpentsDead:
	if ('sa_stage != 7 || mobcount(instance_mapname("1@ep21a"),instance_npcname(strnpcinfo(0))+"::OnSerpentsDead") > 0) end;
	'sa_stage = 8;
	enablenpc instance_npcname("Giant Egg#ep21_sa");
	instance_warpall "1@ep21a",126,155,instance_id();
	mapannounce instance_mapname("1@ep21a"),"The cult's last vessels collapse. Examine the Giant Egg.",bc_map,0xFFDD88;
	end;
}
