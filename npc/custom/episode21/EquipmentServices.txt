mbase_in,220,127,4	script	Wigner Merchant Popo#ep21_hammer	4_M_MANAGER,{
	mes "[Wigner Merchant Popo]";
	if (!callfunc("EP21_MainComplete")) {
		mes "The refine-hammer counter is available after completing Episode 21.";
		close;
	}
	mes "Which Gaebolg refine hammer do you need?";
	next;
	switch (select("+9 to +10 hammer - 50 vouchers.:+10 to +11 hammer - 4,000,000 Zeny and 100 vouchers.:Leave.")) {
	case 1:
		if (countitem(1001618) < 50) {
			mes "You need 50 Wigner Merchant Guild Vouchers.";
			close;
		}
		if (!checkweight(102881,1)) {
			mes "You cannot carry the hammer.";
			close;
		}
		delitem 1001618,50;
		getitem 102881,1;
		mes "Use the hammer on eligible +9 Gaebolg armor.";
		close;
	case 2:
		if (get_reputation_points(13) < 1000) {
			mes "This hammer requires 1,000 Gaebolg Resistance reputation.";
			close;
		}
		if (Zeny < 4000000 || countitem(1001618) < 100) {
			mes "You need 4,000,000 Zeny and 100 Wigner Merchant Guild Vouchers.";
			close;
		}
		if (!checkweight(102882,1)) {
			mes "You cannot carry the hammer.";
			close;
		}
		Zeny -= 4000000;
		delitem 1001618,100;
		getitem 102882,1;
		mes "Use the hammer on eligible +10 Gaebolg armor.";
		close;
	default:
		close;
	}
}
