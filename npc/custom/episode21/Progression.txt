function	script	EP21_QuestInRange	{
	for (.@qid = getarg(0); .@qid <= getarg(1); ++.@qid) {
		if (isbegin_quest(.@qid))
			return true;
	}
	return false;
}

function	script	EP21_EntryRequirement	{
	return BaseLevel >= 230 && isbegin_quest(18233) == 2;
}

function	script	EP21_Started	{
	return EP21_StoryStage || callfunc("EP21_QuestInRange",18334,18361) ||
		callfunc("EP21_QuestInRange",23240,23255) ||
		callfunc("EP21_QuestInRange",17752,17769) || EP21_Main_Complete;
}

function	script	EP21_LunaformaAccess	{
	return EP21_Lunaforma_Access || callfunc("EP21_QuestInRange",23240,23255) ||
		callfunc("EP21_QuestInRange",17752,17769) ||
		callfunc("EP21_QuestInRange",16816,16826) ||
		callfunc("EP21_QuestInRange",18338,18361) || EP21_Main_Complete;
}

function	script	EP21_GaebolgComplete	{
	return EP21_Gaebolg_Complete || isbegin_quest(23248) == 2 ||
		callfunc("EP21_QuestInRange",17752,17769) ||
		callfunc("EP21_QuestInRange",16816,16826) ||
		callfunc("EP21_QuestInRange",18338,18361) || EP21_Main_Complete;
}

function	script	EP21_AlbertaAccess	{
	// The warper to Alberta is a Gaebolg Resistance completion reward; the
	// village introduction itself is tracked separately by EP21_Alberta_Complete.
	return callfunc("EP21_GaebolgComplete");
}

function	script	EP21_TempleAccess	{
	return EP21_Temple_Access || callfunc("EP21_QuestInRange",17761,17769) ||
		callfunc("EP21_QuestInRange",16816,16826) ||
		callfunc("EP21_QuestInRange",18338,18361) || EP21_Main_Complete;
}

function	script	EP21_CultComplete	{
	return EP21_Cult_Complete || isbegin_quest(17764) ||
		callfunc("EP21_QuestInRange",17765,17769) ||
		callfunc("EP21_QuestInRange",16816,16826) ||
		callfunc("EP21_QuestInRange",18338,18361) || EP21_Main_Complete;
}

function	script	EP21_WarperAccess	{
	.@map$ = getarg(0);
	if (.@map$ == "jor_tail" || .@map$ == "jor_crk") {
		if (callfunc("EP21_EntryRequirement") || callfunc("EP21_Started"))
			return true;
		message strcharinfo(0),"Episode 21 requires Base Level 230 and completion of Episode 20: The Immortal.";
		return false;
	}
	if (.@map$ == "jor_mbase") {
		if (callfunc("EP21_LunaformaAccess")) return true;
		message strcharinfo(0),"Reach Lunaforma through Shufapa's Episode 21 investigation first.";
		return false;
	}
	if (.@map$ == "jor_albe") {
		if (callfunc("EP21_AlbertaAccess")) return true;
		message strcharinfo(0),"Complete the Gaebolg Resistance and Old Alberta introduction first.";
		return false;
	}
	if (.@map$ == "jor_raise1" || .@map$ == "jor_raise2") {
		if (callfunc("EP21_GaebolgComplete")) return true;
		message strcharinfo(0),"Complete the Gaebolg Resistance assignment first.";
		return false;
	}
	if (.@map$ == "jor_tmple1" || .@map$ == "jor_tmple2") {
		if (callfunc("EP21_TempleAccess")) return true;
		message strcharinfo(0),"The Jormungandr Temple route has not been opened by the resistance.";
		return false;
	}
	if (.@map$ == "jor_sklf1" || .@map$ == "jor_sklf2") {
		if (callfunc("EP21_MainComplete")) return true;
		message strcharinfo(0),"Complete Episode 21 before entering the cult's end-game floors.";
		return false;
	}
	return true;
}

function	script	EP21_InstanceWarperAccess	{
	switch (getarg(0)) {
	case 21:
		if (isbegin_quest(17764) == 1) return true;
		message strcharinfo(0),"Complete Cult Infiltration and receive Nadoyo's Gimli operation first.";
		return false;
	case 22:
		if (callfunc("EP21_GimliComplete")) return true;
		message strcharinfo(0),"Complete Gimli Infiltration first.";
		return false;
	case 23:
		if (isbegin_quest(18347) || isbegin_quest(18348) || isbegin_quest(18349) || callfunc("EP21_MainComplete")) return true;
		message strcharinfo(0),"Advance the Black-Haired Beast investigation to the Final Battle.";
		return false;
	case 24:
		if (isbegin_quest(18350) || isbegin_quest(18351) || isbegin_quest(18352) || callfunc("EP21_MainComplete")) return true;
		message strcharinfo(0),"Reach the Suspicious Book and Shining Door first.";
		return false;
	case 25:
		if (isbegin_quest(18358) || isbegin_quest(18359) || isbegin_quest(18360) || callfunc("EP21_MainComplete")) return true;
		message strcharinfo(0),"Reach the Lasgand investigation before entering Silent Sanctuary.";
		return false;
	}
	return false;
}
