function	script	EP21_MainComplete	{
	return EP21_Main_Complete || isbegin_quest(18360) == 2 || isbegin_quest(18361) == 2;
}

function	script	EP21_EnterInstance	{
	.@name$ = getarg(0);
	.@party_id = getcharid(1);
	if (.@party_id) {
		.@iid = instance_id(IM_PARTY);
		if (.@iid && instance_live_info(ILI_NAME,.@iid) != .@name$) {
			mes "[Episode 21]";
			mes "Your party already owns another active instance.";
			close;
		}
		if (!.@iid) {
			if (!is_party_leader()) {
				mes "[Episode 21]";
				mes "Your party leader must create the entrance first.";
				close;
			}
			.@iid = instance_create(.@name$,IM_PARTY,.@party_id);
		}
	} else {
		.@iid = instance_id(IM_CHAR);
		if (.@iid && instance_live_info(ILI_NAME,.@iid) != .@name$) {
			mes "[Episode 21]";
			mes "You already own another active solo instance.";
			close;
		}
		if (!.@iid)
			.@iid = instance_create(.@name$,IM_CHAR,getcharid(0));
	}
	if (.@iid < 1) {
		mes "[Episode 21]";
		mes "The temporal route could not be stabilized. Try again shortly.";
		close;
	}
	if (instance_enter(.@name$,-1,-1,getcharid(0),.@iid) != IE_OK) {
		mes "[Episode 21]";
		mes "The instance map is not ready.";
		close;
	}
	end;
}
