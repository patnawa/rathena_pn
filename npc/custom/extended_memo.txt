// Independent implementation of the published Missionary Work quest.
// PN extended Warp Portal memo slots.
function	script	F_PNMemoEligible	{
	return (Class == Job_Cardinal || Class == Job_Inquisitor) && getskilllv("AL_WARP") >= 4;
}

prontera,225,330,4	script	Missionary Rosetta#PNMemo	4_F_SISTER,{
	mes "[Missionary Rosetta]";
	if (!callfunc("F_PNMemoEligible")) {
		mes "Cardinals and Inquisitors with Warp Portal level 4 can join our missionary work.";
		close;
	}
	if (PN_MissionStage == 4) {
		mes "Your Warp Portal remembers six locations. Use /memo to save a destination.";
		mes "Resetting your skills removes this reward and you must repeat our work.";
		close;
	}
	if (PN_MissionStage == 0) {
		mes "Help our people in Umbala and Einbroch. Complete both missions to gain three extra memo slots.";
		mes "A skill reset clears this quest and its extra slots.";
		if (select("I'll participate:Not now") == 2)
			close;
		if (!callfunc("F_PNMemoEligible") || PN_MissionStage != 0)
			close;
		PN_MissionStage = 1;
		PN_MemoExtra = 1;
		setquest 16586;
		mes "You may now remember four locations. Save the town with /memo when you arrive.";
	}
	if (PN_MissionStage == 3 && PN_MissionDone == 3) {
		PN_MemoExtra = 3;
		PN_MissionStage = 4;
		completequest 16586;
		mes "Both communities are safe. You can now remember six destinations.";
		close;
	}
	if (PN_MissionRegion) {
		mes "Your current mission is in " + ((PN_MissionRegion == 1) ? "Umbala. Speak with Tita." : "Einbroch. Speak with Suan.");
		if (select("Travel to mission:Stay here") == 2)
			close;
		.@region = PN_MissionRegion;
	} else {
		if (PN_MissionDone == 0)
			.@region = select("Umbala:Einbroch:Not now");
		else {
			.@region = (PN_MissionDone == 1) ? 2 : 1;
			mes "Your next mission is in " + ((.@region == 1) ? "Umbala. Speak with Tita." : "Einbroch. Speak with Suan.");
			if (select("Accept and travel:Not now") == 2)
				close;
		}
		if (.@region == 3)
			close;
		if (!callfunc("F_PNMemoEligible") || PN_MissionStage != 1 || PN_MissionRegion)
			close;
		PN_MissionRegion = .@region;
		PN_MissionRescue = 0;
		PN_MissionStage = 2;
		setquest 16586 + .@region;
	}
	// Scripts cannot read memo slots, so ask before replacing the fourth one.
	if (PN_MissionDone) {
		mes "I can also record the town in your fourth memo slot when you arrive. This replaces anything saved there.";
		.@memo = (select("Keep my memo slots:Record the town in memo slot 4") == 2);
	}
	if (!callfunc("F_PNMemoEligible") || PN_MissionRegion != .@region)
		close;
	close2;
	if (.@region == 1)
		warp "umbala",127,138;
	else
		warp "einbroch",178,146;
	// The second assignment records its town in slot four only when the player agreed.
	if (.@memo)
		atcommand "@memo 3";
	end;
}

function	script	F_PNMemoMission	{
	.@region = getarg(0);
	.@hunt = (.@region == 1) ? 16592 : 16596;
	mes "[" + getarg(1) + "]";
	if (!callfunc("F_PNMemoEligible") || PN_MissionStage != 2 || PN_MissionRegion != .@region) {
		mes "Missionary Rosetta in Prontera coordinates our rescue work.";
		close;
	}
	if (checkquest(.@hunt) == -1) {
		mes (.@region == 1) ? "Defeat 100 Brutal Murderers in the Dead's Banquet Hall and rescue Jhon, Maria and Blue." : "Defeat 100 Abyssmen in Einbech Mine 3 and rescue Las, Minas and Tiris.";
		if (select("I'll help:Not now") == 2)
			close;
		if (!callfunc("F_PNMemoEligible") || PN_MissionStage != 2 || PN_MissionRegion != .@region || checkquest(.@hunt) != -1)
			close;
		setquest .@hunt;
		for (.@q = .@hunt - 3; .@q < .@hunt; ++.@q)
			setquest .@q;
		mes "Find all three missing people and complete the hunt, then report to me.";
		close;
	}
	if (checkquest(.@hunt,HUNTING) != 2 || PN_MissionRescue != 7) {
		mes "The hunt and all three rescues must be complete.";
		close;
	}
	// No yield between checking completion and updating the character's progress.
	completequest .@hunt;
	completequest 16586 + .@region;
	PN_MissionDone |= .@region;
	PN_MissionRegion = 0;
	PN_MissionRescue = 0;
	PN_MissionStage = (PN_MissionDone == 3) ? 3 : 1;
	mes "Everyone has returned safely. Report to Missionary Rosetta in Prontera.";
	close;
}

umbala,127,140,4	script	Tita#PNMemo	4_F_UMGIRL,{
	callfunc "F_PNMemoMission",1,"Tita";
	end;
}
einbroch,178,148,4	script	Suan#PNMemo	4_M_EINMAN,{
	callfunc "F_PNMemoMission",2,"Suan";
	end;
}

function	script	F_PNMemoRescue	{
	.@region = getarg(0);
	.@bit = getarg(1);
	.@quest = getarg(2);
	.@hunt = (.@region == 1) ? 16592 : 16596;
	mes "[" + getarg(3) + "]";
	if (!callfunc("F_PNMemoEligible") || PN_MissionStage != 2 || PN_MissionRegion != .@region || checkquest(.@hunt) < 0) {
		mes "Please speak with the missionary coordinator in town first.";
		close;
	}
	if (PN_MissionRescue & .@bit) {
		mes "You already helped me home. Thank you!";
		close;
	}
	PN_MissionRescue |= .@bit;
	completequest .@quest;
	specialeffect EF_PORTAL;
	cloakonnpcself;
	mes "Your portal will take me home safely. Thank you!";
	close;
}
