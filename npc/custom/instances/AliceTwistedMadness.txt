// Alice's Twisted Madness: original maze encounter on purpose-built terrain.
// Daily admission is shared by Normal and Hard. Timed bonus is measured from reservation.
function	script	F_AliceTravel	{
	.@cid = getcharid(0);
	'alice_phase[.@cid] = getarg(0);
	'alice_map[.@cid] = getarg(1);
	'alice_x[.@cid] = getarg(2); 'alice_y[.@cid] = getarg(3);
	warp instance_mapname(getarg(1) == 2 ? "2@alice_mad":"1@alice_mad"),getarg(2),getarg(3);
	return;
}

function	script	F_AliceMember	{
	if (getcharid(1) != 'alice_party) return 0;
	if (!'alice_started_member[getcharid(0)]) return 0;
	if ('alice_boss_started) return 0;
	return 1;
}

function	script	F_AliceCorridor	{
	if (!callfunc("F_AliceMember")) return;
	.@cid = getcharid(0); .@corridor = getarg(0);
	if (.@corridor != 'alice_corridor[.@cid]) return;
	if ('alice_phase[.@cid] != 1 && 'alice_phase[.@cid] != 3) return;
	if (rand(100) < 10) {
		'alice_return_phase[.@cid] = 'alice_phase[.@cid];
		callfunc "F_AliceTravel",2,1,14,114;
		return;
	}
	if (.@corridor == 6) {
		callfunc "F_AliceTravel",4,1,360,160;
	} else if ('alice_door[.@cid] == 'alice_solution[.@cid]) {
		'alice_corridor[.@cid] = 6;
		callfunc "F_AliceTravel",3,1,72,20;
	} else callfunc "F_AliceTravel",0,1,110,110;
	return;
}

function	script	F_AliceLeadership	{
	if (!'alice_started || 'alice_boss_started || getcharid(1) != 'alice_party || !is_party_leader()) return;
	.@party = getcharid(1);
	.@n = getpartymember(.@party,1,.@cid);
	getpartymember .@party,2,.@aid;
	.@count = 0;
	for (.@i = 0; .@i < .@n; ++.@i) {
		if (.@cid[.@i] == getcharid(0)) continue;
		if (!isloggedin(.@aid[.@i],.@cid[.@i])) continue;
		if (!'alice_started_member[.@cid[.@i]]) continue;
		.@target[.@count] = .@cid[.@i]; .@account[.@count] = .@aid[.@i];
		.@menu$ += "Traveler "+.@cid[.@i]+":";
		mes .@cid[.@i]+" - "+rid2name(.@aid[.@i]);
		++.@count;
	}
	if (!.@count) { dispbottom "No connected expedition member is available to lead."; return; }
	mes "Choose a connected expedition member to lead and recall the group.";
	.@choice = select(.@menu$+"Cancel")-1;
	if (.@choice >= .@count) return;
	if (getcharid(1) != .@party || !is_party_leader() || 'alice_boss_started) return;
	.@cid = .@target[.@choice]; .@aid = .@account[.@choice];
	if (!isloggedin(.@aid,.@cid)) return;
	if (getcharid(1,rid2name(.@aid)) != .@party || !'alice_started_member[.@cid]) return;
	if (party_changeleader(.@party,.@cid) == 1) dispbottom "Expedition leadership has changed.";
	return;
}
