function	script	OGHCM_MaxLevel	{
	.@points = getarg(0);
	setarray .@need[1],0,5,11,18,26,35,45,60,80,100;
	.@level = 1;
	for (.@i = 2; .@i <= 10; ++.@i)
		if (.@points >= .@need[.@i]) .@level = .@i;
	return .@level;
}

function	script	OGHCM_ScaleMob	{
	.@gid = getarg(0);
	.@level = getarg(1);
	if (!unitexists(.@gid)) return 0;
	getunitdata .@gid,.@data;
	setunitdata .@gid,UMOB_ATKMIN,min(65000,.@data[UMOB_ATKMIN] * .@level);
	setunitdata .@gid,UMOB_ATKMAX,min(65000,.@data[UMOB_ATKMAX] * .@level);
	return 1;
}

function	script	OGHCM_IsCirclet	{
	.@id = getarg(0);
	return .@id >= 19474 && .@id <= 19492;
}

function	script	OGHCM_IsSpecialEnchant	{
	.@id = getarg(0);
	setarray .@base[0],29706,29061,29111,29101,29091,29071,29081;
	for (.@i = 0; .@i < getarraysize(.@base); ++.@i)
		if (.@id >= .@base[.@i] && .@id <= .@base[.@i] + 9)
			return .@base[.@i];
	return 0;
}

// Rebuilds the equipped circlet while preserving refine, grade, physical card,
// identification, break status and all five random options.
function	script	OGHCM_RebuildCirclet	{
	.@new0 = getarg(0); .@new1 = getarg(1);
	.@new2 = getarg(2); .@new3 = getarg(3);
	getinventorylist;
	.@idx = -1;
	for (.@i = 0; .@i < @inventorylist_count; ++.@i) {
		if ((@inventorylist_equip[.@i] & EQP_HEAD_TOP) && callfunc("OGHCM_IsCirclet",@inventorylist_id[.@i])) {
			.@idx = .@i;
			break;
		}
	}
	if (.@idx < 0) return 0;
	setarray .@oid[0],@inventorylist_option_id1[.@idx],@inventorylist_option_id2[.@idx],@inventorylist_option_id3[.@idx],@inventorylist_option_id4[.@idx],@inventorylist_option_id5[.@idx];
	setarray .@oval[0],@inventorylist_option_value1[.@idx],@inventorylist_option_value2[.@idx],@inventorylist_option_value3[.@idx],@inventorylist_option_value4[.@idx],@inventorylist_option_value5[.@idx];
	setarray .@opar[0],@inventorylist_option_parameter1[.@idx],@inventorylist_option_parameter2[.@idx],@inventorylist_option_parameter3[.@idx],@inventorylist_option_parameter4[.@idx],@inventorylist_option_parameter5[.@idx];
	.@id = @inventorylist_id[.@idx];
	.@identify = @inventorylist_identify[.@idx];
	.@refine = @inventorylist_refine[.@idx];
	.@attribute = @inventorylist_attribute[.@idx];
	.@grade = @inventorylist_enchantgrade[.@idx];
	delitemidx @inventorylist_idx[.@idx],1;
	getitem4 .@id,1,.@identify,.@refine,.@attribute,.@new0,.@new1,.@new2,.@new3,.@grade,.@oid,.@oval,.@opar;
	return 1;
}

1@gl_he,1,1,0	script	#OGHCM_Control	HIDDEN_NPC,{
	end;

OnInstanceInit:
	'ogh_stage = 0;
	'ogh_map$ = instance_mapname("1@gl_he");
	'ogh_token$ = "OGHCM:"+instance_id()+":"+gettimetick(2);
	disablenpc instance_npcname("West Gate#oghcm");
	disablenpc instance_npcname("East Gate#oghcm");
	disablenpc instance_npcname("Oscar#oghcm_center");
	disablenpc instance_npcname("Oscar#oghcm_boss");
	disablenpc instance_npcname("Temporal Treasure#oghcm");
	disablenpc instance_npcname("Oscar#oghcm_reward");
	disablenpc instance_npcname("Exit Distortion#oghcm");
	mapannounce 'ogh_map$,"Old Glast Heim Challenge: speak with Oscar to select the contamination level.",bc_map,0x77AAFF;
	end;

OnSpawnWings:
	if ('ogh_stage != 1) end;
	setarray .@mobs[0],20574,20576,20577,20578,20579,20580;
	.@left_event$ = instance_npcname(strnpcinfo(0))+"::OnLeftDead";
	.@right_event$ = instance_npcname(strnpcinfo(0))+"::OnRightDead";
	for (.@i = 0; .@i < 50; ++.@i) {
		areamonster 'ogh_map$,35,45,130,260,"Distorted Glast Heim Soldier",.@mobs[rand(getarraysize(.@mobs))],1,.@left_event$;
		callfunc "OGHCM_ScaleMob",$@mobid[0],'ogh_level;
		areamonster 'ogh_map$,170,45,265,260,"Distorted Glast Heim Soldier",.@mobs[rand(getarraysize(.@mobs))],1,.@right_event$;
		callfunc "OGHCM_ScaleMob",$@mobid[0],'ogh_level;
	}
	.@count = getmapunits(BL_PC,'ogh_map$,.@players[0]);
	for (.@i = 0; .@i < .@count; ++.@i)
		unitwarp .@players[.@i],'ogh_map$,rand(2) ? 70 : 230,150;
	end;

OnLeftDead:
	if ('ogh_stage != 1) end;
	++'ogh_left_kills;
	if ('ogh_left_kills == 47) {
		'ogh_left_open = 1;
		enablenpc instance_npcname("West Gate#oghcm");
		mapannounce 'ogh_map$,"Oscar: The west wing is stable. Its return gate is now open!",bc_map,0x55FFAA;
		if (!'ogh_root_started) donpcevent instance_npcname(strnpcinfo(0))+"::OnRootStart";
		callsub L_TryCenter;
	}
	end;

OnRightDead:
	if ('ogh_stage != 1) end;
	++'ogh_right_kills;
	if ('ogh_right_kills == 47) {
		'ogh_right_open = 1;
		enablenpc instance_npcname("East Gate#oghcm");
		mapannounce 'ogh_map$,"Oscar: The east wing is stable. Its return gate is now open!",bc_map,0x55FFAA;
		if (!'ogh_root_started) donpcevent instance_npcname(strnpcinfo(0))+"::OnRootStart";
		callsub L_TryCenter;
	}
	end;

OnRootStart:
	if ('ogh_root_started) end;
	'ogh_root_started = 1;
	.@maximum = ('ogh_level >= 8) ? 4 : (('ogh_level >= 6) ? 3 : (('ogh_level >= 4) ? 2 : 1));
	'ogh_root_active = rand(1,.@maximum);
	.@event$ = instance_npcname(strnpcinfo(0))+"::OnRootDead";
	for (.@i = 0; .@i < 'ogh_root_active; ++.@i) {
		monster 'ogh_map$,142 + (.@i % 2) * 16,178 + (.@i / 2) * 14,"Prime Corruption Root",20575,1,.@event$;
		callfunc "OGHCM_ScaleMob",$@mobid[0],'ogh_level;
	}
	mapannounce 'ogh_map$,"Oscar: Prime Corruption Roots appeared in the center! You have 30 seconds before they erupt!",bc_map,0xFF5555;
	sleep 30000;
	if ('ogh_root_done || 'ogh_stage != 1) end;
	'ogh_root_timeout = 1;
	'ogh_root_done = 1;
	'ogh_root_active = 0;
	.@count = getareaunits(BL_PC,'ogh_map$,125,160,175,210,.@players[0]);
	for (.@i = 0; .@i < .@count; ++.@i)
		unitkill .@players[.@i];
	killmonster 'ogh_map$,.@event$;
	mapannounce 'ogh_map$,"Oscar destroys the unstable roots. Anyone caught near the center is consumed by the eruption!",bc_map,0xFF3333;
	callsub L_TryCenter;
	end;

OnRootDead:
	if ('ogh_root_timeout || 'ogh_stage != 1) end;
	++'ogh_root_kills;
	--'ogh_root_active;
	if ('ogh_root_active <= 0) {
		'ogh_root_done = 1;
		mapannounce 'ogh_map$,"All Prime Corruption Roots were destroyed before the eruption.",bc_map,0x55FFAA;
		callsub L_TryCenter;
	}
	end;

OnBossStart:
	if ('ogh_stage != 3 || 'ogh_boss) end;
	.@bossid = (rand(100) < 5 * 'ogh_level) ? 20572 : 20573;
	'ogh_himmelmez = (.@bossid == 20572);
	monster 'ogh_map$,155,280,'ogh_himmelmez ? "Phantom Himmelmez" : "Phantom Amdarais",.@bossid,1,instance_npcname(strnpcinfo(0))+"::OnBossDead";
	'ogh_boss = $@mobid[0];
	setarray .@hp[1],600000000,750000000,900000000,1050000000,1200000000,1350000000,1500000000,1650000000,1800000000,2050000000;
	'ogh_boss_maxhp = .@hp['ogh_level];
	setunitdata 'ogh_boss,UMOB_MAXHP,'ogh_boss_maxhp;
	setunitdata 'ogh_boss,UMOB_HP,'ogh_boss_maxhp;
	callfunc "OGHCM_ScaleMob",'ogh_boss,'ogh_level;
	'ogh_phase = 0; 'ogh_outside = 0; 'ogh_boss_tick = 0;
	.@thorn_event$ = instance_npcname(strnpcinfo(0))+"::OnThornDead";
	setarray .@tx[0],125,135,145,155,165,175,125,175,125,175,125,135,145,155,165,175;
	setarray .@ty[0],255,255,255,255,255,255,265,265,280,280,295,295,295,295,295,295;
	for (.@i = 0; .@i < getarraysize(.@tx); ++.@i) {
		monster 'ogh_map$,.@tx[.@i],.@ty[.@i],"Mutated Thorn",20581,1,.@thorn_event$;
		setunitdata $@mobid[0],UMOB_MAXHP,5 + 'ogh_level * 4;
		setunitdata $@mobid[0],UMOB_HP,5 + 'ogh_level * 4;
	}
	mapannounce 'ogh_map$,'ogh_himmelmez ? "Phantom Himmelmez tears through the distortion!" : "Phantom Amdarais rises from the distortion!",bc_map,0xFF7777;
	donpcevent instance_npcname(strnpcinfo(0))+"::OnBossMonitor";
	end;

OnBossMonitor:
	while ('ogh_stage == 3 && unitexists('ogh_boss)) {
		sleep 2000;
		if ('ogh_stage != 3 || !unitexists('ogh_boss)) end;
		getunitdata 'ogh_boss,.@data;
		.@newhp = min('ogh_boss_maxhp,.@data[UMOB_HP] + 300000 * 'ogh_level);
		if (.@newhp > .@data[UMOB_HP]) setunitdata 'ogh_boss,UMOB_HP,.@newhp;
		if (.@data[UMOB_X] < 120 || .@data[UMOB_X] > 182 || .@data[UMOB_Y] < 250 || .@data[UMOB_Y] > 298)
			'ogh_outside += 2;
		else
			'ogh_outside = 0;
		if ('ogh_outside >= 30) {
			'ogh_boss_reset = 1;
			'ogh_outside = 0;
			'ogh_phase = 0;
			unitwarp 'ogh_boss,'ogh_map$,155,280;
			setunitdata 'ogh_boss,UMOB_HP,'ogh_boss_maxhp;
			mapannounce 'ogh_map$,"Oscar: The boss left the arena too long. The battle has reset at full health!",bc_map,0xFF3333;
			continue;
		}
		.@percent = .@data[UMOB_HP] / max(1,'ogh_boss_maxhp / 100);
		if ('ogh_phase == 0 && .@percent <= 70) { 'ogh_phase = 1; callsub L_SpawnSlaves,20583; }
		if ('ogh_phase == 1 && .@percent <= 50) { 'ogh_phase = 2; callsub L_SpawnSlaves,20584; }
		if ('ogh_phase == 2 && .@percent <= 30) { 'ogh_phase = 3; callsub L_SpawnSlaves,20585; }
		++'ogh_boss_tick;
		if ('ogh_level >= 3 && 'ogh_boss_tick % 15 == 0) {
			mapannounce 'ogh_map$,"The phantom gathers every wound into Max Pain!",bc_map,0xFF55FF;
			unitskilluseid 'ogh_boss,"NPC_MAXPAIN",min(10,'ogh_level),'ogh_boss,-10000;
		}
		if ('ogh_level >= 9 && 'ogh_boss_tick % 10 == 0) {
			unitskilluseid 'ogh_boss,"NPC_WIDECURSE",5,'ogh_boss,-10000;
			unitskilluseid 'ogh_boss,"NPC_WIDEBLEEDING",5,'ogh_boss,-10000;
		}
	}
	end;

OnBossDead:
	if ('ogh_stage != 3) end;
	'ogh_stage = 4;
	'ogh_boss = 0;
	killmonster 'ogh_map$,"All";
	enablenpc instance_npcname("Temporal Treasure#oghcm");
	enablenpc instance_npcname("Oscar#oghcm_reward");
	enablenpc instance_npcname("Exit Distortion#oghcm");
	mapannounce 'ogh_map$,"The distortion collapses. Claim the Temporal Treasure and report to Oscar.",bc_map,0xFFDD77;
	end;

OnSlaveDead:
	if ('ogh_stage == 3) ++'ogh_slave_kills;
	end;

OnThornDead:
	end;

L_TryCenter:
	if ('ogh_stage == 1 && 'ogh_left_open && 'ogh_right_open && 'ogh_root_done) {
		'ogh_stage = 2;
		enablenpc instance_npcname("Oscar#oghcm_center");
		mapannounce 'ogh_map$,"Both wings and the center are stable. Meet Oscar at the north end of the central hall.",bc_map,0x55FFAA;
	}
	return;

L_SpawnSlaves:
	.@mob = getarg(0);
	.@event$ = instance_npcname(strnpcinfo(0))+"::OnSlaveDead";
	for (.@i = 0; .@i < 34; ++.@i) {
		areamonster 'ogh_map$,128,258,178,294,"Phantom Guard",.@mob,1,.@event$;
		callfunc "OGHCM_ScaleMob",$@mobid[0],'ogh_level;
	}
	mapannounce 'ogh_map$,"A regiment of phantom guards answers the boss!",bc_map,0xFFAA55;
	return;
}
