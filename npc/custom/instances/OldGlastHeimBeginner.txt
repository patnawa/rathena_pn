function	script	OGHB_NextWeeklyReset	{
	// The official reset is Thursday. This returns the next Thursday
	// at 00:00 in the map-server's local timezone.
	.@now = gettimetick(2);
	.@seconds_today = gettime(DT_HOUR) * 3600 + gettime(DT_MINUTE) * 60 + gettime(DT_SECOND);
	.@days = (THURSDAY - gettime(DT_DAYOFWEEK) + 7) % 7;
	if (.@days == 0)
		.@days = 7;
	return .@now - .@seconds_today + .@days * 86400;
}

glast_01,196,235,4	script	Unique Clown#oghb	4_M_BARD,{
	mes "[Unique Clown]";
	mes "I'm studying a small time gap connected to the fall of old Glast Heim. Adventurers can enter it alone or with a party.";
	next;
	if (BaseLevel < 65) {
		mes "[Unique Clown]";
		mes "The distortion rejects anyone below Base Level 65. Come back when you are ready.";
		close;
	}

	.@party_id = getcharid(1);
	.@mode = .@party_id ? IM_PARTY : IM_CHAR;
	.@owner = .@party_id ? .@party_id : getcharid(0);
	.@iid = instance_id(.@mode);
	if (.@iid && instance_live_info(ILI_NAME,.@iid) != "Old Glast Heim (Beginner)") {
		mes "[Unique Clown]";
		mes "You already own another active instance. Close it before generating this time gap.";
		close;
	}

	if (OGHB_Next > gettimetick(2)) {
		.@minutes = (OGHB_Next - gettimetick(2) + 59) / 60;
		mes "[Unique Clown]";
		mes "Your memory is still tied to the previous distortion. The Thursday reset is in about ^FF0000"+.@minutes+" minute(s)^000000.";
		close;
	}

	if (!.@iid) {
		if (.@party_id && !is_party_leader()) {
			mes "[Unique Clown]";
			mes "Your party leader must generate the time gap.";
			close;
		}
		if (select("Generate Time Gap.:Cancel.") == 2)
			close;
		.@iid = instance_create("Old Glast Heim (Beginner)",.@mode,.@owner);
		if (.@iid < 1) {
			mes "[Unique Clown]";
			mes "The time gap could not be generated. Please try again shortly.";
			close;
		}
		mes "[Unique Clown]";
		mes "The time gap is open. Speak to me once more to enter it.";
		close;
	}

	if (select("Enter Old Glast Heim.:Cancel.") == 2)
		close;
	.@result = instance_enter("Old Glast Heim (Beginner)",-1,-1,getcharid(0),.@iid);
	if (.@result != IE_OK) {
		mes "[Unique Clown]";
		mes "The time gap is not ready, or you do not belong to its owner.";
		close;
	}

	// Beginner OGH officially cannot be re-entered after leaving.
	OGHB_Next = callfunc("OGHB_NextWeeklyReset");
	OGHB_Disguised = 1;
	disguise 1019;
	if (isbegin_quest(19990))
		erasequest 19990;
	setquest 19990;
	end;

OnInit:
	questinfo QTYPE_QUEST,QMARK_YELLOW,"BaseLevel >= 65";
	end;
}
