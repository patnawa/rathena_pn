function	script	F_TombRemorseCleanup	{
	setarray .@active[0],16465,16467,16469,16471;
	setarray .@standby[0],16466,16468,16470,16472;
	for (.@i = 0; .@i < 4; ++.@i) {
		if (checkquest(.@standby[.@i],PLAYTIME) == 2) {
			erasequest .@standby[.@i];
			erasequest .@active[.@i];
		}
	}
	if (checkquest(16464,PLAYTIME) == 2)
		erasequest 16464;
	return;
}

function	script	F_TombRemorsePresence	{
	.@count = 'member_count;
	.@party = 'party_id;
	.@map$ = 'map$;
	copyarray .@aid[0],'member_aid[0],.@count;
	copyarray .@cid[0],'member_cid[0],.@count;
	.@caller = getcharid(3);
	.@error$ = "";
	for (.@i = 0; .@i < .@count; ++.@i) {
		if (!isloggedin(.@aid[.@i],.@cid[.@i]) || !attachrid(.@aid[.@i])) {
			.@error$ = "Every starting member must remain online.";
			break;
		}
		if (getcharid(1) != .@party) {
			.@error$ = strcharinfo(0)+" is no longer in the starting party.";
			break;
		}
		if (strcharinfo(3) != .@map$ || Hp <= 0) {
			.@error$ = strcharinfo(0)+" must be alive and inside the Tomb.";
			break;
		}
	}
	attachrid .@caller;
	return .@error$;
}

function	script	F_TombRemorseBossCredit	{
	.@boss = getarg(0);
	.@count = 'member_count;
	.@map$ = 'map$;
	copyarray .@aid[0],'member_aid[0],.@count;
	copyarray .@cid[0],'member_cid[0],.@count;
	.@caller = getcharid(3);
	for (.@i = 0; .@i < .@count; ++.@i) {
		if (!isloggedin(.@aid[.@i],.@cid[.@i]) || !attachrid(.@aid[.@i]))
			continue;
		if (strcharinfo(3) != .@map$)
			continue;
		if (.@boss == 20893 && checkquest(16467) != -1 && checkquest(16468) == -1)
			TombRemorse_TiaraDone = 1;
		if (.@boss == 20892 && checkquest(16465) != -1 && checkquest(16466) == -1)
			TombRemorse_SakrayDone = 1;
	}
	attachrid .@caller;
	return;
}

function	script	F_TombRemorseCollect	{
	.@kind = getarg(0); // 0: memory (3), 1: past (10)
	.@bit = getarg(1);
	mes "[Lingering Fragment]";
	if (instance_live_info(ILI_NAME,instance_id()) != "Tomb of Remorse"
		|| TombRemorse_ActiveInstance != instance_id()) {
		mes "The fragment does not respond to you.";
		close;
	}
	if (.@kind == 0) {
		if (checkquest(16469) == -1 || checkquest(16470) != -1) {
			mes "You have not accepted Fragmented Memory.";
			close;
		}
		.@flag = 1 << .@bit;
		if (TombRemorse_MemoryMask & .@flag) {
			mes "You already recovered this memory today.";
			close;
		}
		TombRemorse_MemoryMask |= .@flag;
		++TombRemorse_MemoryCount;
		mes "A fractured memory settles into place ("+min(TombRemorse_MemoryCount,3)+"/3).";
		if (TombRemorse_MemoryCount >= 3)
			dispbottom "Fragmented Memory is complete. Report to the Marquess.";
		close;
	}
	if (checkquest(16471) == -1 || checkquest(16472) != -1) {
		mes "You have not accepted Fragmented Past.";
		close;
	}
	.@flag = 1 << .@bit;
	if (TombRemorse_PastMask & .@flag) {
		mes "You already recovered this trace of the past today.";
		close;
	}
	TombRemorse_PastMask |= .@flag;
	++TombRemorse_PastCount;
	mes "You preserve a fragment of the past ("+min(TombRemorse_PastCount,10)+"/10).";
	if (TombRemorse_PastCount >= 10)
		dispbottom "Fragmented Past is complete. Report to the Marquess.";
	close;
}

// Client quest navigation points here for access and cooldown information.
dali02,78,116,4	script	Princess Tiara#tor_gate	20893,{
	callfunc "F_TombRemorseCleanup";
	mes "[Princess Tiara]";
	if (BaseLevel < 220) {
		mes "Only adventurers of Base Level 220 or higher may face these memories.";
		close;
	}
	if (!TombRemorse_Access) {
		if (checkquest(16463) == -1) {
			mes "Please begin my story at dali,124,88.";
			close;
		}
		mes "Beyond this point, the past will take physical form. Please release us from it.";
		completequest 16463;
		TombRemorse_Access = 1;
		next;
	}
	.@party = getcharid(1);
	if (!.@party) {
		mes "Form a party first. A one-person party is permitted.";
		close;
	}
	.@iid = instance_id(IM_PARTY);
	if (.@iid && instance_live_info(ILI_NAME,.@iid) != "Tomb of Remorse") {
		mes "Your party already owns another active memorial dungeon.";
		close;
	}
	.@returning = (.@iid > 0 && TombRemorse_ActiveInstance == .@iid
		&& TombRemorse_ActiveUntil > gettimetick(2));
	if (.@iid && getinstancevar('started,.@iid) && !.@returning) {
		mes "The expedition has already begun. Only its snapshotted starting members may enter.";
		close;
	}
	if (!.@returning && checkquest(16464,PLAYTIME) != -1) {
		mes "You have already entered today. The memory resets at 04:00 server time.";
		close;
	}
	if (!.@iid) {
		if (!is_party_leader()) {
			mes "Only the party leader may reserve the Tomb.";
			close;
		}
		if (!instance_check_party(.@party,1,220)) {
			mes "Every online party member must be Base Level 220 or higher.";
			close;
		}
		mes "The Tomb lasts 60 minutes. The daily cooldown is recorded on entry.";
		next;
		if (select("Create and enter.:Cancel.") == 2)
			close;
		.@iid = instance_create("Tomb of Remorse",IM_PARTY,.@party);
		if (.@iid < 1) {
			mes "The Tomb could not be created (error "+.@iid+").";
			close;
		}
	} else {
		.@choice = select("Enter the active Tomb.:Review the rules.:Cancel.");
		if (.@choice == 2) {
			mes "Base Level 220+, party required, once daily at 04:00, with a 60-minute limit.";
			mes "All starting members must be online, alive, and inside before each passage opens.";
			close;
		}
		if (.@choice == 3)
			close;
	}
	.@result = instance_enter("Tomb of Remorse",-1,-1,getcharid(0),.@iid);
	if (.@result != IE_OK) {
		mes "The instance map is not ready (error "+.@result+").";
		close;
	}
	if (!.@returning) {
		setquest 16464;
		TombRemorse_ActiveInstance = .@iid;
		TombRemorse_ActiveUntil = gettimetick(2)+3600;
	}
	end;

OnInit:
	questinfo QTYPE_QUEST,QMARK_YELLOW,"BaseLevel >= 220 && checkquest(16463) != -1 && !TombRemorse_Access";
	end;
}

		close;
	}
	mes "The palace holds four tasks. Their rewards total 20 Mineas each day.";
	next;
	switch(select("Accept available tasks.:Report completed tasks.:Reward information.:Leave.")) {
	case 1:
		.@accepted = 0;
		if (checkquest(16465) == -1 && checkquest(16466) == -1) {
			setquest 16465;
			TombRemorse_SakrayDone = 0;
			++.@accepted;
		}
		if (checkquest(16467) == -1 && checkquest(16468) == -1) {
			setquest 16467;
			TombRemorse_TiaraDone = 0;
			++.@accepted;
		}
		if (checkquest(16469) == -1 && checkquest(16470) == -1) {
			setquest 16469;
			TombRemorse_MemoryCount = 0;
			TombRemorse_MemoryMask = 0;
			++.@accepted;
		}
		if (checkquest(16471) == -1 && checkquest(16472) == -1) {
			setquest 16471;
			TombRemorse_PastCount = 0;
			TombRemorse_PastMask = 0;
			++.@accepted;
		}
		if (.@accepted)
			mes "All "+.@accepted+" available task(s) have been recorded.";
		else
			mes "You already hold every task currently available to you.";
		close;
	case 2:
		.@claims = 0;
		.@eligible[0] = checkquest(16465) != -1 && checkquest(16466) == -1
			&& (TombRemorse_SakrayDone || checkquest(16465,HUNTING) == 2);
		.@eligible[1] = checkquest(16467) != -1 && checkquest(16468) == -1
			&& (TombRemorse_TiaraDone || checkquest(16467,HUNTING) == 2);
		.@eligible[2] = checkquest(16469) != -1 && checkquest(16470) == -1
			&& TombRemorse_MemoryCount >= 3;
		.@eligible[3] = checkquest(16471) != -1 && checkquest(16472) == -1
			&& TombRemorse_PastCount >= 10;
		.@mineas = .@eligible[0] * 10 + .@eligible[1] * 8 + .@eligible[2] + .@eligible[3];
		if (.@mineas && !checkweight(1000367,.@mineas)) {
			mes "Make enough inventory room for all " + .@mineas + " Mineas before filing these reports.";
			close;
		}
		if (.@eligible[0]) {
			completequest 16465;
			getitem 1000367,10; // Mineas
			if (checkquest(16466) == -1) setquest 16466;
			TombRemorse_SakrayDone = 0;
			++.@claims;
		}
		if (.@eligible[1]) {
			completequest 16467;
			getitem 1000367,8;
			if (checkquest(16468) == -1) setquest 16468;
			TombRemorse_TiaraDone = 0;
			++.@claims;
		}
		if (.@eligible[2]) {
			completequest 16469;
			getitem 1000367,1;
			if (checkquest(16470) == -1) setquest 16470;
			TombRemorse_MemoryCount = 0;
			TombRemorse_MemoryMask = 0;
			++.@claims;
		}
		if (.@eligible[3]) {
			completequest 16471;
			getitem 1000367,1;
			if (checkquest(16472) == -1) setquest 16472;
			TombRemorse_PastCount = 0;
			TombRemorse_PastMask = 0;
			++.@claims;
		}
		if (.@claims)
			mes "Your completed reports have been exchanged for Mineas.";
		else
			mes "None of your active reports is complete yet.";
		close;
	case 3:
		mes "Broken Bond: defeat Cursed Sakray, ^0000FF10 Mineas^000000.";
		mes "Shedding Tears: defeat Princess Tiara, ^0000FF8 Mineas^000000.";
		mes "Fragmented Memory: recover 3 memories, ^0000FF1 Mineas^000000.";
		mes "Fragmented Past: preserve 10 fragments, ^0000FF1 Mineas^000000.";
		next;
		mes "The official notices also describe a random Mineas boss drop. Its rate is unpublished, so this clean-room release does not guess one.";
		close;
	case 4:
		close;
	}

OnInit:
	questinfo QTYPE_DAILYQUEST,QMARK_YELLOW,"TombRemorse_Access && BaseLevel >= 220";
	end;
}
