function	script	F_AirshipCrashCleanup	{
	if (AirshipCrash_Next && AirshipCrash_Next <= gettimetick(2)) {
		AirshipCrash_Next = 0;
		AirshipCrash_ActiveInstance = 0;
		AirshipCrash_ActiveUntil = 0;
		if (checkquest(5909) != -1)
			erasequest 5909;
	}
	return;
}

// Return the third future 04:00 boundary: the next 04:00 plus two days.
function	script	F_AirshipCrashNextReset	{
	.@seconds = (4-gettime(DT_HOUR))*3600-gettime(DT_MINUTE)*60-gettime(DT_SECOND);
	if (.@seconds <= 0)
		.@seconds += 86400;
	return gettimetick(2)+.@seconds+172800;
}

// Strict transition/reward check for the party snapshot made by Ferlock.
// Returns an empty string on success or a user-facing failure reason.
function	script	F_AirshipCrashPresence	{
	.@map$ = getarg(0);
	.@count = 'member_count;
	.@party = 'party_id;
	copyarray .@aid[0],'member_aid[0],.@count;
	copyarray .@cid[0],'member_cid[0],.@count;
	.@caller = getcharid(3);
	.@error$ = "";
	for (.@i = 0; .@i < .@count; ++.@i) {
		detachrid;
		if (!isloggedin(.@aid[.@i],.@cid[.@i]) || !attachrid(.@aid[.@i])) {
			.@error$ = "Every registered member must remain online.";
			break;
		}
		if (getcharid(1) != .@party) {
			.@error$ = strcharinfo(0)+" is no longer in the registered party.";
			break;
		}
		if (strcharinfo(3) != .@map$ || Hp <= 0) {
			.@error$ = strcharinfo(0)+" must be alive and on the same instance map.";
			break;
		}
	}
	detachrid;
	if (!attachrid(.@caller))
		return "The requesting member disconnected.";
	if (.@error$ == "" && getmapusers(.@map$) != .@count)
		.@error$ = "The number inside no longer matches the registered party count.";
	return .@error$;
}

function	script	F_AirshipCrashBriefed	{
	.@count = 'member_count;
	copyarray .@aid[0],'member_aid[0],.@count;
	copyarray .@cid[0],'member_cid[0],.@count;
	.@caller = getcharid(3);
	.@ready = 1;
	for (.@i = 0; .@i < .@count; ++.@i) {
		detachrid;
		if (!isloggedin(.@aid[.@i],.@cid[.@i]) || !attachrid(.@aid[.@i]) || checkquest(5911) != 2) {
			.@ready = 0;
			break;
		}
	}
	detachrid;
	if (!attachrid(.@caller))
		return 0;
	return .@ready;
}

function	script	F_AirshipCrashMissionStart	{
	.@quest = getarg(0);
	.@count = getarg(1);
	.@members = 'member_count;
	copyarray .@aid[0],'member_aid[0],.@members;
	copyarray .@cid[0],'member_cid[0],.@members;
	.@caller = getcharid(3);
	// Validate every attachment before modifying any member's quest state.
	for (.@i = 0; .@i < .@members; ++.@i) {
		detachrid;
		if (!isloggedin(.@aid[.@i],.@cid[.@i]) || !attachrid(.@aid[.@i])) {
			detachrid;
			attachrid .@caller;
			return 0;
		}
	}
	for (.@i = 0; .@i < .@members; ++.@i) {
		detachrid;
		if (!attachrid(.@aid[.@i])) {
			detachrid;
			attachrid .@caller;
			return 0;
		}
		for (.@q = 5912; .@q <= 5914; ++.@q)
			if (checkquest(.@q) != -1)
				erasequest .@q;
		setquest .@quest;
	}
	detachrid;
	if (!attachrid(.@caller))
		return 0;
	'mission_quest = .@quest;
	'mission_remaining = .@count;
	'mission_active = 1;
	'mission_ready = 0;
	.@each = .@count/5;
	.@extra = .@count-(.@each*5);
	.@label$ = instance_npcname("#AirshipCrashControl")+"::OnMissionMobDead";
	areamonster 'cave_map$,20,20,380,380,"Rotten Tree",20886,.@each+.@extra,.@label$;
	areamonster 'cave_map$,20,20,380,380,"Melted Poring",20887,.@each,.@label$;
	areamonster 'cave_map$,20,20,380,380,"Grave Worm",20888,.@each,.@label$;
	areamonster 'cave_map$,20,20,380,380,"Brain Sucker",20889,.@each,.@label$;
	areamonster 'cave_map$,20,20,380,380,"Mosquilo",20890,.@each,.@label$;
	instance_announce 0,"Airship Crash: joint research began - hunt "+.@count+" local monsters.",bc_map,0x88CCFF;
	return 1;
}

function	script	F_AirshipCrashMissionReward	{
	.@quest = 'mission_quest;
	.@tier = .@quest-5911;
	setarray .@base[1],10000000,20000000,30000000;
	setarray .@job[1],7500000,15000000,22500000;
	.@count = 'member_count;
	copyarray .@aid[0],'member_aid[0],.@count;
	copyarray .@cid[0],'member_cid[0],.@count;
	.@caller = getcharid(3);
	for (.@i = 0; .@i < .@count; ++.@i) {
		detachrid;
		if (!isloggedin(.@aid[.@i],.@cid[.@i]) || !attachrid(.@aid[.@i])) {
			detachrid;
			attachrid .@caller;
			return 0;
		}
	}
	for (.@i = 0; .@i < .@count; ++.@i) {
		detachrid;
		if (!attachrid(.@aid[.@i])) {
			detachrid;
			attachrid .@caller;
			return 0;
		}
		completequest .@quest;
		getexp .@base[.@tier],.@job[.@tier];
		erasequest .@quest;
	}
	detachrid;
	if (!attachrid(.@caller))
		return 0;
	return 1;
}

// Exact current client exchange costs.  Rusbald calls this with one output.
function	script	F_AirshipCrashExchange	{
	.@item = getarg(0);
	.@stones = getarg(1);
	.@extra = getarg(2,0);
	.@extra_amount = getarg(3,0);
	.@zeny = getarg(4,0);
	if (countitem(1000363) < .@stones) {
		mes "You need "+.@stones+" Magical Stones.";
		return 0;
	}
	if (.@extra && countitem(.@extra) < .@extra_amount) {
		mes "You also need "+.@extra_amount+" "+getitemname(.@extra)+".";
		return 0;
	}
	if (Zeny < .@zeny) {
		mes "You also need "+.@zeny+" Zeny.";
		return 0;
	}
	if (!checkweight(.@item,1)) {
		mes "You cannot carry the reward.";
		return 0;
	}
	delitem 1000363,.@stones;
	if (.@extra)
		delitem .@extra,.@extra_amount;
	if (.@zeny)
		Zeny -= .@zeny;
	getitem .@item,1;
	mes "Exchange complete: "+getitemname(.@item)+".";
	return 1;
}

// Boss-death labels cannot wait for inventory management. If the slayer cannot
// carry the antiquity, send exactly one copy through the in-game mail system.
function	script	F_AirshipCrashBossReward	{
	if (checkweight(102573,1)) {
		getitem 102573,1;
		return 1;
	}
	setarray .@mail_item[0],102573;
	setarray .@mail_amount[0],1;
	mail getcharid(0),"Dr. Strong","Airship Crash Reward","Your inventory was full when the boss fell, so the crash-site antiquity was delivered by mail.",0,.@mail_item,.@mail_amount;
	dispbottom "Your inventory was full. The crash-site antiquity was delivered by mail.";
	return 1;
}
