function	script	F_PNHoL_ProtectionLevel	{
	if (getequipid(EQI_HEAD_LOW) != 420231)
		return 0;

	.@card = getequipcardid(EQI_HEAD_LOW, 3);
	if (.@card < 312453 || .@card > 312472)
		return 0;
	return .@card - 312452;
}

// Return the next Friday 09:45 in local server time.
function	script	F_PNHoL_NextReset	{
	.@now = gettimetick(2);
	.@today_seconds = gettime(DT_HOUR) * 3600 + gettime(DT_MINUTE) * 60 + gettime(DT_SECOND);
	.@reset_seconds = 9 * 3600 + 45 * 60;
	.@days = (FRIDAY - gettime(DT_DAYOFWEEK) + 7) % 7;
	.@delta = .@days * 86400 + .@reset_seconds - .@today_seconds;
	if (.@delta <= 0)
		.@delta += 604800;
	return .@now + .@delta;
}

// The barrier rank is account data.  The enchant on the equipped
// protection is its visible/client representation and is repaired here.
function	script	F_PNHoL_SyncProtection	{
	.@item_level = callfunc("F_PNHoL_ProtectionLevel");
	if (!.@item_level)
		return 0;

	if (#PNHoL_Level < 1 || #PNHoL_Level > 20)
		#PNHoL_Level = .@item_level;
	if (.@item_level == #PNHoL_Level)
		return #PNHoL_Level;

	if (getequipuniqueid(EQI_HEAD_LOW)) {
		dispbottom "Hall of Life: unique-ID protections cannot be synchronized safely.";
		return 0;
	}
	getinventorylist;
	.@idx = -1;
	for (.@i = 0; .@i < @inventorylist_count; ++.@i) {
		if ((@inventorylist_equip[.@i] & EQP_HEAD_LOW) && @inventorylist_id[.@i] == 420231) {
			.@idx = .@i;
			break;
		}
	}
	if (.@idx < 0 || @inventorylist_expire[.@idx]) {
		dispbottom "Hall of Life: rental or unlocatable protections cannot be synchronized safely.";
		return 0;
	}
	setarray .@card[0],@inventorylist_card1[.@idx],@inventorylist_card2[.@idx],@inventorylist_card3[.@idx],@inventorylist_card4[.@idx];
	setarray .@oid[0],@inventorylist_option_id1[.@idx],@inventorylist_option_id2[.@idx],@inventorylist_option_id3[.@idx],@inventorylist_option_id4[.@idx],@inventorylist_option_id5[.@idx];
	setarray .@oval[0],@inventorylist_option_value1[.@idx],@inventorylist_option_value2[.@idx],@inventorylist_option_value3[.@idx],@inventorylist_option_value4[.@idx],@inventorylist_option_value5[.@idx];
	setarray .@opar[0],@inventorylist_option_parameter1[.@idx],@inventorylist_option_parameter2[.@idx],@inventorylist_option_parameter3[.@idx],@inventorylist_option_parameter4[.@idx],@inventorylist_option_parameter5[.@idx];
	.@card[3] = 312452 + #PNHoL_Level;
	.@identify = @inventorylist_identify[.@idx];
	.@refine = @inventorylist_refine[.@idx];
	.@attribute = @inventorylist_attribute[.@idx];
	.@grade = @inventorylist_enchantgrade[.@idx];
	.@bound = @inventorylist_bound[.@idx];
	delequip EQI_HEAD_LOW;
	if (!delitemidx(@inventorylist_idx[.@idx],1))
		return 0;
	if (.@bound)
		getitembound4 420231,1,.@identify,.@refine,.@attribute,.@card[0],.@card[1],.@card[2],.@card[3],.@bound,.@grade,.@oid,.@oval,.@opar;
	else
		getitem4 420231,1,.@identify,.@refine,.@attribute,.@card[0],.@card[1],.@card[2],.@card[3],.@grade,.@oid,.@oval,.@opar;
	// `equip` selects the first copy by item ID. Only auto-equip when this is the
	// sole protection; otherwise let the player select the newly ranked copy.
	if (countitem(420231) != 1) {
		dispbottom "Hall of Life: the synchronized protection was returned unequipped. Equip the correctly ranked copy and try again.";
		return 0;
	}
	equip 420231;
	return #PNHoL_Level;
}

// Apply an unfinished-run penalty once the referenced instance no
// longer exists.  Reward-locked accounts cannot gain or lose levels.
function	script	F_PNHoL_CheckFailure	{
	if (#PNHoL_PendingIID < 1)
		return 0;
	if (#PNHoL_RewardUntil > gettimetick(2)) {
		#PNHoL_PendingIID = 0;
		#PNHoL_PendingLevel = 0;
		if (isbegin_quest(12617) > 0)
			erasequest 12617;
		return 0;
	}
	if (instance_live_info(ILI_NAME, #PNHoL_PendingIID) == "Hall of Life")
		return 0;

	if (#PNHoL_Level > 1)
		--#PNHoL_Level;
	#PNHoL_PendingIID = 0;
	#PNHoL_PendingLevel = 0;
	if (isbegin_quest(12617) > 0)
		erasequest 12617;
	if (callfunc("F_PNHoL_ProtectionLevel"))
		callfunc("F_PNHoL_SyncProtection");
	return 1;
}

// Exact Hall of Life table from the current MuhRO raw wiki page.
// Returns 0 without granting anything when inventory capacity is low.
function	script	F_PNHoL_GrantReward	{
	.@level = getarg(0);
	if (.@level < 1 || .@level > 20)
		return 0;

	setarray .@circulation[1], 15,23,32,42,52,62,72,83,98,118,139,161,183,205,227,250,273,297,321,346;
	setarray .@birth[1],        0,8,15,20,26,31,37,43,49,55,62,69,77,86,96,106,118,129,142,158;
	setarray .@extinction[1],   0,8,15,20,26,31,37,43,49,55,62,69,77,86,96,106,118,129,142,158;
	setarray .@truth[1],        0,0,0,0,0,3,6,21,26,28,31,34,38,42,46,49,53,59,64,70;
	setarray .@light[1],        0,0,0,0,0,0,0,5,8,9,11,12,13,14,16,17,18,20,22,26;
	setarray .@genesis[1],      0,0,0,0,0,0,0,0,3,5,7,10,11,12,13,15,16,17,19,22;
	setarray .@rune[1],         0,0,0,0,0,0,0,0,5,6,7,8,10,11,12,13,15,16,17,20;

	setarray .@all_item[0], 1001456,1001457,1001458,1001459,1001460,1001601,1001593;
	setarray .@all_amount[0], .@circulation[.@level],.@birth[.@level],.@extinction[.@level],.@truth[.@level],.@light[.@level],.@genesis[.@level],.@rune[.@level];
	for (.@i = 0; .@i < getarraysize(.@all_item); ++.@i) {
		if (.@all_amount[.@i] < 1)
			continue;
		.@give_item[.@count] = .@all_item[.@i];
		.@give_amount[.@count] = .@all_amount[.@i];
		++.@count;
	}
	if (!checkweight2(.@give_item, .@give_amount))
		return 0;

	for (.@i = 0; .@i < .@count; ++.@i)
		getitem .@give_item[.@i], .@give_amount[.@i];
	return 1;
}

	}

	mes "Re-entry delay: ^0000FF1 minute^000000. Reward reset: ^0000FFFriday 09:45^000000.";
	next;

		close;

		mes "Equip a valid " + mesitemlink(420231) + " in the lower-headgear slot.";
		mes "The Sealing Spirit Stone at 166,235 can reissue a lost protection.";
		close;

	.@iid = instance_id(IM_PARTY);
	if (#PNHoL_PendingIID > 0 && #PNHoL_PendingIID != .@iid && #PNHoL_RewardUntil <= .@now) {
		mes "[Hall of Life]";
		mes "This account is still registered in another active Hall of Life correction.";
		mes "Finish that correction or wait for its 60-minute lifetime to end.";
		close;
	}
	if (.@iid && instance_live_info(ILI_NAME, .@iid) == .@instance$) {
		switch(select("Enter the reserved Hall.:Review barrier status.:Cancel.")) {
		case 1:
			if (PNHoL_NextEntry > .@now) {
				mes "[Hall of Life]";
				mes "The dimensional after-effect remains for " + (PNHoL_NextEntry - .@now) + " more second(s).";
				close;
			}
			if (instance_enter(.@instance$) != IE_OK) {
				mes "[Hall of Life]";
				mes "The reserved passage did not respond. Please try again.";
				close;

			PNHoL_NextEntry = .@now + 60;

			if (isbegin_quest(12619) > 0)
				erasequest 12619;
			setquest 12619;
			if (#PNHoL_RewardUntil <= .@now) {
				#PNHoL_PendingIID = .@iid;
				#PNHoL_PendingLevel = .@protection_level;
				if (isbegin_quest(12617) > 0)
					erasequest 12617;
				setquest 12617;
			}
			end;
		case 2:
			mes "[Hall of Life]";
			mes "Account barrier: ^0000FF" + #PNHoL_Level + "^000000.";
			if (#PNHoL_RewardUntil > .@now)
				mes "This account has claimed its weekly reward. Progress is locked until the Friday reset.";
			else
				mes "No weekly reward lock is active.";
			close;
		case 3:
			close;
		}
	}
	if (.@iid) {
		mes "[Hall of Life]";
		mes "Your party already owns another memorial dungeon: ^0000FF" + instance_live_info(ILI_NAME, .@iid) + "^000000.";
		close;
	}
	if (!is_party_leader()) {
		mes "[Hall of Life]";
		mes "Only the party leader may create the prison. Once it is reserved, every eligible member may enter.";
		close;
	}

		mes "[Hall of Life]";
		mes "The party leader needs one " + mesitemlink(1001415) + ".";
		close;
	}
	if (select("Use one Hall Key and reserve.:Cancel.") == 2)
		close;

	.@created = instance_create(.@instance$);
	if (.@created < 1) {
		mes "[Hall of Life]";
		mes "Reservation failed (code " + .@created + ").";
		close;
	}
	setinstancevar 'party_id, getcharid(1), .@created;
	setinstancevar 'leader_aid, getcharid(3), .@created;
	delitem 1001415, 1;
	mes "[Hall of Life]";
	mes "The Hall is reserved for 60 minutes. Speak to the gate again to enter.";
	close;

	initnpctimer;
	end;

		disablenpc "Dimension Prison#2";
	if (getnpcid(0,"Dimensional Prison#2"))
		disablenpc "Dimensional Prison#2";
	end;
}

		close;

		mes "[Rigel]";

		close;

		close;

	'started = 1;
	'party_id = getcharid(1);
	deletearray 'member_aid;
	deletearray 'member_cid;
	getpartymember 'party_id, 1;
	.@count = $@partymembercount;
	copyarray .@cid[0], $@partymembercid[0], .@count;
	getpartymember 'party_id, 2;
	copyarray .@aid[0], $@partymemberaid[0], .@count;
	.@caller_aid = getcharid(3);
	for (.@i = 0; .@i < .@count; ++.@i) {
		if (!isloggedin(.@aid[.@i], .@cid[.@i]) || !attachrid(.@aid[.@i]))
			continue;
		if (strcharinfo(3) == 'map$) {
			'member_aid['member_count] = .@aid[.@i];
			'member_cid['member_count] = .@cid[.@i];
			++'member_count;
		}
		detachrid;
	}
	attachrid .@caller_aid;

	disablenpc instance_npcname(strnpcinfo(0));
	donpcevent instance_npcname("#pn_hol_control") + "::OnStartPhase1";
	end;
}

	'phantom_entries = 0;

// One-second state loop: member/death lock, regeneration, and sanctuary.
1@ba_go,1,1,0	script	#pn_hol_pulse	-1,{
	end;

OnTimer1000:
	stopnpctimer;
	if (('stage != 1 && 'stage != 2) || !unitexists('boss_gid))
		end;
	++'fight_seconds;

	.@blocked = 0;
	for (.@i = 0; .@i < 'member_count; ++.@i) {
		if (!isloggedin('member_aid[.@i], 'member_cid[.@i]) || !attachrid('member_aid[.@i])) {
			.@blocked = 1;
			continue;
		}
		.@dead = (Hp < 1);
		if (strcharinfo(3) != 'map$ || .@dead)
			.@blocked = 1;
		if (.@dead && !'member_dead[.@i]) {
			'member_dead[.@i] = 1;
			++'death_count;
			'regen_bonus_pct += 3;
			if ('stage == 1 && 'dimension) {
				areamonster 'map$,333,30,371,68,"Rigel's Illusion",22176,1,instance_npcname("#pn_hol_control") + "::OnIllusionDead";
				hateffect HAT_EF_WHITEBODY, true, $@mobid[0];
			}
			instance_announce instance_id(), "A death strengthened Rigel and created another phantom.", BC_MAP, 0xFF6666;
		}
		else if (!.@dead)
			'member_dead[.@i] = 0;
		detachrid;
	}
	if (.@blocked != 'member_block) {
		'member_block = .@blocked;
		if (.@blocked)
			instance_announce instance_id(), "Rigel is invulnerable while a registered member is absent or dead.", BC_MAP, 0xFF6666;
		else
			instance_announce instance_id(), "Every registered member is present and alive; Rigel is vulnerable again.", BC_MAP, 0x66FF66;
	}

	if ('stage == 2 && 'sanctuary) {
		getunitdata 'boss_gid, .@data;
		if (.@data[UMOB_X] < 'sanctuary_x - 6 || .@data[UMOB_X] > 'sanctuary_x + 6 || .@data[UMOB_Y] < 'sanctuary_y - 6 || .@data[UMOB_Y] > 'sanctuary_y + 6)
			donpcevent instance_npcname("#pn_hol_control") + "::OnBreakSanctuary";
	}
	if ('groggy)
		setunitdata 'boss_gid, UMOB_DMGIMMUNE, false;
	else
		setunitdata 'boss_gid, UMOB_DMGIMMUNE, ('member_block || 'sanctuary);

	if (!'groggy && 'fight_seconds % 5 == 0) {
		getunitdata 'boss_gid, .@data;
		.@pct = 'heal_pct + 'regen_bonus_pct;
		.@hp = .@data[UMOB_HP] + 'base_hp * .@pct / 100;
		setunitdata 'boss_gid, UMOB_HP, min(.@hp, 'base_hp);
	}
	if ('stage == 2 && !'sanctuary && !'groggy && 'fight_seconds >= 'next_sanctuary) {
		'next_sanctuary = 'fight_seconds + 'sanctuary_interval;
		donpcevent instance_npcname("#pn_hol_control") + "::OnOpenSanctuary";
	}
	if ('stage == 2 && !'next_sanctuary)
		'next_sanctuary = 'sanctuary_interval;

	setnpctimer 0;
	initnpctimer;
	end;
}
