function	script	FOGH_IsEquipment	{
	.@id = getarg(0);
	if (.@id == 15388 || .@id == 15389)
		return 1;
	return .@id >= 32228 && .@id <= 32233;
}

// Recreates the selected equipped artifact while preserving its physical
// card, refine, grade, identify/broken state, binding, and random options.
function	script	FOGH_RebuildEquipment	{
	.@part = getarg(0);
	.@new1 = getarg(1);
	.@new2 = getarg(2);
	.@new3 = getarg(3);
	switch (.@part) {
	case EQI_ARMOR:
		.@mask = EQP_ARMOR;
		break;
	case EQI_GARMENT:
		.@mask = EQP_GARMENT;
		break;
	case EQI_ACC_L:
		.@mask = EQP_ACC_L;
		break;
	default:
		return 0;
	}
	getinventorylist;
	.@idx = -1;
	for (.@i = 0; .@i < @inventorylist_count; ++.@i) {
		if ((@inventorylist_equip[.@i] & .@mask) && callfunc("FOGH_IsEquipment",@inventorylist_id[.@i])) {
			.@idx = .@i;
			break;
		}
	}
	if (.@idx < 0)
		return 0;
	setarray .@oid[0],@inventorylist_option_id1[.@idx],@inventorylist_option_id2[.@idx],@inventorylist_option_id3[.@idx],@inventorylist_option_id4[.@idx],@inventorylist_option_id5[.@idx];
	setarray .@oval[0],@inventorylist_option_value1[.@idx],@inventorylist_option_value2[.@idx],@inventorylist_option_value3[.@idx],@inventorylist_option_value4[.@idx],@inventorylist_option_value5[.@idx];
	setarray .@opar[0],@inventorylist_option_parameter1[.@idx],@inventorylist_option_parameter2[.@idx],@inventorylist_option_parameter3[.@idx],@inventorylist_option_parameter4[.@idx],@inventorylist_option_parameter5[.@idx];
	.@id = @inventorylist_id[.@idx];
	.@identify = @inventorylist_identify[.@idx];
	.@refine = @inventorylist_refine[.@idx];
	.@attribute = @inventorylist_attribute[.@idx];
	.@grade = @inventorylist_enchantgrade[.@idx];
	.@bound = @inventorylist_bound[.@idx];
	.@card0 = @inventorylist_card1[.@idx];
	delitemidx @inventorylist_idx[.@idx],1;
	if (.@bound)
		getitembound4 .@id,1,.@identify,.@refine,.@attribute,.@card0,.@new1,.@new2,.@new3,.@bound,.@grade,.@oid,.@oval,.@opar;
	else
		getitem4 .@id,1,.@identify,.@refine,.@attribute,.@card0,.@new1,.@new2,.@new3,.@grade,.@oid,.@oval,.@opar;
	return 1;
}

	end;
}
