// EDDA Bioresearch Laboratory. Original encounter dialogue and wave tuning.
// Both modes share admission quest 16390; reconnects use this live instance's CID roster.
function	script	F_BioReady	{
	.@party = getarg(0);
	.@map$ = getarg(1);
	.@radius = getarg(2,0);
	.@count = getpartymember(.@party,1,.@cid);
	getpartymember .@party,2,.@aid;
	if (!.@count) return 0;
	for (.@i = 0; .@i < .@count; ++.@i) {
		if (!isloggedin(.@aid[.@i],.@cid[.@i])) {
			if ('bio_started_count) continue;
			return 0;
		}
		++.@online;
		if ('bio_started_count) {
			if (!'bio_started_member[.@cid[.@i]]) return 0;
		}
		if (!'bio_admitted[.@cid[.@i]]) return 0;
		if (readparam(Hp,.@cid[.@i]) <= 0) return 0;
		getmapxy .@where$,.@x,.@y,BL_PC,rid2name(.@aid[.@i]);
		if (.@where$ != .@map$) return 0;
		if (.@radius) {
			if (distance(.@x,.@y,118,22) > .@radius) return 0;
		}
	}
	return .@online > 0;
}

function	script	F_BioMove	{
	.@map$ = getarg(0); .@x = getarg(1); .@y = getarg(2);
	for (.@i = 0; .@i < getarraysize('bio_roster); ++.@i) {
		.@cid = 'bio_roster[.@i]; .@aid = 'bio_accounts[.@i];
		if (!isloggedin(.@aid,.@cid)) continue;
		if (getcharid(1,rid2name(.@aid)) != 'bio_party) continue;
		getmapxy .@where$,.@px,.@py,BL_PC,rid2name(.@aid);
		if (.@where$ != instance_mapname("1@gol1") && .@where$ != instance_mapname("1@gol2")) continue;
		warp .@map$,.@x,.@y,.@cid;
	}
	return;
}

function	script	F_BioBarrier	{
	.@wave = getarg(0); .@closed = getarg(1);
	setarray .@x[0],110,119,105,126;
	setarray .@y[0],18,21,69,72;
	.@cx = .@x[.@wave]; .@cy = .@y[.@wave];
	.@map$ = instance_mapname("1@gol1");
	// Each 7x7 perimeter is natively walkable; restore only those same cells.
	setcell .@map$,.@cx-3,.@cy-3,.@cx+3,.@cy-3,CELL_WALKABLE,!.@closed;
	setcell .@map$,.@cx-3,.@cy+3,.@cx+3,.@cy+3,CELL_WALKABLE,!.@closed;
	setcell .@map$,.@cx-3,.@cy-2,.@cx-3,.@cy+2,CELL_WALKABLE,!.@closed;
	setcell .@map$,.@cx+3,.@cy-2,.@cx+3,.@cy+2,CELL_WALKABLE,!.@closed;
	for (.@i = 0; .@i < 4; ++.@i) {
		.@npc$ = instance_npcname("Containment Flame#bio"+.@wave+"_"+.@i);
		if (.@closed) enablenpc .@npc$;
		else disablenpc .@npc$;
	}
	return;
}

		setinstancevar 'bio_party,.@party,.@iid;
	} else {
		if (select("Enter the laboratory.:Leave.") == 2) close;

	// Revalidate after menu suspension; another member may have started meanwhile.
	if (getcharid(1) != .@party || instance_id(IM_PARTY) != .@iid) close;
	if (instance_live_info(ILI_NAME,.@iid) != "Bioresearch Laboratory") close;
	if (getinstancevar('bio_party,.@iid) != .@party) close;
	// Numeric IDs may be reused while the menu is open; read live admission again.
	.@returning = getinstancevar('bio_admitted[getcharid(0)],.@iid);
	if (!.@returning) {
		if (checkquest(16390,PLAYTIME) == 2) erasequest 16390;
		if (isbegin_quest(16390)) { mes "Your daily admission is already used."; close; }
	}
	.@zone = getinstancevar('bio_zone,.@iid);
	if (!.@returning && .@zone) { mes "This expedition has already started."; close; }
	if (.@zone) {
		if (!getinstancevar('bio_started_member[getcharid(0)],.@iid)) { mes "You were not part of this expedition's starting roster."; close; }
	}
	if (.@zone >= 9 || getinstancevar('bio_boss_started,.@iid)) {
		if (!getinstancevar('bio_eligible[getcharid(0)],.@iid)) { mes "Only the locked encounter roster may return now."; close; }
	}
	if (instance_enter("Bioresearch Laboratory",-1,-1,getcharid(0),.@iid) != IE_OK) { mes "The laboratory map is not ready."; close; }
	if (!.@returning) {
		setinstancevar 'bio_admitted[getcharid(0)],1,.@iid;
		if (!isbegin_quest(16388)) setquest 16388;
		if (isbegin_quest(16392)) erasequest 16392;
		if (isbegin_quest(16399)) erasequest 16399;
		if (isbegin_quest(16400)) erasequest 16400;
		setquest 16390;
	}
	if (.@zone >= 8) warp instance_mapname("1@gol2",.@iid),50,12;
	else if (.@zone) warp instance_mapname("1@gol1",.@iid),getinstancevar('bio_x,.@iid),getinstancevar('bio_y,.@iid);
	end;
OnInit:
	questinfo QTYPE_QUEST,QMARK_YELLOW,"BaseLevel >= 170";
	end;
