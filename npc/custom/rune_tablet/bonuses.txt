// Stable client set/piece IDs; original implementation of published numeric rules.
// Hook is unique, survives death/dispel, and is removed on logout then restored.
// Recalculation reads persistent state, so no effects accumulate across refreshes.
function	script	PN_RT_Refresh	{
	// Keep milliseconds within bonus_script's uint32 duration. Renew daily.
	bonus_script "{ callfunc \"PN_RT_Bonus\"; }",604800,1544;
	recalculatestat;
	return;
}

-	script	PN Rune Bonus Lifecycle	-1,{
OnPCLoginEvent:
OnRenew:
	callfunc "PN_RT_Refresh";
	addtimer 86400000,"PN Rune Bonus Lifecycle::OnRenew";
	end;
}
