//===== rAthena Script =======================================
//= Episode 20 - four side stories and nine daily categories
//===== Description: =========================================
//= Uses the official 2026 client quest IDs. Daily rewards are
//= committed only after inventory checks and reset at 04:00.
//============================================================

function	script	EP20_DailyCanAccept	{
	.@qid = getarg(0);
	.@next = getd("EP20_DailyNext_"+.@qid);
	if (.@next > gettimetick(2))
		return false;
	return isbegin_quest(.@qid) != 1;
}

function	script	EP20_DailyStart	{
	.@qid = getarg(0);
	if (!callfunc("EP20_DailyCanAccept",.@qid))
		return false;
	if (isbegin_quest(.@qid) == 2)
		erasequest .@qid;
	if (isbegin_quest(.@qid) == 0)
		setquest .@qid;
	return true;
}

function	script	EP20_DailyFinish	{
	.@qid = getarg(0);
	.@base_whiskers = getarg(1,3);
	.@reputation = getarg(2,5);
	.@base_exp = getarg(3,13500000);
	.@job_exp = getarg(4,9500000);
	.@reward = callfunc("EP20_DailyWhiskers",.@base_whiskers);
	if (!callfunc("EP20_CheckReward",.@reward))
		return false;
	completequest .@qid;
	setd("EP20_DailyNext_"+.@qid,callfunc("EP20_NextReset"));
	callfunc "EP20_Reward",.@reward,.@reputation,.@base_exp,.@job_exp;
	return true;
}

function	script	EP20_RecordWorkAssignment	{
	.@index = getarg(0);
	.@name$ = getarg(1);
	if (isbegin_quest(8853) != 1) {
		message strcharinfo(0),"No intermediate Rgan is waiting for this assignment.";
		return false;
	}
	if (!callfunc("EP20_RequireRganDisguise"))
		return false;
	.@bit = 1 << .@index;
	if (EP20_WorkMask & .@bit) {
		message strcharinfo(0),.@name$ + " is already assigned for today.";
		return false;
	}
	EP20_WorkMask |= .@bit;
	message strcharinfo(0),"You assign an intermediate Rgan to " + .@name$ + ".";
	return true;
}
