function	script	EP20_QuestInRange	{
	for (.@qid = getarg(0); .@qid <= getarg(1); ++.@qid) {
		if (isbegin_quest(.@qid))
			return true;
	}
	return false;
}

function	script	EP20_MainComplete	{
	return EP20_Main_Complete || isbegin_quest(18233) == 2;
}

function	script	EP20_EffectiveStep	{
	.@step = EP20_Step;
	if (callfunc("EP20_MainComplete")) return 11;
	if (isbegin_quest(18232) || isbegin_quest(18233)) return max(.@step,10);
	if (callfunc("EP20_QuestInRange",18220,18231)) return max(.@step,9);
	if (callfunc("EP20_QuestInRange",23112,23118)) return max(.@step,8);
	if (callfunc("EP20_QuestInRange",23100,23111)) return max(.@step,7);
	if (callfunc("EP20_QuestInRange",17702,17709)) return max(.@step,6);
	if (callfunc("EP20_QuestInRange",17696,17701)) return max(.@step,5);
	if (isbegin_quest(16704) || callfunc("EP20_QuestInRange",17690,17695)) return max(.@step,4);
	if (callfunc("EP20_QuestInRange",16701,16703)) return max(.@step,3);
	if (callfunc("EP20_QuestInRange",16695,16700)) return max(.@step,2);
	if (callfunc("EP20_QuestInRange",16691,16694)) return max(.@step,1);
	return .@step;
}

function	script	EP20_SyncStep	{
	.@step = callfunc("EP20_EffectiveStep");
	if (EP20_Step < .@step)
		EP20_Step = .@step;
	return .@step;
}

function	script	EP20_Prerequisite	{
	// Do not strand characters that began Episode 20 before this gate was
	// installed. New starts require every prerequisite listed by MuhRO.
	if (callfunc("EP20_EffectiveStep") > 0)
		return true;
	return BaseLevel >= 215 && isbegin_quest(17649) == 2 &&
		isbegin_quest(17650) == 2 && ep19_side_saint >= 20 && ep19_side_sea >= 17;
}

function	script	EP20_RequireRganDisguise	{
	if (getstatus(SC_MONSTER_TRANSFORM,1) == 21530)
		return true;
	message strcharinfo(0),"Use a Rgan Transformation Scroll first; Lunch in the Frozen Scale fields can supply another.";
	return false;
}

function	script	EP20_WarperAccess	{
	.@map$ = getarg(0);
	.@step = callfunc("EP20_EffectiveStep");
	if (callfunc("EP20_MainComplete")) .@step = 11;
	.@need = 0;
	// Ice Castle and its original interior are Episode 19 public hubs.
	if (.@map$ == "icecastle" || .@map$ == "icas_in")
		return true;
	if (.@map$ == "jor_back4" || .@map$ == "jor_back5" || .@map$ == "jor_back6" || .@map$ == "jor_safty1")
		.@need = 1;
	else if (.@map$ == "jor_root1" || .@map$ == "jor_root2")
		.@need = 4;
	else if (.@map$ == "icas_in2" || .@map$ == "jor_safty2" || .@map$ == "jor_sanct")
		.@need = 5;
	else if (.@map$ == "jor_maze")
		.@need = 6;
	else if (.@map$ == "jor_twice")
		.@need = 7;
	else if (.@map$ == "jor_twig")
		.@need = 9;
	else if (.@map$ == "jor_root3")
		.@need = 11;
	else
		return true;
	if (.@step >= .@need)
		return true;
	message strcharinfo(0),"Advance Episode 20: The Immortal before warping to this area.";
	return false;
}

function	script	EP20_CheckReward	{
	.@whiskers = getarg(0,0);
	.@item = getarg(1,0);
	.@amount = getarg(2,0);
	if (.@whiskers > 0 && .@item > 0 && .@amount > 0) {
		setarray .@items[0],1001217,.@item;
		setarray .@amounts[0],.@whiskers,.@amount;
		if (!checkweight2(.@items,.@amounts)) {
			message strcharinfo(0),"Make enough inventory room for every quest reward.";
			return false;
		}
		return true;
	}
	if (.@whiskers > 0 && !checkweight(1001217,.@whiskers)) {
		message strcharinfo(0),"Make room for the Sacred Cat Whiskers reward.";
		return false;
	}
	if (.@item > 0 && .@amount > 0 && !checkweight(.@item,.@amount)) {
		message strcharinfo(0),"Make room for the quest item reward.";
		return false;
	}
	return true;
}

function	script	EP20_Reward	{
	.@whiskers = getarg(0,0);
	.@reputation = getarg(1,0);
	.@base_exp = getarg(2,0);
	.@job_exp = getarg(3,0);
	if (!callfunc("EP20_CheckReward",.@whiskers))
		return false;
	if (.@whiskers > 0) getitem 1001217,.@whiskers;
	if (.@reputation > 0) add_reputation_points 4,.@reputation;
	if (.@base_exp > 0 || .@job_exp > 0) getexp .@base_exp,.@job_exp;
	return true;
}

function	script	EP20_DailyWhiskers	{
	.@amount = getarg(0);
	.@tier = get_reputation_points(4) / 1000;
	if (.@tier > 3) .@tier = 3;
	return .@amount + (.@tier * 2);
}

function	script	EP20_NextReset	{
	.@now = gettimetick(2);
	.@seconds_today = gettime(DT_HOUR) * 3600 + gettime(DT_MINUTE) * 60 + gettime(DT_SECOND);
	.@reset = .@now - .@seconds_today + 14400;
	if (.@reset <= .@now) .@reset += 86400;
	return .@reset;
}

function	script	EP20_DailyKey	{
	// A monotonically increasing 04:00-server-time day key.
	.@now = gettimetick(2);
	.@seconds_today = gettime(DT_HOUR) * 3600 + gettime(DT_MINUTE) * 60 + gettime(DT_SECOND);
	if (.@seconds_today < 14400) .@now -= 86400;
	return .@now / 86400;
}

function	script	EP20_MoveQuest	{
	.@old = getarg(0);
	.@next = getarg(1,0);
	if (isbegin_quest(.@old) == 1)
		completequest .@old;
	if (.@next > 0 && isbegin_quest(.@next) == 0)
		setquest .@next;
	return;
}

function	script	EP20_EnterInstance	{
	.@name$ = getarg(0);
	// Canyon Exploration and Nest of Small Branches are the two documented
	// solo story expeditions. Do not bind either to an unrelated party slot.
	.@solo = (.@name$ == "Canyon Exploration" || .@name$ == "Nest of Small Branches");
	.@party_id = getcharid(1);
	if (.@party_id > 0 && !.@solo) {
		.@iid = instance_id(IM_PARTY);
		if (.@iid > 0 && instance_live_info(ILI_NAME,.@iid) != .@name$) {
			mes "[Episode 20 Expedition]";
			mes "Your party already owns another active instance.";
			close;
		}
		if (.@iid <= 0) {
			if (!is_party_leader()) {
				mes "[Episode 20 Expedition]";
				mes "The party leader must create the expedition.";
				close;
			}
			.@iid = instance_create(.@name$,IM_PARTY,.@party_id);
		}
	} else {
		.@iid = instance_id(IM_CHAR);
		if (.@iid > 0 && instance_live_info(ILI_NAME,.@iid) != .@name$) {
			mes "[Episode 20 Expedition]";
			mes "You already own another active solo instance.";
			close;
		}
		if (.@iid <= 0)
			.@iid = instance_create(.@name$,IM_CHAR,getcharid(0));
	}
	if (.@iid <= 0) {
		mes "[Episode 20 Expedition]";
		mes "The expedition could not be created. Try again shortly.";
		close;
	}
	.@result = instance_enter(.@name$,-1,-1,getcharid(0),.@iid);
	if (.@result != IE_OK) {
		mes "[Episode 20 Expedition]";
		mes "The instance map is not ready (error " + .@result + ").";
		close;
	}
	end;
}
