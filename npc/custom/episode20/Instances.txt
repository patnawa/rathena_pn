1@twbs,82,81,4	script(DISABLED)	Mysterious Storage#ep20	4_ENERGY_BLUE,{
	if (instance_live_info(ILI_NAME) != "The Undying" || 'ep20_undying_stage != 2) end;
	mes "[Mysterious Storage]";
	.@now = gettimetick(2);
	if ('ep20_undying_mode == 1) {
		if (EP20_UndyingNormalNext > .@now) { mes "You already claimed today's Normal Mode storage."; close; }
		.@token = 102567;
		setarray .@rates[0],4000,4000,4990,3387,3387,4000,0,0,1000,1000,1000,1000,1000,1000;
	} else if ('ep20_undying_mode == 2) {
		if (EP20_UndyingHighNext > .@now) { mes "You already claimed today's High Mode storage."; close; }
		.@token = 102568;
		setarray .@rates[0],5773,5773,6957,4990,4990,5773,4990,900,2000,2000,2000,2000,2000,2000;
	} else {
		mes "The storage is inert.";
		close;
	}
	// Keep this order synchronized with .@rates. reference server's 2026-08-11
	// Suspicious Storage table lists these as independent rolls, including
	// the Pure materials that can also drop directly from Lasgand.
	setarray .@pool[0],1001029,1001030,1001248,1001031,1001032,1001249,1001250,1001251,450270,470197,480283,480284,450271,470198;
	if (getarraysize(.@pool) != 14 || getarraysize(.@rates) != 14) {
		debugmes "Episode 20: Suspicious Storage pool/rate array mismatch.";
		mes "The storage seal is unstable. Please report this to a Game Master.";
		close;
	}
	.@count = 0;
	for (.@i = 0; .@i < getarraysize(.@pool); ++.@i) {
		if (rand(10000) < .@rates[.@i]) {
			.@items[.@count] = .@pool[.@i];
			.@amounts[.@count] = 1;
			++.@count;
		}
	}
	.@items[.@count] = .@token;
	.@amounts[.@count] = 1;
	++.@count;
	if (!checkweight2(.@items,.@amounts)) {
		mes "Make enough inventory room for every storage reward.";
		close;
	}
