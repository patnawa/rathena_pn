1@twbs,78,80,0	script	#EP20_Undying_Control	HIDDEN_NPC,{
	end;
OnInstanceInit:
	if (instance_live_info(ILI_NAME) != "The Undying") end;
	'ep20_undying_stage = 0;
	'ep20_undying_mode = 0;
	enablenpc instance_npcname("Lasgand#ep20_undying_start");
	mapannounce instance_mapname("1@twbs"),"The Immortal: choose a mode at Lasgand.",bc_map,0x77AAFF;
	end;
OnTimer3000:
	if (instance_live_info(ILI_NAME) != "The Undying" || 'ep20_undying_stage != 1) { stopnpctimer; end; }
	if (unitexists('ep20_undying_gid)) {
		sc_end SC_ARMOR,'ep20_undying_gid;
		setunitdata 'ep20_undying_gid,UMOB_DAMAGETAKEN,('ep20_undying_mode == 2 ? 1 : 10);
		setunitdata 'ep20_undying_gid,UMOB_MODE,'ep20_undying_boss_mode;
	}
	end;
// One-second encounter clock; no player attachment is needed.
OnTimer4000:
	if (instance_live_info(ILI_NAME) != "The Undying" || 'ep20_undying_stage != 1 || !unitexists('ep20_undying_gid)) { stopnpctimer; end; }
	'ep20_undying_ticks++;
	.@dead = 0;
	for (.@i = 0; .@i < getarraysize('ep20_undying_roster); ++.@i) {
		// Script boolean expressions evaluate both operands, including offline reads.
		if (!isloggedin('ep20_undying_accounts[.@i],'ep20_undying_roster[.@i])) continue;
		if (readparam(Hp,'ep20_undying_roster[.@i]) <= 0) .@dead = 1;
	}
	setunitdata 'ep20_undying_gid,UMOB_DMGIMMUNE,.@dead;
	if (.@dead != 'ep20_undying_dead) {
		'ep20_undying_dead = .@dead;
		mapannounce instance_mapname("1@twbs"),(.@dead ? "Lasgand feeds on a fallen challenger. Revive your companions!" : "Lasgand's protection fades."),bc_map,0xFF8888;
	}
	.@map$ = instance_mapname("1@twbs");
	.@n = getmapunits(BL_PC,.@map$,.@players);
	.@count = 0;
	for (.@i = 0; .@i < .@n; ++.@i) {
		.@cid = getcharid(0,rid2name(.@players[.@i]));
		if (readparam(Hp,.@cid) > 0) .@targets[.@count++] = .@players[.@i];
	}
	if (.@count) {
		getunitdata 'ep20_undying_gid,.@boss;
		for (.@i = 0; .@i < 4; ++.@i) {
			.@gid = 'ep20_undying_visions[.@i];
			if (!unitexists(.@gid) || 'ep20_undying_ticks < 'ep20_undying_cast[.@i]) continue;
			getunitdata .@gid,.@vision;
			.@dx = .@vision[UMOB_X] - .@boss[UMOB_X];
			.@dy = .@vision[UMOB_Y] - .@boss[UMOB_Y];
			.@distance = max(.@dx,-.@dx,.@dy,-.@dy);
			// The published encounter specifies distance scaling, not exact timings.
			// Near/middle/far bands use 1/3/5 additional cast seconds.
			.@delay = (.@distance <= 5 ? 1 : (.@distance <= 10 ? 3 : 5));
			getmapxy .@targetmap$,.@x,.@y,BL_PC,rid2name(.@targets[rand(.@count)]);
			setarray .@skills$[0],"NPC_RAINOFMETEOR","NPC_GROUNDDRIVE","WZ_STORMGUST";
			unitskillusepos .@gid,.@skills$[('ep20_undying_ticks + .@i) % 3],5,.@x,.@y,.@delay,false,0,true;
			'ep20_undying_cast[.@i] = 'ep20_undying_ticks + .@delay + 5;
		}
	}
	setnpctimer 3000;
	end;
OnVisionDead:
	end;
OnBossDead:
	if (instance_live_info(ILI_NAME) != "The Undying" || 'ep20_undying_stage != 1) end;
	if (mobcount(instance_mapname("1@twbs"),instance_npcname(strnpcinfo(0))+"::OnBossDead") > 0) end;
	'ep20_undying_stage = 2;
	stopnpctimer;
	killmonster instance_mapname("1@twbs"),instance_npcname(strnpcinfo(0))+"::OnVisionDead";
	enablenpc instance_npcname("Mysterious Storage#ep20");
	enablenpc instance_npcname("White Cat#ep20_undying_exit");
	mapannounce instance_mapname("1@twbs"),"Lasgand is suppressed. The Mysterious Storage has appeared.",bc_map,0x55FFAA;
	end;
}

	.@now = gettimetick(2);

	.@control$ = instance_npcname("#EP20_Undying_Control");
	setnpctimer 0,.@control$;
	initnpctimer .@control$;

	if ('ep20_undying_claimed[.@member]) { mes "You already claimed this instance's storage."; close; }

	if ('ep20_undying_mode == 1) {
		if (EP20_UndyingNormalNext > .@now) { mes "You already claimed today's Normal Mode storage."; close; }

		setarray .@rates[0],4000,4000,4990,3387,3387,4000,0,0,1000,1000,1000,1000,1000,1000;

		if (EP20_UndyingHighNext > .@now) { mes "You already claimed today's Hard Mode storage."; close; }
		.@token = 102568;
		setarray .@rates[0],5773,5773,6957,4990,4990,5773,4990,900,2000,2000,2000,2000,2000,2000;
	} else {
		mes "The storage is inert.";
		close;
	}
	// Keep this order synchronized with .@rates. reference server's 2026-08-11
	// Suspicious Storage table lists these as independent rolls, including
	// the Pure materials that can also drop directly from Lasgand.
	setarray .@pool[0],1001029,1001030,1001248,1001031,1001032,1001249,1001250,1001251,450270,470197,480283,480284,450271,470198;
	if (getarraysize(.@pool) != 14 || getarraysize(.@rates) != 14) {
		debugmes "Episode 20: Suspicious Storage pool/rate array mismatch.";
		mes "The storage seal is unstable. Please report this to a Game Master.";
		close;
	}
	// Persist the rolls before checking capacity so retries cannot reroll loot.
	if (!'ep20_undying_rolled[.@member]) {
		for (.@i = 0; .@i < getarraysize(.@pool); ++.@i)
			if (rand(10000) < .@rates[.@i]) 'ep20_undying_loot[.@member] |= (1 << .@i);
		'ep20_undying_rolled[.@member] = 1;
	}
	.@count = 0;
	for (.@i = 0; .@i < getarraysize(.@pool); ++.@i) {
		if ('ep20_undying_loot[.@member] & (1 << .@i)) {
			.@items[.@count] = .@pool[.@i];
			.@amounts[.@count] = 1;
			++.@count;
		}
	}
	.@items[.@count] = .@token;
	.@amounts[.@count] = 1;
	++.@count;
	if (!checkweight2(.@items,.@amounts)) {
		mes "Make enough inventory room for every storage reward.";
		close;
	}
	'ep20_undying_claimed[.@member] = 1;
	for (.@i = 0; .@i < .@count; ++.@i)
		getitem .@items[.@i],.@amounts[.@i];
	callfunc "EP20_Reward",0,20,42000000,29500000;
	if ('ep20_undying_mode == 1)
		EP20_UndyingNormalNext = callfunc("EP20_NextReset");
	else
		EP20_UndyingHighNext = callfunc("EP20_NextReset");
	mes "The storage yields its sealed materials and this mode's Antiquity Issgard. This mode resets at 04:00.";
	close;
}
