// Independently implemented #243 Infinite Catalyst Box 3.
// Acquisition is deliberately left to existing reward/admin grant systems.
function	script	F_PNInfiniteCatalystBox	{
	mes "[Infinite Catalyst Box 3]";
	mes "Choose one catalyst rental. It lasts 30 days, including time offline.";
	mes "Keep it in your inventory to waive the corresponding skill material cost.";
	.@choice = select("Soul Talisman:Trap:Special Alloy Trap:Cancel");
	if (.@choice == 4)
		close;
	.@item = 50150 + .@choice;
	mes "Receive " + getitemname(.@item) + "?";
	if (select("Receive rental:Cancel") == 2)
		close;
	// Revalidate after the last dialogue yield. Refuse duplicate tokens so
	// checkweight must reserve a new inventory slot, including for Etc rentals.
	if (countitem(50150) < 1) {
		mes "The catalyst box is no longer in your inventory.";
		close;
	}
	if (countitem(.@item) || rentalcountitem(.@item)) {
		mes "You already carry this catalyst. Use your existing rental first.";
		close;
	}
	if (!checkweight(.@item, 1)) {
		mes "Make room for one item in your inventory first.";
		close;
	}
	// No dialogue or other yield between grant and consumption. If native
	// rentitem fails it aborts before box consumption; success is checked too.
	rentitem .@item, 2592000;
	if (rentalcountitem(.@item) == 1) {
		delitem 50150, 1;
		mes "Your 30-day rental has started.";
	}
	close;
}
