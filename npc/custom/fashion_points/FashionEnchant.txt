//===== rAthena Script =======================================
//= Complete Fashion Stone application service
//===== Provenance / compatibility ===========================
//= Stone -> enchant mappings and slot families are the complete 366-pair set
//= from revision 61305 of:
//= https://wiki.muhro.eu/Costume_Enchants_(Fashion_Points)
//=
//= The page does not publish an application chance. This service therefore
//= applies one supported stone deterministically; a validation or mutation
//= failure consumes nothing. This is a declared compatibility policy, not a
//= claim about an unpublished MuhRO probability.
//=
//= The stock Malangdo NPC covers only 196 pairs. This service covers all 366
//= pairs, including Druid/Karnos/Alitea with the integrated class skills.
//= FP_EnchantSupported still validates every record's card/enchant type.
//=
//= The custom modifyequipitem command changes the enchant slot on the original
//= inventory record. Unique ID, binding, expiry, favorite state, refine, grade,
//= attribute, unrelated cards and random options are retained exactly. The
//= selected stone is debited by exact inventory index and refunded with its
//= binding intact if the guarded mutation fails.
//============================================================

mal_in01,25,113,3	script	Complete Fashion Enchanter#FP	4_WHITETIGER,{
	disable_items;
	mes "[Complete Fashion Enchanter]";
	mes "I support every currently functional stone in the Fashion Point boxes.";
	mes "Application is guaranteed, and an invalid or interrupted validation consumes nothing.";
	mes "The target enchant slot must be empty.";
	next;
	.@cat=select("Top stone:Middle stone:Lower stone:Garment first-slot stone:Garment second-slot stone:Leave")-1;
	if (.@cat == 5) close;
	switch (.@cat) {
	case 0:
		.@first_box=0; .@last_box=4; .@part=EQI_COSTUME_HEAD_TOP;
		.@mask=EQP_COSTUME_HEAD_TOP; .@slot=0; .@part_name$="upper costume";
		break;
	case 1:
		.@first_box=5; .@last_box=9; .@part=EQI_COSTUME_HEAD_MID;
		.@mask=EQP_COSTUME_HEAD_MID; .@slot=1; .@part_name$="middle costume";
		break;
	case 2:
		.@first_box=10; .@last_box=14; .@part=EQI_COSTUME_HEAD_LOW;
		.@mask=EQP_COSTUME_HEAD_LOW; .@slot=2; .@part_name$="lower costume";
		break;
	case 3:
		.@first_box=15; .@last_box=19; .@part=EQI_COSTUME_GARMENT;
		.@mask=EQP_COSTUME_GARMENT; .@slot=0; .@part_name$="costume garment first slot";
		break;
	case 4:
		.@first_box=20; .@last_box=20; .@part=EQI_COSTUME_GARMENT;
		.@mask=EQP_COSTUME_GARMENT; .@slot=1; .@part_name$="costume garment second slot";
		break;
	}

	// Build an owned-only selection, keeping menus bounded to 12 entries/page.
	.@owned=0;
	for (.@box=.@first_box; .@box<=.@last_box; ++.@box) {
		.@n=callfunc("FP_LoadBox",.@box);
		for (.@i=0; .@i<.@n; ++.@i) {
			.@stone=@FP_BoxStone[.@i];
			.@enchant=@FP_BoxEnchant[.@i];
			if (countitem(.@stone) < 1) continue;
			if (getiteminfo(.@stone,ITEMINFO_TYPE) < 0 || !callfunc("FP_EnchantSupported",.@enchant)) continue;
			.@stone_id[.@owned]=.@stone;
			.@enchant_id[.@owned]=.@enchant;
			++.@owned;
		}
	}
	if (!.@owned) {
		mes "[Complete Fashion Enchanter]";
		mes "You do not have a supported " + .@part_name$ + " stone.";
		close;
	}
	.@page=0;
	while (1) {
		.@from=.@page*12;
		.@last=.@from+12;
		if (.@last>.@owned) .@last=.@owned;
		.@menu$="";
		for (.@i=.@from; .@i<.@last; ++.@i)
			.@menu$+=getitemname(.@stone_id[.@i])+" ["+countitem(.@stone_id[.@i])+"]:";
		.@shown=.@last-.@from;
		if (.@last<.@owned) .@menu$+="Next page:";
		if (.@page>0) .@menu$+="Previous page:";
		.@menu$+="Cancel:";
		.@pick=select(.@menu$)-1;
		if (.@pick<.@shown) {
			.@pick+=.@from;
			break;
		}
		.@cursor=.@shown;
		if (.@last<.@owned) {
			if (.@pick==.@cursor) { ++.@page; continue; }
			++.@cursor;
		}
		if (.@page>0 && .@pick==.@cursor) { --.@page; continue; }
		close;
	}
	.@stone=.@stone_id[.@pick];
	.@enchant=.@enchant_id[.@pick];

	.@equip_id=getequipid(.@part);
	mes "[Complete Fashion Enchanter]";
	if (.@equip_id < 1 || getiteminfo(.@equip_id,ITEMINFO_TYPE) != IT_ARMOR
		|| !(getiteminfo(.@equip_id,ITEMINFO_LOCATIONS)&.@mask)) {
		mes "Equip a compatible " + .@part_name$ + " first.";
		close;
	}
	if (.@slot < getitemslots(.@equip_id)) {
		mes "That position is a real card socket, not a costume-enchant slot, so it was refused.";
		close;
	}
	if (getequipcardid(.@part,.@slot)) {
		mes "The target slot already contains an enchant. Recover it first; it will not be overwritten.";
		close;
	}
	mes "Apply ^0000ff"+getitemname(.@stone)+"^000000 to ^0000ff"+getitemname(.@equip_id)+"^000000?";
	mes "The application is guaranteed. The original item and all of its metadata are retained.";
	next;
	if (select("Apply enchant:Cancel")==2) close;

	// Re-read mutable state after confirmation and debit one exact, refundable
	// stone stack before making the in-place compare-and-swap mutation.
	if (!callfunc("FP_EnchantSupported",.@enchant) || countitem(.@stone)<1
		|| getequipid(.@part)!=.@equip_id || getequipcardid(.@part,.@slot)!=0) {
		mes "[Complete Fashion Enchanter]";
		mes "The stone or equipped costume changed. Nothing was consumed.";
		close;
	}
	getinventorylist;
	.@stone_list=-1;
	for (.@i=0; .@i<@inventorylist_count; ++.@i) {
		if (@inventorylist_id[.@i]==.@stone && @inventorylist_amount[.@i]>0
			&& !@inventorylist_expire[.@i] && !@inventorylist_favorite[.@i] && .@stone_list<0)
			.@stone_list=.@i;
	}
	if (.@stone_list<0) {
		mes "[Complete Fashion Enchanter]";
		mes "A permanent, non-favorite stone stack could not be located. Nothing was consumed.";
		close;
	}
	if (.@slot < getitemslots(.@equip_id)) {
		mes "The selected position is not an available costume-enchant slot.";
		close;
	}
	.@stone_idx=@inventorylist_idx[.@stone_list];
	.@stone_bound=@inventorylist_bound[.@stone_list];
	if (!delitemidx(.@stone_idx,1)) {
		mes "[Complete Fashion Enchanter]";
		mes "The exact stone stack changed. Nothing was consumed.";
		close;
	}
	if (!modifyequipitem(.@part,.@equip_id,0,-1,.@slot,0,.@enchant)) {
		if (.@stone_bound) getitembound .@stone,1,.@stone_bound;
		else getitem .@stone,1;
		mes "[Complete Fashion Enchanter]";
		mes "The equipped item changed or failed a safety check. Your stone was refunded.";
		close;
	}
	mes "[Complete Fashion Enchanter]";
	mes "Success. The same costume and unique ID now carry the enchant; all unrelated metadata was preserved.";
	mes "If an equipment restriction prevented automatic re-equipping, the item is safely in your inventory.";
	close;
}
