function	script	FP_StoneKnown	{
	.@needle = getarg(0);
	for (.@box = 0; .@box < 21; ++.@box) {
		.@n = callfunc("FP_LoadBox",.@box);
		for (.@i = 0; .@i < .@n; ++.@i)
			if (@FP_BoxStone[.@i] == .@needle) return 1;
	}
	return 0;
}

function	script	FP_StoneFromEnchant	{
	.@needle = getarg(0);
	for (.@box = 0; .@box < 21; ++.@box) {
		.@n = callfunc("FP_LoadBox",.@box);
		for (.@i = 0; .@i < .@n; ++.@i)
			if (@FP_BoxEnchant[.@i] == .@needle) return @FP_BoxStone[.@i];
	}
	return 0;
}

function	script	FP_IsTombCostume	{
	// Stock kRO Bio5 costume versions of Old Headgears.
	.@id = getarg(0);
	return (.@id >= 19961 && .@id <= 19974);
}

function	script	FP_OpenBox	{
	.@box_index=getarg(0);
	.@box_item=getarg(1);
	.@n=callfunc("FP_LoadBox",.@box_index);
	deletearray .@available[0],128;
	.@count=0;
	for (.@i=0; .@i<.@n; ++.@i) {
		// Require both the physical stone and its matching enchant database
		// record. Enchanter support is a separate compatibility concern: this
		// snapshot's stock Malangdo script predates the newest published pairs.
		if (getiteminfo(@FP_BoxStone[.@i],ITEMINFO_TYPE) < 0 || getiteminfo(@FP_BoxEnchant[.@i],ITEMINFO_TYPE) < 0) continue;
		.@available[.@count++]=@FP_BoxStone[.@i];
	}
	if (!.@count) {
		dispbottom "This Fashion box has no supported published outcomes. It was not consumed.";
		return 0;
	}
	.@reward=.@available[rand(.@count)];
	if (countitem(.@box_item) < 1 || !checkweight(.@reward,1)) {
		dispbottom "The Fashion box was not consumed. Make room in your inventory and try again.";
		return 0;
	}
	// DelayConsume: consume only after every validation and immediately before
	// the successful award.
	delitem .@box_item,1;
	getitem .@reward,1;
	dispbottom "The Fashion box awarded " + getitemname(.@reward) + ". (Uniform compatibility selection.)";
	return 1;
}

// Account login + online-only Gold Point clock.
-	script	FashionPointClock	FAKE_NPC,{
	end;

OnInit:
	bindatcmd "activity",strnpcinfo(3) + "::OnAtActivity",0,99;
	end;

OnPCLoginEvent:
	if (#FP_Gold >= 50) {
		#FP_Gold = 50;
		dispbottom "Gold Points are capped at 50. Visit the Fashion Point Manager to exchange them.";
		@FP_TimerRunning = 0;
		end;
	}
	@FP_TimerRunning = 1;
	addtimer 180000,strnpcinfo(3) + "::OnGoldTick";
	end;

OnGoldTick:
	@FP_TimerRunning = 0;
	if (#FP_Gold < 50) ++#FP_Gold;
	if (#FP_Gold >= 50) {
		#FP_Gold = 50;
		dispbottom "Gold Points reached the cap of 50. Exchange them for Fashion Points.";
		end;
	}
	@FP_TimerRunning = 1;
	addtimer 180000,strnpcinfo(3) + "::OnGoldTick";
	end;

OnAtActivity:
	dispbottom "Gold Points: " + #FP_Gold + "/50 | Fashion Points: " + #FP_Fashion;
	end;
}

	mes "Fashion Points: ^cc33cc" + #FP_Fashion + "^000000";
	next;
	.@choice=select("Exchange Gold Points 1:1:About account scope:Leave");
	if (.@choice == 3) close;
	if (.@choice == 2) {
		mes "[Gold Point Manager]";
		mes "This implementation stores points on your rAthena login account. MuhRO's published 'master account' can span linked game accounts, but stock rAthena has no equivalent relationship.";
		close;
	}
	if (#FP_Gold < 1) {
		mes "[Gold Point Manager]";
		mes "You do not have any Gold Points to exchange.";
		close;
	}
	mes "[Gold Point Manager]";
	mes "How many? (1-" + #FP_Gold + ")";
	input .@amount,1,#FP_Gold;
	#FP_Gold -= .@amount;
	#FP_Fashion += .@amount;
	if (!@FP_TimerRunning) {
		@FP_TimerRunning = 1;
		addtimer 180000,"FashionPointClock::OnGoldTick";
	}
	mes "[Gold Point Manager]";
	mes "Exchanged ^ff8800" + .@amount + " Gold Points^000000 for ^cc33cc" + .@amount + " Fashion Points^000000.";
	close;
}
