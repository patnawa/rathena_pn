// Existing certificate IDs, with native refinability and equipment-type checks.
// Supports modern weapons/armor; excludes shadow and costume equipment.
function	script	F_PNRefine15Ticket	{
	disable_items;
	mes "[Master Refiner]";
	mes "One matching +15 certificate raises an equipped weapon or armor directly to +15 without breaking it.";
	setarray .@slots[0],EQI_HEAD_TOP,EQI_ARMOR,EQI_HAND_L,EQI_HAND_R,EQI_GARMENT,EQI_SHOES,EQI_ACC_L,EQI_ACC_R,EQI_HEAD_MID,EQI_HEAD_LOW;
	.@menu$ = "";
	.@count = 0;
	for (.@i = 0; .@i < getarraysize(.@slots); ++.@i) {
		.@slot = .@slots[.@i];
		if (!getequipisequiped(.@slot))
			continue;
		if (!getequipisenableref(.@slot) || getequiprefinerycnt(.@slot) >= 15)
			continue;
		.@id = getequipid(.@slot);
		.@type = getiteminfo(.@id,ITEMINFO_TYPE);
		if (.@type != IT_WEAPON && .@type != IT_ARMOR)
			continue;
		.@eligible[.@count++] = .@slot;
		.@menu$ += getequipname(.@slot)+":";
	}
	if (!.@count) {
		mes "Equip a refinable weapon or armor below +15 first.";
		return;
	}
	.@choice = select(.@menu$+"Cancel")-1;
	if (.@choice >= .@count)
		return;
	.@slot = .@eligible[.@choice];
	.@id = getequipid(.@slot);
	.@refine = getequiprefinerycnt(.@slot);
	.@unique$ = getequipuniqueid(.@slot);
	.@type = getiteminfo(.@id,ITEMINFO_TYPE);
	.@ticket = (.@type == IT_WEAPON) ? 6872 : 6878;
	if (.@unique$ == "" || .@unique$ == "0" || !getequipisenableref(.@slot) || .@refine >= 15 || (.@type != IT_WEAPON && .@type != IT_ARMOR)) {
		mes "That equipment is no longer eligible.";
		return;
	}
	if (countitem(.@ticket) < 1) {
		mes "Bring one "+getitemname(.@ticket)+" for this equipment.";
		return;
	}
	mes "Raise "+getequipname(.@slot)+" from +"+.@refine+" to +15?";
	mes "Consumes one "+getitemname(.@ticket)+".";
	next;
	if (select("Use certificate:Cancel") != 1)
		return;
	// Revalidate after the yielding menu, before consuming the certificate.
	if (!getequipisequiped(.@slot) || getequipid(.@slot) != .@id || getequipuniqueid(.@slot) != .@unique$ || getequiprefinerycnt(.@slot) != .@refine || !getequipisenableref(.@slot) || countitem(.@ticket) < 1) {
		mes "Your equipment or certificate changed. Please start again.";
		return;
	}
	// No yield between validation, consumption and refinement.
	delitem .@ticket,1;
	successrefitem .@slot,15-.@refine;
	mes "Your equipment is now +15.";
	return;
}
