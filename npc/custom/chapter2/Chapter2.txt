// Chapter 2 - Song of the World Tree: Fragments of the End
// Playable implementation based on the published MuhRO/RockMMO progression.

function	script	CH2_Complete	{
	return isbegin_quest(27101) == 2;
}

function	script	CH2_RepLevel	{
	if (CH2_Reputation >= 300) return 3;
	if (CH2_Reputation >= 200) return 2;
	if (CH2_Reputation >= 100) return 1;
	return 0;
}

function	script	CH2_Instance	{
	.@name$ = getarg(0);
	.@iid = instance_id(IM_CHAR);
	if (.@iid && instance_live_info(ILI_NAME,.@iid) != .@name$) {
		mes "[Chapter 2 Expedition]";
		mes "You already own another active instance. Close it from the instance window before creating this one.";
		close;
	}
	if (!.@iid) {
		.@iid = instance_create(.@name$,IM_CHAR,getcharid(0));
		if (.@iid < 0) {
			mes "[Chapter 2 Expedition]";
			mes "The instance could not be created. Please try again shortly.";
			close;
		}
	}
	if (instance_enter(.@name$,-1,-1,getcharid(0),.@iid) != IE_OK) {
		mes "[Chapter 2 Expedition]";
		mes "The instance entrance is unavailable.";
		close;
	}
	end;
}

function	script	CH2_GiveReward	{
	.@quest = getarg(0);
	.@coins = getarg(1);
	if (!checkweight(1002700,.@coins)) {
		mes "[Flame Branch Mission]";
		mes "Make room for " + .@coins + " Flame Coins before filing this mission.";
		close;
	}
	completequest .@quest;
	setd("CH2_Daily_CD_"+.@quest,gettimetick(2)+14400);
	getitem 1002700,.@coins;
	CH2_Reputation += 2;
	getexp 12000000,8400000;
	mes "[Flame Branch Mission]";
	mes "Reward: ^ff8800"+.@coins+" Flame Coins^000000 and ^33aa33+2 reputation^000000.";
	close;
}

function	script	CH2_RecreateItem	{
	.@idx = getarg(0);
	.@newid = getarg(1);
	.@newref = getarg(2);
	.@part = getarg(3);
	if (.@idx < 0 || .@idx >= @inventorylist_count || !@inventorylist_equip[.@idx]
		|| @inventorylist_expire[.@idx] || getequipuniqueid(.@part))
		return 0;
	setarray .@oid[0],@inventorylist_option_id1[.@idx],@inventorylist_option_id2[.@idx],@inventorylist_option_id3[.@idx],@inventorylist_option_id4[.@idx],@inventorylist_option_id5[.@idx];
	setarray .@oval[0],@inventorylist_option_value1[.@idx],@inventorylist_option_value2[.@idx],@inventorylist_option_value3[.@idx],@inventorylist_option_value4[.@idx],@inventorylist_option_value5[.@idx];
	setarray .@opar[0],@inventorylist_option_parameter1[.@idx],@inventorylist_option_parameter2[.@idx],@inventorylist_option_parameter3[.@idx],@inventorylist_option_parameter4[.@idx],@inventorylist_option_parameter5[.@idx];
	.@identify = @inventorylist_identify[.@idx];
	.@attribute = @inventorylist_attribute[.@idx];
	.@c1 = @inventorylist_card1[.@idx]; .@c2 = @inventorylist_card2[.@idx];
	.@c3 = @inventorylist_card3[.@idx]; .@c4 = @inventorylist_card4[.@idx];
	.@grade = @inventorylist_enchantgrade[.@idx];
	.@bound = @inventorylist_bound[.@idx];
	delequip .@part;
	if (!delitemidx(@inventorylist_idx[.@idx],1))
		return 0;
	if (.@bound)
		getitembound4 .@newid,1,.@identify,.@newref,.@attribute,.@c1,.@c2,.@c3,.@c4,.@bound,.@grade,.@oid,.@oval,.@opar;
	else
		getitem4 .@newid,1,.@identify,.@newref,.@attribute,.@c1,.@c2,.@c3,.@c4,.@grade,.@oid,.@oval,.@opar;
	return 1;
}

// Monster-death labels cannot pause for inventory management. Deliver their
// one-item rewards by mail when the killer cannot carry them immediately.
function	script	CH2_GiveInstanceItem	{
	.@item = getarg(0);
	.@amount = getarg(1,1);
	if (checkweight(.@item,.@amount)) {
		getitem .@item,.@amount;
		return 1;
	}
	setarray .@mail_item[0],.@item;
	setarray .@mail_amount[0],.@amount;
	mail getcharid(0),"Flame Branch","Chapter 2 Reward","Your inventory was full when the instance reward was issued, so it was delivered by mail.",0,.@mail_item,.@mail_amount;
	dispbottom "Your inventory was full. The Chapter 2 reward was delivered by mail.";
	return 1;
}
