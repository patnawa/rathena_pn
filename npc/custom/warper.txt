// ------------------- Functions -------------------
// * Go("<map>",<x>,<y>);
//	~ Warps directly to a map.
//
// * Disp("<Menu Option>",<first option>,<last option>);
// * Pick("<map_prefix>"{,<index offset>});
//	~ Dynamic menu and map selection (auto-numbered).
//	~ Fields and Dungeons must use Disp and Pick Functions.
//
// * Disp("<Option 1>:<Option 2>:<etc.>");
// * Pick("","<map1>","<map2>","<etc.>");
//	~ Manual menu and map selection (listed).
//
// * Restrict("<RE | Pre-RE>"{,<menu option numbers>});
//	~ Only allows map for Renewal or Pre-Renewal modes.
//     If menu option numbers are given, only those maps
//     will be restricted (i.e. not for "Go").
//
// Other notes:
//   ~ Array @c[] holds all (x,y) coordinates.
//   ~ Use @c[2] EXCEPT when maps begin dynamically
//	  at 0: use @c[0] and Pick() offset 1.
// --------------------------------------------------

function Go {
	.@map$ = getarg(0);
	.@x = getarg(1,0); .@y = getarg(2,0);
	// Reject stale saved destinations before invoking story access helpers.
	// Script boolean operators are eager: never check a cell on an unloaded map.
	.@available = (getmapusers(.@map$) >= 0);
	if (.@available) {
		if (.@x || .@y)
			.@available = checkcell(.@map$,.@x,.@y,CELL_CHKPASS);
	}
	if (!.@available) {
		message strcharinfo(0),"That destination is unavailable. Please choose another route.";
		end;
	}
	if (!ValidateWarp(.@map$,.@x,.@y))
		end;
	warp .@map$,.@x,.@y;
	// The native warp command can fail without ending the script. Save history
	// only after confirmed arrival; random arrivals retain their actual cell.
	if (getmapxy(.@arrived$,.@arrivedx,.@arrivedy,BL_PC) != 0 || .@arrived$ != .@map$ ||
		((.@x || .@y) && (.@arrivedx != .@x || .@arrivedy != .@y))) {
		message strcharinfo(0),"Unable to warp right now. Your previous Last Warp has been kept.";
		end;
	}
	if (.@map$ == "bl_depth2")
		callfunc "F_BiosphereDepth2Purge";
	set lastwarp$, .@arrived$;
	set lastwarpx, .@arrivedx;
	set lastwarpy, .@arrivedy;
	end;
}

// Permanent access mirrors each normal quest entrance. Do not require the
// whole campaign where a story milestone already permits returning to a map.
function ValidateQuestMap {
	.@map$ = getarg(0);
	// quests_amatsu.txt: the restored lord grants the physical transit permit.
	if (substr(.@map$,0,6) == "ama_dun") {
		if (event_amatsu != 6 || !countitem(7160)) {
			message strcharinfo(0),"Complete the Amatsu dungeon entrance quest and bring your Transit Permit.";
			return false;
		}
	}
	// quests_ayothaya.txt: the stronger thread opens the second floor.
	if (.@map$ == "ayo_fild02" || .@map$ == "ayo_dun01") {
		if (ayodunquest < 3 || !(countitem(7285) || countitem(7287))) {
			message strcharinfo(0),"Receive Boonthom's Holy Thread and open the Ayothaya ruins first.";
			return false;
		}
	}
	if (.@map$ == "ayo_dun02" && (ayodunquest < 11 || !countitem(7287))) {
		message strcharinfo(0),"Receive Boonthom's Holier Thread before entering the second Ayothaya floor.";
		return false;
	}
	if (substr(.@map$,0,6) == "bra_dun" && brazil_ghost < 8) {
		message strcharinfo(0),"Complete the Brasilis bathroom ghost investigation first.";
		return false;
	}
	// quests_nameless.txt: Larjes' first visit and the night island transition.
	if (.@map$ == "nameless_i" && aru_monas < 17) {
		message strcharinfo(0),"Unlock Nameless Island with Larjes in the Veins investigation first.";
		return false;
	}
	if ((.@map$ == "nameless_n" || substr(.@map$,0,4) == "abbey") && aru_monas < 20) {
		message strcharinfo(0),"Advance the Nameless Island investigation until the night island opens.";
		return false;
	}
	// quests_rachel.txt and ra_san.txt: the first visit is the story trial.
	if (substr(.@map$,0,5) == "ra_san" && !(MISC_QUEST & 8192)) {
		if (ra_tem_q < ((.@map$ == "ra_san01")?21:22)) {
			message strcharinfo(0),"Unlock Rachel Sanctuary with High Priest Zhed first. Complete the first-floor trial before using deeper floors.";
			return false;
		}
	}
	// quests_moscovia.txt: the stranger offers the forest after the Whale quest.
	if ((.@map$ == "mosk_fild02" || substr(.@map$,0,7) == "mosk_dun") && mos_whale_edq <= 38) {
		message strcharinfo(0),"Complete Finding the Moving Island in Moscovia first.";
		return false;
	}
	// lhz_dun.txt: the sewer permit is a character flag, not a carried item.
	if (substr(.@map$,0,6) == "lhz_dun") {
		if (!(MISC_QUEST & 512)) {
			message strcharinfo(0),"Complete the Lighthalzen laboratory entrance quest with Fishbone first.";
			return false;
		}
		if (.@map$ == "lhz_dun03" || .@map$ == "lhz_dun04") {
			if (BaseLevel < ((Upper == 1)?90:95)) {
				message strcharinfo(0),"Bio Lab floors 3 and 4 require Base Level 95, or 90 for reborn characters.";
				return false;
			}
		}
		// WolfchevLaboratory.txt: investigation visits use the quest map instead.
		if (.@map$ == "lhz_dun04" && lght_duk01 < 6 && lhz_curse < 31) {
			message strcharinfo(0),"Finish the Wolfchev laboratory investigation or Cursed Spirit access quest first.";
			return false;
		}
	}
	// kiel_hyre_quest.txt: initial factory unlock and the permanent keycard.
	if (.@map$ == "kh_dun01" || .@map$ == "kh_dun02") {
		if (KielHyreQuest < 38) {
			message strcharinfo(0),"Open the robot factory during the Kiel Hyre investigation first.";
			return false;
		}
		if (.@map$ == "kh_dun02" || KielHyreQuest >= 106) {
			if (!countitem(7509)) {
				message strcharinfo(0),"Bring the Luxurious Keycard from Kiel Hyre to enter this factory route.";
				return false;
			}
		}
	}
	if (substr(.@map$,0,6) == "nyd_dun" && ep13_mdrama < 6) {
		message strcharinfo(0),"Follow the footprints and enter the Yggdrasil roots in the Two Tribes investigation first.";
		return false;
	}
	// quests_dicastes.txt: tunnel invitation and capital-city approval.
	if ((.@map$ == "dicastes01" || .@map$ == "dicastes02") && ep13_3_invite < 5) {
		message strcharinfo(0),"Receive the Sapha invitation and city guard's approval before visiting El Dicastes.";
		return false;
	}
	if (substr(.@map$,0,6) == "dic_dun" && ep13_3_invite < 4) {
		message strcharinfo(0),"Advance Sapha's Visit until the El Dicastes tunnel is opened.";
		return false;
	}
	// quests_17_1.txt: completed railroad access; the OS clear also proves access
	// for older characters whose introduction quest record is no longer present.
	if (.@map$ == "sp_cor" && isbegin_quest(11603) != 2 && isbegin_quest(12452) != 2) {
		message strcharinfo(0),"Complete Dien's railroad investigation before entering Cor.";
		return false;
	}
	// quests_17_2.txt / ba_maison.txt: power plant, interior, then outer estate.
	if (substr(.@map$,0,2) == "ba_") {
		.@step = (.@map$ == "ba_pw01")?4:((.@map$ == "ba_in01")?6:7);
		if (ep17_2_main < .@step) {
			message strcharinfo(0),"Advance the Episode 17.2 mansion introduction to step "+.@step+" before using this route.";
			return false;
		}
	}
	if (.@map$ == "wolfvill" && ep18_main < 35 && isbegin_quest(8703) == 0) {
		message strcharinfo(0),"Receive Suad's village invitation during Episode 18 first.";
		return false;
	}
	if (.@map$ == "icecastle" && ep19_main < 7) {
		message strcharinfo(0),"Finish the Episode 19 airship introduction to Issgard first.";
		return false;
	}
	return true;
}

function Disp {
	if (getargcount() < 3)
		set @menu$, getarg(0);
	else {
		set @menu$,"";
		for (set .@i,getarg(1); .@i<=getarg(2); set .@i,.@i+1)
			set @menu$, @menu$+getarg(0)+" "+.@i+":";
	}
	return;
}
function Pick {
	// Disp's numbered menus end with ':'; remove it before adding controls.
	.@choices$ = @menu$;
	if (charat(.@choices$,getstrlen(.@choices$)-1) == ":")
		.@choices$ = substr(.@choices$,0,getstrlen(.@choices$)-2);
	explode .@options$,.@choices$,":";
	.@count = getarraysize(.@options$);
	set .@warp_block,@warp_block;
	set @warp_block,0;
	if((@f && .OnlyFirstFld) || (@d && .OnlyFirstDun)){
		set .@select,1;
		if(.@warp_block){
			while(.@warp_block & (1<<.@select)){
				.@select += 1;
			}
		}
	}else{
		set .@select, select(.@choices$+":Back to main menu:Cancel");
		if (.@select == .@count+1)
			return true;
		if (.@select == .@count+2)
			end;
	}
	if (.@select > .@count) {
		message strcharinfo(0),"No routes in this group are enabled.";
		end;
	}
	if (getarg(0) == "") {
		set .@i, .@select;
		set .@map$, getarg(.@i);
	} else {
		set .@i, .@select-getarg(1,0);
		set .@map$, getarg(0)+((.@i<10)?"0":"")+.@i;
	}
	if (.@warp_block & (1<<.@select)) {
		message strcharinfo(0),"This map is not enabled in "+((checkre(0))?"":"Pre-")+"Renewal.";
		end;
	}
	set .@x, @c[.@i*2];
	set .@y, @c[.@i*2+1];
	deletearray @c[0],getarraysize(@c);
	@f = false; @d = false;
	Go(.@map$,.@x,.@y);
}
function Restrict {
	.@type$ = getarg(0);
	
	if (.@type$ == "Brasilis") {
		// If the official warper to Brasilis is enabled, meaning is either
		// Renewal mode or Pre-renewal Brasilis is enabled, we lift the restriction
		if (getnpcid(0, "Crewman_bra2"))
			return;
		// Otherwise we apply the renewal restriction
		.@type$ = "RE";
	}

	if ((.@type$ == "RE" && !checkre(0)) || (.@type$ == "Pre-RE" && checkre(0))) {
		if (getarg(1,0)) {
			set @warp_block,0;
			for (set .@i,1; .@i<getargcount(); set .@i,.@i+1)
				set @warp_block, @warp_block | (1<<getarg(.@i));
		} else {
			message strcharinfo(0),"This map is not enabled in "+((checkre(0))?"":"Pre-")+"Renewal.";
			end;
		}
	}
	return;
}
