function	script	ZC_HasAccess	{
	return ZC_Access || isbegin_quest(27093) == 2;
}

function	script	ZC_DailyKey	{
	// Beginning of the current daily period, which resets at 04:00
	// in the map-server's local timezone.
	.@now = gettimetick(2);
	.@reset = .@now - (gettime(DT_HOUR) * 3600 + gettime(DT_MINUTE) * 60 + gettime(DT_SECOND)) + 14400;
	if (.@now < .@reset)
		.@reset -= 86400;
	return .@reset;
}

function	script	ZC_CanEnter	{
	.@tier = getarg(0);
	if (!callfunc("ZC_HasAccess")) {
		message strcharinfo(0), "Complete Robin's Zero Cell investigation first.";
		return false;
	}
	if (.@tier == 1) {
		if (BaseLevel < 260) {
			message strcharinfo(0), "Lux and Nox require Base Level 260.";
			return false;
		}
		if (ZC_Auth_LN <= gettimetick(2)) {
			message strcharinfo(0), "Your Lux/Nox authorization is inactive. Speak to Newt in ch1fild1.";
			return false;
		}
	} else {
		if (BaseLevel < 270) {
			message strcharinfo(0), "Vita and Ark require Base Level 270.";
			return false;
		}
		if (ZC_Auth_VA <= gettimetick(2)) {
			message strcharinfo(0), "Your Vita/Ark authorization is inactive. Speak to Newt in ch1fild1.";
			return false;
		}
	}
	return true;
}

function	script	ZC_DailyQuest	{
	.@quest = getarg(0);
	.@stamp$ = getarg(1);
	.@reward = getarg(2);
	.@rings = getarg(3);
	.@title$ = getarg(4);
	.@key = callfunc("ZC_DailyKey");

	if (getd(.@stamp$) == .@key) {
		mes "[Robin]";
		mes "You already completed " + .@title$ + " during this daily period.";
		mes "Cleanup assignments reset at 04:00 server time.";
		close;
	}

	.@state = isbegin_quest(.@quest);
	if (.@state == 2) {
		erasequest .@quest;
		.@state = 0;
	}
	if (.@state == 0) {
		mes "[Robin]";
		mes "I need a cleanup team for " + .@title$ + ".";
		mes "The quest log counts any Zero Cell monster defeated on that sector map.";
		if (select("Accept.:Not now.") == 2)
			close;
		setquest .@quest;
		mes "[Robin]";
		mes "The assignment has been added to your quest log.";
		close;
	}

	if (checkquest(.@quest, HUNTING) != 2) {
		mes "[Robin]";
		mes "The cleanup is still in progress. Check your quest window for the remaining count.";
		close;
	}
	if (!checkweight(.@reward, 1, 1001996, .@rings)) {
		mes "[Robin]";
		mes "Please make room in your inventory before receiving the reward.";
		close;
	}
	completequest .@quest;
	setd .@stamp$, .@key;
	getitem .@reward, 1;
	getitem 1001996, .@rings;
	mes "[Robin]";
	mes "Excellent work. Here is your reward.";
	mes "^3366FF" + getitemname(.@reward) + " x1^000000";
	mes "^3366FF" + getitemname(1001996) + " x" + .@rings + "^000000";
	close;
}

ch1fild1,106,84,4	script	Newt#zc_service	4_CH1_DELEGACY02,{
	questinfo QTYPE_QUEST, QMARK_YELLOW, "isbegin_quest(27093) == 1";
	mes "[Newt]";
	if (isbegin_quest(27093) == 1) {
		mes "Robin explained the trace. I can now tune a temporary authorization to the Zero Cell sectors.";
		completequest 27093;
		ZC_Access = 1;
		mes "The investigation is complete. Choose the authorization tier you need.";
		next;
	}
	if (!callfunc("ZC_HasAccess")) {
		mes "We need Robin's investigation data before I can stabilize a route.";
		close;
	}
	.@ln = ZC_Auth_LN - gettimetick(2);
	.@va = ZC_Auth_VA - gettimetick(2);
	if (.@ln < 0) .@ln = 0;
	if (.@va < 0) .@va = 0;
	mes "Lux/Nox authorization: ^3366FF" + ((.@ln) ? ((.@ln / 60) + " minute(s) remaining") : "inactive") + "^000000";
	mes "Vita/Ark authorization: ^3366FF" + ((.@va) ? ((.@va / 60) + " minute(s) remaining") : "inactive") + "^000000";
	next;
	menu "Authorize Lux / Nox (50 Mana Rings)",L_AuthLN,
		 "Authorize Vita / Ark (70 Mana Rings)",L_AuthVA,
		 "Enter an authorized sector",L_Enter,
		 "Cancel",L_End;
L_AuthLN:
	if (BaseLevel < 260) {
		mes "[Newt]";
		mes "Lux and Nox require Base Level 260.";
		close;
	}
	if (countitem(1001996) < 50) {
		mes "[Newt]";
		mes "Bring me 50 Mana Rings.";
		close;
	}
	if (ZC_Auth_LN > gettimetick(2) && select("Replace the remaining authorization time?:Keep the current time.") == 2)
		close;
	delitem 1001996,50;
	ZC_Auth_LN = gettimetick(2) + 3600;
	mes "[Newt]";
	mes "Lux and Nox are authorized for one hour.";
	next;
	goto L_Enter;
L_AuthVA:
	if (BaseLevel < 270) {
		mes "[Newt]";
		mes "Vita and Ark require Base Level 270.";
		close;
	}
	if (countitem(1001996) < 70) {
		mes "[Newt]";
		mes "Bring me 70 Mana Rings.";
		close;
	}
	if (ZC_Auth_VA > gettimetick(2) && select("Replace the remaining authorization time?:Keep the current time.") == 2)
		close;
	delitem 1001996,70;
	ZC_Auth_VA = gettimetick(2) + 3600;
	mes "[Newt]";
	mes "Vita and Ark are authorized for one hour.";
	next;
L_Enter:
	menu "Lux",L_Lux,"Nox",L_Nox,"Vita",L_Vita,"Ark",L_Ark,"Cancel",L_End;
L_Lux:
	if (!callfunc("ZC_CanEnter",1)) close;
	warp "ch1zero1",99,100;
	end;
L_Nox:
	if (!callfunc("ZC_CanEnter",1)) close;
	warp "ch1zero2",99,100;
	end;
L_Vita:
	if (!callfunc("ZC_CanEnter",2)) close;
	warp "ch1zero3",99,100;
	end;
L_Ark:
	if (!callfunc("ZC_CanEnter",2)) close;
	warp "ch1zero4",99,100;
	end;
L_End:
	close;
}

// Enforce expiration after entry. Each sector enables the load event, and
// the timer rechecks once per minute while the character remains inside.
-	script	ZC_ZeroGuard	-1,{
OnPCLoadMapEvent:
	.@map$ = strcharinfo(3);
	if (.@map$ == "ch1zero1" || .@map$ == "ch1zero2" || .@map$ == "ch1zero3" || .@map$ == "ch1zero4") {
		deltimer "ZC_ZeroGuard::OnAuthCheck";
		addtimer 1000,"ZC_ZeroGuard::OnAuthCheck";
	}
	end;
OnAuthCheck:
	.@map$ = strcharinfo(3);
	if (.@map$ == "ch1zero1" || .@map$ == "ch1zero2") {
		if (ZC_Auth_LN <= gettimetick(2)) {
			message strcharinfo(0), "Your Zero Cell authorization expired.";
			warp "ch1fild1",106,81;
			end;
		}
		addtimer 60000,"ZC_ZeroGuard::OnAuthCheck";
	} else if (.@map$ == "ch1zero3" || .@map$ == "ch1zero4") {
		if (ZC_Auth_VA <= gettimetick(2)) {
			message strcharinfo(0), "Your Zero Cell authorization expired.";
			warp "ch1fild1",106,81;
			end;
		}
		addtimer 60000,"ZC_ZeroGuard::OnAuthCheck";
	}
	end;
}
