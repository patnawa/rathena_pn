// Varmundt's Biosphere entrances, exits, and official open-map spawns.
// Source: wiki.muhro.eu/Varmundt's_Biosphere and current kRO map data.

function	script	F_BiosphereAccess	{
	if (BaseLevel < 240) {
		dispbottom "Varmundt's Biosphere requires Base Level 240.";
		return false;
	}
	return true;
}

function	script	F_BiosphereDepth1Access	{
	if (BaseLevel < 250) {
		dispbottom "Biosphere Depth 1 requires Base Level 250.";
		return false;
	}
	if (ep17_2_main < 33) {
		dispbottom "Complete the Episode 17.2 main story before entering Biosphere Depth 1.";
		return false;
	}
	return true;
}

function	script	F_BiosphereDepth2Access	{
	if (BaseLevel < 250) {
		dispbottom "Biosphere Depth 2 requires Base Level 250.";
		return false;
	}
	if (ep17_2_main < 33) {
		dispbottom "Complete the Episode 17.2 main story before entering Biosphere Depth 2.";
		return false;
	}
	if (get_reputation_points(REPUTATION_BIOSPHERE_DEPTH1) < 2000) {
		dispbottom "Biosphere Depth 2 requires 2,000 Depth 1 Research reputation.";
		return false;
	}
	return true;
}

function	script	F_BiosphereDepth2Purge	{
	// Official Depth Abyss entry removes these persistent combat buffs.
	sc_end SC_COMPETENTIA;
	sc_end SC_MUSICAL_INTERLUDE;
	sc_end SC_MOONLITSERENADE;
	sc_end SC_TALISMAN_OF_PROTECTION;
	sc_end SC_TALISMAN_OF_WARRIOR;
	sc_end SC_TALISMAN_OF_MAGICIAN;
	sc_end SC_HEAVEN_AND_EARTH;
	sc_end SC_TALISMAN_OF_FIVE_ELEMENTS;
	sc_end SC_SPELL_ENCHANTING;
	sc_end SC_D_MACHINE;
	return;
}

// Depth maps are deliberately not valid relog destinations.  The nosave
// mapflag protects future position saves; this guard also handles characters
// whose position was persisted before the flag or during an interrupted save.
-	script	Biosphere Relog Guard#depth	-1,{
	end;
OnPCLoginEvent:
	.@map$ = strcharinfo(3);
	if (.@map$ == "bl_depth1" || .@map$ == "bl_depth2") {
		dispbottom "Relogging inside Biosphere Depth is not allowed. Returning to your save point.";
		warp getsavepoint(0),getsavepoint(1),getsavepoint(2);
	}
	end;
}
