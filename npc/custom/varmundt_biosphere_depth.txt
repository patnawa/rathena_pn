function	script	F_BioDepthQuestAccess	{
	if (BaseLevel < 250) {
		mes "This research program requires Base Level 250.";
		return false;
	}
	if (ep17_2_main < 33) {
		mes "Complete the Episode 17.2 mansion story before joining the Depth research program.";
		return false;
	}
	return true;
}

function	script	F_BioExpireWait	{
	.@quest = getarg(0);
	if (checkquest(.@quest,PLAYTIME) == 2)
		erasequest .@quest;
	return isbegin_quest(.@quest);
}

// MuhRO maps the eight I quests to 30-sample submissions and the eight II
// quests to 80+80 monster hunts. This intentionally differs from stock kRO's
// combined QuestInfo wording; see SOURCE_INFO.txt for the required client text.
function	script	F_BioD1ElementResearch	{
	.@i = getarg(0);
	setarray .@name$[0],"Fire","Ice","Earth","Storm","Purification","Corruption","Soul","Venom";
	setarray .@sample[0],1001330,1001332,1001331,1001333,1001335,1001336,1001334,1001337;
	setarray .@sample_q[0],16739,16743,16747,16751,16755,16759,16763,16767;
	setarray .@sample_wait[0],16740,16744,16748,16752,16756,16760,16764,16768;
	setarray .@hunt_q[0],16741,16745,16749,16753,16757,16761,16765,16769;
	setarray .@hunt_wait[0],16742,16746,16750,16754,16758,16762,16766,16770;

	mes "[" + .@name$[.@i] + " Research Administrator]";
	if (!callfunc("F_BioDepthQuestAccess"))
		close;
	callfunc "F_BioExpireWait",.@sample_wait[.@i];
	callfunc "F_BioExpireWait",.@hunt_wait[.@i];
	mes "I handle the " + .@name$[.@i] + " research assignments.";
	next;
	switch(select("Submit 30 samples (2 documents).:Hunt 80 of each paired monster (3 documents).:Leave.")) {
	case 1:
		.@quest = .@sample_q[.@i];
		.@wait = .@sample_wait[.@i];
		if (isbegin_quest(.@wait)) {
			mes "[" + .@name$[.@i] + " Research Administrator]";
			mes "The sample assignment is on cooldown until 04:00.";
			close;
		}
		if (!isbegin_quest(.@quest)) {
			setquest .@quest;
			mes "[" + .@name$[.@i] + " Research Administrator]";
			mes "Collect 30 units of item ^3366FF" + .@sample[.@i] + "^000000 and bring them here.";
			close;
		}
		if (countitem(.@sample[.@i]) < 30) {
			mes "[" + .@name$[.@i] + " Research Administrator]";
			mes "You need 30 samples. You currently have " + countitem(.@sample[.@i]) + ".";
			close;
		}
		if (!checkweight(1001289,2)) {
			mes "[" + .@name$[.@i] + " Research Administrator]";
			mes "Make room for two research documents first.";
			close;
		}
		delitem .@sample[.@i],30;
		erasequest .@quest;
		setquest .@wait;
		getitem 1001289,2;
		mes "[" + .@name$[.@i] + " Research Administrator]";
		mes "Submission accepted. You received 2 Depth 1 Research Documents.";
		close;
	case 2:
		.@quest = .@hunt_q[.@i];
		.@wait = .@hunt_wait[.@i];
		if (isbegin_quest(.@wait)) {
			mes "[" + .@name$[.@i] + " Research Administrator]";
			mes "The hunting assignment is on cooldown until 04:00.";
			close;
		}
		if (!isbegin_quest(.@quest)) {
			setquest .@quest;
			mes "[" + .@name$[.@i] + " Research Administrator]";
			mes "Hunt 80 of each paired " + .@name$[.@i] + " monster on Biosphere Depth 1.";
			close;
		}
		if (checkquest(.@quest,HUNTING) != 2) {
			mes "[" + .@name$[.@i] + " Research Administrator]";
			mes "The paired hunt is not complete yet. Check your quest window.";
			close;
		}
		if (!checkweight(1001289,3)) {
			mes "[" + .@name$[.@i] + " Research Administrator]";
			mes "Make room for three research documents first.";
			close;
		}
		erasequest .@quest;
		setquest .@wait;
		getitem 1001289,3;
		mes "[" + .@name$[.@i] + " Research Administrator]";
		mes "Hunt verified. You received 3 Depth 1 Research Documents.";
		close;
	default:
		close;
	}
}

// Research documents become reputation at a fixed 2:3 ratio.
ba_in01,292,104,4	script	Depth Research Administrator#bio_d1	4_EP17_MASTER_A,{
	mes "[Depth Research Administrator]";
	if (!callfunc("F_BioDepthQuestAccess"))
		close;

	mes "I exchange every 2 Depth 1 Research Documents for 3 research reputation.";
	mes "Current reputation: ^3366FF" + .@rep + " / 5000^000000";
	if (.@rep >= 5000) {
		mes "Your Depth 1 research reputation is already at its maximum.";
		close;
	}
	.@max = countitem(1001289) / 2;
	.@limit = (5000 - .@rep + 2) / 3;
	if (.@max > .@limit)
		.@max = .@limit;
	if (.@max < 1) {
		mes "Bring at least 2 Depth 1 Research Documents.";
		close;
	}
	next;
	mes "[Depth Research Administrator]";
	mes "How many pairs should I process? Maximum: " + .@max + ".";
	input .@amount,1,.@max;
	delitem 1001289,.@amount * 2;
	.@gain = .@amount * 3;
	if (.@gain > 5000 - .@rep)
		.@gain = 5000 - .@rep;
	add_reputation_points REPUTATION_BIOSPHERE_DEPTH1,.@gain;
	mes "Research registered. You gained ^33AA33" + .@gain + " Depth 1 reputation^000000.";
	close;
}

// Crown transaction guard. Arguments: explicit ID array, mode (0 capture,
// 1 validate, 2 mutate), caller-local numeric snapshot, caller-local UID string,
// synthetic card index (1 or 3), new enchant. Capture MUST precede dialogue
// suspension. Validation never replaces that expectation with current values.
// Numeric snapshot: inventory index/id/amount/equip/refine/identify/attribute/
// grade, four cards, expiry/binding/favorite, and all five option triples.
function	script	F_BioD2ModifyCrown	{
	.@mode = getarg(1);
	if (.@mode < 0 || .@mode > 2)
		return 0;
	getinventorylist;
	.@idx = -1;
	for (.@i = 0; .@i < @inventorylist_count; ++.@i) {
		if ((@inventorylist_equip[.@i] & EQP_HEAD_TOP) && inarray(getarg(0),@inventorylist_id[.@i]) >= 0) {
			.@idx = .@i;
			break;
		}
	}
	if (.@idx < 0)
		return 0;
	.@uid$ = @inventorylist_uniqueid$[.@idx];
	// Normal non-stackable equipment has a generated UID. A legacy zero UID
	// cannot distinguish an otherwise identical replacement: refuse, not guess.
	if (.@uid$ == "0" || .@uid$ == "" || @inventorylist_amount[.@idx] != 1 || !@inventorylist_identify[.@idx])
		return 0;
	setarray .@seen[0],@inventorylist_idx[.@idx],@inventorylist_id[.@idx],@inventorylist_amount[.@idx],@inventorylist_equip[.@idx],@inventorylist_refine[.@idx],@inventorylist_identify[.@idx],@inventorylist_attribute[.@idx],@inventorylist_enchantgrade[.@idx],@inventorylist_card1[.@idx],@inventorylist_card2[.@idx],@inventorylist_card3[.@idx],@inventorylist_card4[.@idx],@inventorylist_expire[.@idx],@inventorylist_bound[.@idx],@inventorylist_favorite[.@idx];
	for (.@i = 1; .@i <= 5; ++.@i) {
		.@offset = 15 + (.@i - 1) * 3;
		.@seen[.@offset] = getd("@inventorylist_option_id" + .@i + "[" + .@idx + "]");
		.@seen[.@offset+1] = getd("@inventorylist_option_value" + .@i + "[" + .@idx + "]");
		.@seen[.@offset+2] = getd("@inventorylist_option_parameter" + .@i + "[" + .@idx + "]");
	}
	if (.@mode == 0) {
		copyarray getarg(2),.@seen[0],30;
		set getarg(3),.@uid$;
		return 1;
	}
	copyarray .@expected[0],getarg(2),30;
	if (.@uid$ != getarg(3))
		return 0;
	for (.@i = 0; .@i < 30; ++.@i)
		if (.@seen[.@i] != .@expected[.@i])
			return 0;
	if (.@mode == 1)
		return 1;
	.@slot = getarg(4);
	if (.@slot != 1 && .@slot != 3)
		return 0;
	// Even a same-result roll reaches native validation, with original values.
	return modifyequipitem(EQI_HEAD_TOP,.@expected[1],0,-1,.@slot,.@expected[8+.@slot],getarg(5));
}

		goto L_StatReroll;
	default:
		close;
	}

L_Crown:
	mes "[Abyss Researcher]";
	mes "Every crown costs 50 Abyss Magic Jewels, 50 Time Dimension Magic Jewels, and 75 Temporal Gemstones.";
	next;
	// Keep every old selection, including Cancel at 19, at its original index.
	.@crown = select("Dragon Knight:Imperial Guard:Meister:Biolo:Shadow Cross:Abyss Chaser:Arch Mage:Elemental Master:Cardinal:Inquisitor:Windhawk:Troubadour / Trouvere:Shinkiro / Shiranui:Night Watch:Sky Emperor:Soul Ascetic:Hyper Novice:Spirit Handler:Cancel:Alitea") - 1;
	if (.@crown == 18)
		close;
	if (.@crown == 19)
		.@crown = 18;
	.@result = .@crown_ids[.@crown];
	mes "[Abyss Researcher]";
	mes "Create item ^3366FF" + .@result + "^000000?";
	next;
	if (select("Craft it.:Cancel.") == 2)
		close;
	if (!callsub(S_Access,500)) {
		mes "Your Biosphere access or reputation changed. No materials were used.";
		close;
	}
	if (countitem(1001552) < 50 || countitem(1001553) < 50 || countitem(25865) < 75) {
		mes "[Abyss Researcher]";
		mes "You do not have all required materials.";
		close;
	}
	if (!checkweight(.@result,1)) {
		mes "Make room for the crown first.";
		close;
	}
	delitem 1001552,50;
	delitem 1001553,50;
	delitem 25865,75;
	getitem .@result,1;
	mes "[Abyss Researcher]";
	mes "Crown crafting complete.";
	close;

L_RandomUpgrade:
	if (!callfunc("F_BioD2ModifyCrown",.@crown_ids[0],0,.@crown_snapshot[0],.@crown_uid$)) {
		mes "[Abyss Researcher]";
		mes "Equip an identified Time Dimensions Rune Crown with a valid unique ID to upgrade.";
		close;
	}
	.@current = .@crown_snapshot[9];
	setarray .@base[0],312719,312729,312739,312749,312759,312769,312779,312789,314249,314259;
	.@family = -1;
	for (.@i = 0; .@i < 10; ++.@i) {
		if (.@current >= .@base[.@i] && .@current <= .@base[.@i] + 9) {
			.@family = .@i;
			.@level = .@current - .@base[.@i] + 1;
			break;
		}
	}
	if (.@family < 0) {
		mes "[Abyss Researcher]";
		mes "The equipped crown does not have a supported special jewel in slot 2.";
		close;
	}
	if (.@level >= 10) {
		mes "[Abyss Researcher]";
		mes "That special jewel is already level 10.";
		close;
	}
	setarray .@chance[1],80,65,50,35,25,20,10,7,5;
	setarray .@cost[1],5,10,20,35,55,80,110,145,185;
	mes "[Abyss Researcher]";
	mes "Level " + .@level + " -> " + (.@level + 1) + ": ^3366FF" + .@chance[.@level] + "% success^000000, " + .@cost[.@level] + " Time Dimension Magic Runes.";
	if (.@level == 1)
		mes "Failure keeps the enchant at level 1.";
	else
		mes "Failure downgrades the enchant by one level.";
	next;
	if (select("Attempt upgrade.:Cancel.") == 2)
		close;
	if (!callsub(S_Access,1500)) {
		mes "Your Biosphere access or reputation changed. No materials were used.";
		close;
	}
	if (!callfunc("F_BioD2ModifyCrown",.@crown_ids[0],1,.@crown_snapshot[0],.@crown_uid$)) {
		mes "The selected crown changed. The upgrade was cancelled without using materials.";
		close;
	}
	if (countitem(1001556) < .@cost[.@level]) {
		mes "[Abyss Researcher]";
		mes "You do not have enough Time Dimension Magic Runes.";
		close;
	}
	if (rand(100) < .@chance[.@level]) {
		.@new = .@current + 1;
		.@success = 1;
	} else {
		.@new = (.@level > 1) ? .@current - 1 : .@current;
		.@success = 0;
	}
	if (!callfunc("F_BioD2ModifyCrown",.@crown_ids[0],2,.@crown_snapshot[0],.@crown_uid$,1,.@new)) {
		mes "[Abyss Researcher]";
		mes "The crown could not be updated. No materials were used.";
		close;
	}
	// Current synchronous equipment/status callback content is regression-
	// audited not to change these materials or access. No yield after success:
	// deduct once, avoiding any potentially lossy delete-then-refund path.
	delitem 1001556,.@cost[.@level];
	mes "[Abyss Researcher]";
	if (.@success)
		mes "Upgrade successful. The special jewel is now level " + (.@level + 1) + ".";
	else if (.@level > 1)
		mes "Upgrade failed. The special jewel fell to level " + (.@level - 1) + ".";
	else
		mes "Upgrade failed. The level 1 jewel was preserved.";
	close;

L_StatReroll:
	if (!callfunc("F_BioD2ModifyCrown",.@crown_ids[0],0,.@crown_snapshot[0],.@crown_uid$)) {
		mes "[Abyss Researcher]";
		mes "Equip an identified Time Dimensions Rune Crown with a valid unique ID to reroll.";
		close;
	}
	.@old = .@crown_snapshot[11];
	setarray .@stat_id[0],4700,4701,4702,4703,4704,4710,4711,4712,4713,4714,4740,4741,4742,4743,4744,4750,4751,4752,4753,4754;
	.@valid = 0;
	for (.@i = 0; .@i < 20; ++.@i) {
		if (.@old == .@stat_id[.@i]) {
			.@valid = 1;
			break;
		}
	}
	if (!.@valid) {
		mes "[Abyss Researcher]";
		mes "The equipped crown does not have a supported stat enchant in slot 4.";
		close;
	}
	mes "[Abyss Researcher]";
	mes "Rerolling only slot 4 costs 180 Abyss Magic Runes. The published normal-enchant probabilities are used.";
	next;
	if (select("Reroll slot 4.:Cancel.") == 2)
		close;
	if (!callsub(S_Access,1500)) {
		mes "Your Biosphere access or reputation changed. No materials were used.";
		close;
	}
	if (!callfunc("F_BioD2ModifyCrown",.@crown_ids[0],1,.@crown_snapshot[0],.@crown_uid$)) {
		mes "The selected crown changed. The reroll was cancelled without using materials.";
		close;
	}
	if (countitem(1001555) < 180) {
		mes "[Abyss Researcher]";
		mes "You need 180 Abyss Magic Runes.";
		close;
	}
	setarray .@stat_weight[0],11500,8000,3500,1500,500,11500,8000,3500,1500,500,11500,8000,3500,1500,500,11500,8000,3500,1500,500;
	.@roll = rand(100000);
	.@total = 0;
	for (.@i = 0; .@i < 20; ++.@i) {
		.@total += .@stat_weight[.@i];
		if (.@roll < .@total) {
			.@new = .@stat_id[.@i];
			break;
		}
	}
	if (!callfunc("F_BioD2ModifyCrown",.@crown_ids[0],2,.@crown_snapshot[0],.@crown_uid$,3,.@new)) {
		mes "[Abyss Researcher]";
		mes "The crown could not be updated. No materials were used.";
		close;
	}
	// Same audited synchronous callback invariant as the jewel upgrade above.
	delitem 1001555,180;
	mes "[Abyss Researcher]";
	mes "Slot 4 was rerolled successfully.";
	close;

// Pure current-state authorization: safe to call after close2 as well as before
// a costed operation. Reputation is a threshold and is never spent here.
S_Access:
	return BaseLevel >= 250 && ep17_2_main >= 33 && get_reputation_points(REPUTATION_BIOSPHERE_DEPTH1) >= 2000 && get_reputation_points(REPUTATION_BIOSPHERE_DEPTH2) >= getarg(0);

L_Convert:
	mes "[Abyss Researcher]";
	.@recipe = select("10 Abyss Jewel Fragments + 10,000z -> Abyss Magic Jewel:10 Time Dimension Jewel Fragments + 10,000z -> Time Dimension Magic Jewel:10 of each Magic Jewel + 20,000z -> Abyss Magic Rune Ore:5 Rune Ore + 5 Temporal Crystals + 30,000z -> Abyss Magic Rune:5 Abyss Runes + spell materials + 50,000z -> Time Dimension Magic Rune:Leave");
	if (.@recipe == 6)
		close;
	switch(.@recipe) {
	case 1:
		setarray .@mat[0],1001550;
		setarray .@need[0],10;
		.@matcount = 1; .@result = 1001552; .@cost = 10000;
		break;
	case 2:
		setarray .@mat[0],1001551;
		setarray .@need[0],10;
		.@matcount = 1; .@result = 1001553; .@cost = 10000;
		break;
	case 3:
		setarray .@mat[0],1001552,1001553;
		setarray .@need[0],10,10;
		.@matcount = 2; .@result = 1001554; .@cost = 20000;
		break;
	case 4:
		setarray .@mat[0],1001554,6607;
		setarray .@need[0],5,5;
		.@matcount = 2; .@result = 1001555; .@cost = 30000;
		break;
	case 5:
		setarray .@mat[0],1001555,6608,6755,25866;
		setarray .@need[0],5,5,5,3;
		.@matcount = 4; .@result = 1001556; .@cost = 50000;
		break;
	}
	.@max = Zeny / .@cost;
	// One getitem output must fit the native MAX_AMOUNT (30000). Input totals
	// may be larger across stacks; each debit is safely chunked below.
	if (.@max > 30000)
		.@max = 30000;
	for (.@i = 0; .@i < .@matcount; ++.@i) {
		.@have = countitem(.@mat[.@i]) / .@need[.@i];
		if (.@max > .@have)
			.@max = .@have;
	}
	if (.@max < 1) {
		mes "[Abyss Researcher]";
		mes "You do not have all required materials and Zeny.";
		close;
	}
	mes "[Abyss Researcher]";
	mes "How many should I create? Maximum: " + .@max + ".";
	// input clamps out-of-range values; a nonzero status must not silently
	// convert that clamped quantity into an unintended transaction.
	if (input(.@amount,1,.@max) != 0)
		close;
	// Revalidate the original conversion access policy after its final yield.
	// Unlike crown services, conversion has no Depth 2 reputation threshold.
	if (strcharinfo(3) != "ba_chess" || BaseLevel < 250 || ep17_2_main < 33 || get_reputation_points(REPUTATION_BIOSPHERE_DEPTH1) < 2000) {
		mes "Your Biosphere access changed. No materials or Zeny were used.";
		close;
	}
	.@total = .@cost * .@amount;
	if (Zeny < .@total) {
		mes "Your materials or Zeny changed. Please try again.";
		close;
	}
	for (.@i = 0; .@i < .@matcount; ++.@i) {
		if (countitem(.@mat[.@i]) < .@need[.@i] * .@amount) {
			mes "Your materials or Zeny changed. Please try again.";
			close;
		}
	}
	// Preserve the conservative pre-payment weight policy: do not credit the
	// weight or slots that consuming materials would release.
	if (Weight + getiteminfo(.@result,ITEMINFO_WEIGHT) * .@amount > MaxWeight) {
		mes "Make room for the converted material first.";
		close;
	}
	// Match the actual plain getitem output, not checkweight's first same-ID
	// approximation. A bound/card/UID-separated first stack may be irrelevant.
	.@slots = getinventoryslots();
	if (.@slots < 1) {
		mes "Your inventory capacity could not be verified. Nothing was used.";
		close;
	}
	getinventorylist;
	.@target = -1;
	.@occupied = 0;
	for (.@i = 0; .@i < @inventorylist_count; ++.@i) {
		if (@inventorylist_idx[.@i] < .@slots)
			++.@occupied;
		if (@inventorylist_id[.@i] != .@result || @inventorylist_bound[.@i] != 0 || @inventorylist_expire[.@i] != 0 || @inventorylist_uniqueid$[.@i] != "0" || @inventorylist_card1[.@i] != 0 || @inventorylist_card2[.@i] != 0 || @inventorylist_card3[.@i] != 0 || @inventorylist_card4[.@i] != 0)
			continue;
		// getinventorylist is in increasing native inventory-index order. The
		// first compatible stack is terminal even if a later stack has room.
		.@target = @inventorylist_idx[.@i];
		.@stack_amount = @inventorylist_amount[.@i];
		break;
	}
	if ((.@target >= 0 && (.@target >= .@slots || .@stack_amount + .@amount > 30000)) || (.@target < 0 && .@occupied >= .@slots)) {
		mes "Make room in the matching output stack or an available inventory slot first.";
		close;
	}
	// Current ba_chess material/achievement callbacks are reviewed as non-
	// mutating. No further suspension is allowed between preflight and commit.
	for (.@i = 0; .@i < .@matcount; ++.@i) {
		.@remaining = .@need[.@i] * .@amount;
		while (.@remaining > 0) {
			.@chunk = (.@remaining > 30000) ? 30000 : .@remaining;
			delitem .@mat[.@i],.@chunk;
			.@remaining -= .@chunk;
		}
	}
	Zeny -= .@total;
	getitem .@result,.@amount;
	mes "Conversion complete. You received " + .@amount + " unit(s) of item ^3366FF" + .@result + "^000000.";
	close;
}
