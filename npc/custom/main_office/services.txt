// PN Main Office: original service hub using existing server rules.
// Map assets are aliases of the owner's standard client maps.
-	script	PN Office Directory	-1,{
	callfunc "PN_OfficeIndex";
	mes "[PN Main Office]";
	mes "Lobby: everyday services and progression.";
	mes "Training: damage lab, repairs and equipment.";
	mes "Upper floor: fashion, clans and lounge.";
	switch(select("Browse all services:Search for a service:Lobby:Training and equipment:Fashion and lounge:Return to Prontera:Leave")) {
	case 3: close2; warp "pn_office",100,40; end;
	case 4: close2; warp "pn_train",50,35; end;
	case 5: close2; warp "pn_style",140,140; end;
	case 6: close2; warp "prontera",150,185; end;
	case 7: close;
	case 2:
		mes "Enter part of a service name, for example card, shadow, fashion or storage.";
		input .@query$;
		if (.@query$ == "") close;
	}
	for (.@i=0; .@i<getarraysize(@PNDeskNames$); .@i++) {
		if (.@query$ != "" && !compare(strtolower(@PNDeskNames$[.@i]),strtolower(.@query$))) continue;
		.@choices[.@count++] = .@i;
		.@menu$ += @PNDeskNames$[.@i]+":";
	}
	if (!.@count) { mes "No matching service. Try a shorter name."; close; }
	.@pick=select(.@menu$+"Cancel")-1;
	if (.@pick == .@count) close;
	.@i=.@choices[.@pick];
	mes "Go to "+@PNDeskNames$[.@i]+"?";
	mes "Map: "+@PNDeskMaps$[.@i]+" ("+@PNDeskX[.@i]+","+@PNDeskY[.@i]+").";
	if (select("Go to the desk:Cancel") == 2) close;
	close2;
	warp @PNDeskMaps$[.@i],@PNDeskX[.@i],@PNDeskY[.@i];
	end;
OnNaviGenerate:
	naviregisterwarp("Directory > Lobby", "pn_office", 100, 40);
	naviregisterwarp("Directory > Training and equipment", "pn_train", 50, 35);
	naviregisterwarp("Directory > Fashion and lounge", "pn_style", 140, 140);
	naviregisterwarp("Directory > Prontera", "prontera", 150, 185);
	end;
}

-	script	PN Office Access	-1,{
	mes "[PN Main Office]";
	mes "Visit our service hub: daily essentials, equipment, training and fashion.";
	if(select("Enter Main Office:Leave") == 2) close;
	close2;
	warp "pn_office",100,40;
	end;
OnNaviGenerate:
	naviregisterwarp("Enter Main Office", "pn_office", 100, 40);
	end;
OnOffice:
	.@map$=strcharinfo(3);
	// A global command NPC has instance_id()==0 even inside an instance.
	// Native instance_generate_mapname always uses the '#' separator.
	if (Hp <= 0 || compare(.@map$,"#") || getmapflag(.@map$,mf_noteleport) || getmapflag(.@map$,mf_nowarp) || getmapflag(.@map$,mf_noreturn)
	 || getmapflag(.@map$,mf_nogo) || getmapflag(.@map$,mf_pvp) || getmapflag(.@map$,mf_gvg)
	 || getmapflag(.@map$,mf_gvg_dungeon) || getmapflag(.@map$,mf_gvg_castle) || getmapflag(.@map$,mf_gvg_te) || getmapflag(.@map$,mf_gvg_te_castle)
	 || getmapflag(.@map$,mf_battleground)) {
		dispbottom "Return to an unrestricted town before using @office."; end;
	}
	warp "pn_office",100,40;
	end;
OnInit:
	bindatcmd "office",strnpcinfo(3)+"::OnOffice",0,99;
	end;
}

-	script	PN Copy Teacher	-1,{
	mes "[Skill Copy Teacher]";
	if (!getskilllv("RG_PLAGIARISM") && !getskilllv("SC_REPRODUCE")) {
		mes "Learn Plagiarism or Reproduce before using this service."; close;
	}
	mes "The normal copy rules apply. Preserve blocks Plagiarism; activate Reproduce to use its copy slot.";
	mes "This replaces a copied skill only if the server accepts the selected skill.";
	setarray .@ids,28,19,14,20,89,79,254,77;
	setarray .@names$,"Heal","Fire Bolt","Cold Bolt","Lightning Bolt","Storm Gust","Magnus Exorcismus","Grand Cross","Turn Undead";
	.@pick=select("Heal:Fire Bolt:Cold Bolt:Lightning Bolt:Storm Gust:Magnus Exorcismus:Grand Cross:Turn Undead:Cancel")-1;
	if (.@pick == 8) close;
	mes "Copy "+.@names$[.@pick]+"? Your learned copy level and skill database determine the result.";
	if(select("Copy skill:Cancel") == 2) close;
	// Every offered skill is copyable by both modes (validated against skill DB).
	// Native eligibility prefers active Reproduce. Its CAST level is val1;
	// the independently learned Plagiarism level must not lower that copy slot.
	.@copycap=getskilllv("RG_PLAGIARISM");
	if (getstatus(SC__REPRODUCE,1)>0 && getskilllv("SC_REPRODUCE"))
		.@copycap=getstatus(SC__REPRODUCE,1);
	if (.@copycap <= 0) { mes "Activate Reproduce or learn Plagiarism first."; close; }
	if (plagiarizeskill(.@ids[.@pick],min(10,.@copycap))) mes "Skill copied.";
	else mes "Copy rejected. Check your copy-skill level, Preserve/Reproduce state and skill eligibility.";
	close;
}

// Advice and status only: these functions never accept/claim quests, reset
// cooldowns, spend currency, give items or move the character.
function	script	PN_GuideNext	{
	if (BaseLevel < 200) {
		mes "Prepare for Base Level 200: choose your job, learn its skills and equip suitable gear.";
		mes "Job Master: lobby 46,40. Druid Mentor: lobby 56,40. Skill Supplies carries consumables; Training has the private damage lab.";
		navigateto "pn_office",46,39;
		return;
	}
	if (!CH1_Complete()) {
		if (!isbegin_quest(18368)) {
			mes "Start Chapter 1 at the Small Ash Tree, prt_fild05 353,252. It requires Base Level 200.";
			navigateto "prt_fild05",353,252;
		} else mes "Continue Chapter 1 in your quest window. Follow the active objective through Bring Down the Air Fortress; Robin's Zero Cell investigation comes afterward.";
		return;
	}
	if (BaseLevel < 225) { mes "Chapter 1 is complete. Reach Base Level 225 to begin Chapter 2 with Guardian El."; return; }
	if (CH2_Complete()) {
		mes "Chapter 2 is complete. Check Daily and repeatable quests for Flame Coins, Biosphere materials and Episode 21 patrols.";
		mes "Equipment goals lists the desks and material sources for your next upgrade.";
		return;
	}
	mes "Chapter 2 stage: "+CH2_Step+" / 12.";
	switch (CH2_Step) {
	case 0:
		mes "Bring 5 Apples to Guardian El. Apples held: "+countitem(512)+".";
		navigateto "ygg_edge",156,179; break;
	case 1: mes "Speak to Guardian El in Veledor."; navigateto "veledor",136,65; break;
	case 2:
		if (CH2_Crossroads) { mes "Crossroads cleared. Report to Guardian El."; navigateto "veledor",136,65; }
		else { mes "Clear Crossroads of Tangled Mana through the gate beside Guardian El."; navigateto "veledor",140,65; }
		break;
	case 3: case 4: mes "Help Karilon: accept his assignment, defeat 20 Shadow Jailers, then report to him."; navigateto "kindlbrk",115,215; break;
	case 5: mes "Meet Orion at the Root of Death."; navigateto "deadroot",136,65; break;
	case 6: case 7: mes "Help the Flame Researcher: defeat the assigned Shadow Jailers and report back."; navigateto "mu_dun01",95,131; break;
	case 8: mes "Bring 5 Apples to the Nyrholt Keeper. Apples held: "+countitem(512)+"."; navigateto "ragsruth",88,58; break;
	case 9:
		if (CH2_Nyrholt) mes "Nyrholt cleared. Report to the Keeper for the next assignment.";
		else mes "Enter Nyrholt through the Keeper and defeat the guardian.";
		navigateto "ragsruth",88,58; break;
	case 10: mes "Defeat 100 Folnir creatures in Distorted Ragsroot, then report to the Scout."; navigateto "rgs_dun1",95,131; break;
	case 11:
		if (CH2_Phantom) mes "The Phantom is defeated. Speak to the Phantom Gate to record the clear and return to Guardian El.";
		else mes "Enter the Phantom Gate and defeat the Phantom of Nyrholt.";
		navigateto "ch2safe4",86,146; break;
	case 12: mes "Report to Guardian El to complete Chapter 2 and claim its reward."; navigateto "veledor",136,65; break;
	default: mes "Your stage is not recognized. Keep your quest state and ask a GM to inspect it.";
	}
	return;
}

function	script	PN_GuideDailies	{
	switch(select("Biosphere habitat hunts:Phantom of Nyrholt:Raised Land patrol:Leave")) {
	case 1:
		if (BaseLevel < 240) { mes "Biosphere hunts unlock at Base Level 240. Your level: "+BaseLevel+"."; return; }
		setarray .@quest,17608,17609,17610,17611,900100,900102,900104;
		setarray .@wait,17612,17613,17614,17615,900101,900103,900105;
		setarray .@names$,"Severe Cold","Fire","Grassland","Death","Soul","Venom","Temple";
		for (.@i=0; .@i<7; ++.@i) {
			.@status$="Available";
			if (isbegin_quest(.@wait[.@i]) && checkquest(.@wait[.@i],PLAYTIME) != 2) .@status$="On cooldown until 04:00";
			else if (isbegin_quest(.@quest[.@i])) {
				if (checkquest(.@quest[.@i],HUNTING) == 2) .@status$="Ready to report";
				else .@status$="In progress - see quest kill count";
			}
			mes .@names$[.@i]+": "+.@status$;
		}
		mes "Each hunt: 100 kills, 5 habitat runes and 2 essences. Research Admin Beta checks the final reward and capacity.";
		navigateto "ba_in01",252,353;
		break;
	case 2:
		if (!CH2_Complete()) { mes "Complete Chapter 2 to unlock the repeatable Phantom expedition."; return; }
		.@state=isbegin_quest(27119);
		.@seconds=max(0,CH2_Daily_CD_27119-gettimetick(2));
		if (.@state == 2 && .@seconds) mes "On cooldown: "+((.@seconds+59)/60)+" minutes remaining.";
		else if (.@state == 1 && CH2_DailyPhantom) mes "Ready to report: claim 3 Flame Coins at the Phantom Gate.";
		else if (.@state == 1) mes "In progress: defeat the Phantom, then report to the gate.";
		else mes "Available: accept the repeatable expedition at the Phantom Gate.";
		mes "The four-hour timer starts when you claim the clear reward.";
		navigateto "ch2safe4",86,146;
		break;
	case 3:
		if (!callfunc("EP21_GimliComplete")) { mes "Complete the Episode 21 Gimli report first."; return; }
		if (!EP21_LugenburgDone && isbegin_quest(17776) != 2) { mes "Finish the Lugenburg Brothers side story with Wilhelm, Reinhardt and Tris."; navigateto "jor_mbase",140,201; return; }
		if (EP21_RaisedDaily == callfunc("EP21_DailyKey")) mes "Today's patrol is complete. It resets at 04:00 server time.";
		else if (EP21_RaisedQuest == 17777 || EP21_RaisedQuest == 17778) {
			if (checkquest(EP21_RaisedQuest,HUNTING) == 2) mes "Ready to report: return to Wilhelm for 10 vouchers and 10 reputation.";
			else mes "In progress: defeat 300 monsters on the assigned Raised Land map.";
		} else mes "Available: choose a northern or southern Raised Land patrol with Wilhelm.";
		navigateto "jor_mbase",140,201;
	}
	return;
}

-	script	PN Build Notes	-1,{
	mes "[Build Notebook]";
	mes "Keep three character-specific build notes for comparison. This notebook does not change stats, skills or equipment.";
	.@slot=select("Notebook 1:Notebook 2:Notebook 3:Leave")-1;
	if (.@slot == 3) close;
	if (PNBuildNote$[.@slot] != "") {
		mes PNBuildNote$[.@slot];
		mes PNBuildStats$[.@slot];
	}
	if(select("Save current stats with a note:Leave") == 2) close;
	mes "Enter a short build or rotation note (up to 120 characters).";
	input .@note$;
	if (.@note$ == "" || getstrlen(.@note$)>120) { mes "Please use 1-120 characters."; close; }
	if(PNBuildNote$[.@slot] != "" && select("Replace this notebook:Cancel") == 2) close;
	PNBuildNote$[.@slot]=.@note$;
	PNBuildStats$[.@slot]="STR "+readparam(bStr)+" AGI "+readparam(bAgi)+" VIT "+readparam(bVit)+" INT "+readparam(bInt)+" DEX "+readparam(bDex)+" LUK "+readparam(bLuk);
	mes "Saved. Use the private damage lab with consistent targets to compare your builds.";
	close;
}
