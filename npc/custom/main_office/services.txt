// PN Main Office: original service hub using existing server rules.
// Map assets are aliases of the owner's standard client maps.
-	script	PN Office Directory	-1,{
	callfunc "PN_OfficeIndex";
	mes "[PN Main Office]";
	mes "Lobby: everyday services and progression.";
	mes "Training: damage lab, repairs and equipment.";
	mes "Upper floor: fashion, clans and lounge.";
	switch(select("Browse all services:Search for a service:Lobby:Training and equipment:Fashion and lounge:Return to Prontera:Leave")) {
	case 3: close2; warp "pn_office",100,40; end;
	case 4: close2; warp "pn_train",50,35; end;
	case 5: close2; warp "pn_style",140,140; end;
	case 6: close2; warp "prontera",150,185; end;
	case 7: close;
	case 2:
		mes "Enter part of a service name, for example card, shadow, fashion or storage.";
		input .@query$;
		if (.@query$ == "") close;
	}
	for (.@i=0; .@i<getarraysize(@PNDeskNames$); .@i++) {
		if (.@query$ != "" && !compare(strtolower(@PNDeskNames$[.@i]),strtolower(.@query$))) continue;
		.@choices[.@count++] = .@i;
		.@menu$ += @PNDeskNames$[.@i]+":";
	}
	if (!.@count) { mes "No matching service. Try a shorter name."; close; }
	.@pick=select(.@menu$+"Cancel")-1;
	if (.@pick == .@count) close;
	.@i=.@choices[.@pick];
	mes "Go to "+@PNDeskNames$[.@i]+"?";
	mes "Map: "+@PNDeskMaps$[.@i]+" ("+@PNDeskX[.@i]+","+@PNDeskY[.@i]+").";
	if (select("Go to the desk:Cancel") == 2) close;
	close2;
	warp @PNDeskMaps$[.@i],@PNDeskX[.@i],@PNDeskY[.@i];
	end;
}

-	script	PN Office Access	-1,{
	mes "[PN Main Office]";
	mes "Visit our service hub: daily essentials, equipment, training and fashion.";
	if(select("Enter Main Office:Leave") == 2) close;
	close2;
	warp "pn_office",100,40;
	end;
OnOffice:
	.@map$=strcharinfo(3);
	// A global command NPC has instance_id()==0 even inside an instance.
	// Native instance_generate_mapname always uses the '#' separator.
	if (Hp <= 0 || compare(.@map$,"#") || getmapflag(.@map$,mf_noteleport) || getmapflag(.@map$,mf_nowarp) || getmapflag(.@map$,mf_noreturn)
	 || getmapflag(.@map$,mf_nogo) || getmapflag(.@map$,mf_pvp) || getmapflag(.@map$,mf_gvg)
	 || getmapflag(.@map$,mf_gvg_dungeon) || getmapflag(.@map$,mf_gvg_castle) || getmapflag(.@map$,mf_gvg_te) || getmapflag(.@map$,mf_gvg_te_castle)
	 || getmapflag(.@map$,mf_battleground)) {
		dispbottom "Return to an unrestricted town before using @office."; end;
	}
	warp "pn_office",100,40;
	end;
OnInit:
	bindatcmd "office",strnpcinfo(3)+"::OnOffice",0,99;
	end;
}

-	script	PN Copy Teacher	-1,{
	mes "[Skill Copy Teacher]";
	if (!getskilllv("RG_PLAGIARISM") && !getskilllv("SC_REPRODUCE")) {
		mes "Learn Plagiarism or Reproduce before using this service."; close;
	}
	mes "The normal copy rules apply. Preserve blocks Plagiarism; activate Reproduce to use its copy slot.";
	mes "This replaces a copied skill only if the server accepts the selected skill.";
	setarray .@ids,28,19,14,20,89,79,254,77;
	setarray .@names$,"Heal","Fire Bolt","Cold Bolt","Lightning Bolt","Storm Gust","Magnus Exorcismus","Grand Cross","Turn Undead";
	.@pick=select("Heal:Fire Bolt:Cold Bolt:Lightning Bolt:Storm Gust:Magnus Exorcismus:Grand Cross:Turn Undead:Cancel")-1;
	if (.@pick == 8) close;
	mes "Copy "+.@names$[.@pick]+"? Your learned copy level and skill database determine the result.";
	if(select("Copy skill:Cancel") == 2) close;
	// Every offered skill is copyable by both modes (validated against skill DB).
	// Native eligibility prefers active Reproduce. Its CAST level is val1;
	// the independently learned Plagiarism level must not lower that copy slot.
	.@copycap=getskilllv("RG_PLAGIARISM");
	if (getstatus(SC__REPRODUCE,1)>0 && getskilllv("SC_REPRODUCE"))
		.@copycap=getstatus(SC__REPRODUCE,1);
	if (.@copycap <= 0) { mes "Activate Reproduce or learn Plagiarism first."; close; }
	if (plagiarizeskill(.@ids[.@pick],min(10,.@copycap))) mes "Skill copied.";
	else mes "Copy rejected. Check your copy-skill level, Preserve/Reproduce state and skill eligibility.";
	close;
}

-	script	PN Build Notes	-1,{
	mes "[Build Notebook]";
	mes "Keep three character-specific build notes for comparison. This notebook does not change stats, skills or equipment.";
	.@slot=select("Notebook 1:Notebook 2:Notebook 3:Leave")-1;
	if (.@slot == 3) close;
	if (PNBuildNote$[.@slot] != "") {
		mes PNBuildNote$[.@slot];
		mes PNBuildStats$[.@slot];
	}
	if(select("Save current stats with a note:Leave") == 2) close;
	mes "Enter a short build or rotation note (up to 120 characters).";
	input .@note$;
	if (.@note$ == "" || getstrlen(.@note$)>120) { mes "Please use 1-120 characters."; close; }
	if(PNBuildNote$[.@slot] != "" && select("Replace this notebook:Cancel") == 2) close;
	PNBuildNote$[.@slot]=.@note$;
	PNBuildStats$[.@slot]="STR "+readparam(bStr)+" AGI "+readparam(bAgi)+" VIT "+readparam(bVit)+" INT "+readparam(bInt)+" DEX "+readparam(bDex)+" LUK "+readparam(bLuk);
	mes "Saved. Use the private damage lab with consistent targets to compare your builds.";
	close;
}
