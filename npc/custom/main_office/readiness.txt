// Read-only guidance. Admission calls the same predicate again after dialogs.
// Empty reason means prerequisites satisfied; it does not reserve an instance.
function	script	PN_InstanceMissing	{
	.@name$ = getarg(0);
	if (.@name$ == "Crossroads of Tangled Mana") {
		if (CH2_Step != 2) return "Missing prerequisite: Chapter 2 stage 2.";
		if (CH2_Crossroads) return "Ready to report: Guardian El in Veledor.";
	} else if (.@name$ == "Nyrholt") {
		if (CH2_Step != 8 && CH2_Step != 9) return "Missing prerequisite: Chapter 2 stage 8 or 9.";
		if (CH2_Nyrholt) return "Ready to report: Nyrholt Keeper.";
		if (CH2_Step == 8 && countitem(512) < 5) return "Missing prerequisite: 5 Apples for the Keeper.";
	} else if (.@name$ == "Phantom of Nyrholt") {
		if (!callfunc("CH2_Complete")) {
			if (CH2_Step != 11) return "Missing prerequisite: Chapter 2 stage 11.";
			if (CH2_Phantom) return "Ready to report: Phantom Gate, then Guardian El.";
		} else {
			if (isbegin_quest(27119) == 2 && CH2_Daily_CD_27119 > gettimetick(2)) return "Cooldown: "+((CH2_Daily_CD_27119-gettimetick(2)+59)/60)+" minutes remain.";
			if (isbegin_quest(27119) == 1 && CH2_DailyPhantom) return "Ready to report: claim 3 Flame Coins at the Phantom Gate.";
		}
	} else if (.@name$ == "Ghost Palace") {
		if (BaseLevel < 120) return "Missing prerequisite: Base Level 120.";
		if (checkquest(1261,PLAYTIME) == 0 || checkquest(1261,PLAYTIME) == 1) return "Cooldown: Cursed Swordsman; "+((max(0,pnquestremaining(1261))+59)/60)+" minutes remain.";
		if (checkquest(1261,PLAYTIME) == 2) return "Timer expired: speak to the Royal Guard to clear it, then speak again.";
		if (!is_party_leader()) return "Missing prerequisite: only the party leader can enter Ghost Palace.";
	} else if (.@name$ == "Charleston in Distress") {
		if (BaseLevel < 130) return "Missing prerequisite: Base Level 130.";
		if (isbegin_quest(13184) != 1) return "Missing prerequisite: accept Charleston's assignment at verus04 75,114.";
		if (checkquest(13185,PLAYTIME) == 0 || checkquest(13185,PLAYTIME) == 1) return "Cooldown: Charleston; "+((max(0,pnquestremaining(13185))+59)/60)+" minutes remain (20-hour timer).";
		if (checkquest(13185,PLAYTIME) == 2) return "Timer expired: speak to the Machine to clear it, then speak again.";
	} else if (.@name$ == "Ominous Dark Whisper") {
		if (isbegin_quest(12663)==1) return "Missing prerequisite: finish Est's infiltration briefing at hem_dun01 209,248.";
		if (isbegin_quest(12666)!=1 && isbegin_quest(12664)!=1) return "Missing prerequisite: Chapter 1 Sky Fortress infiltration assignment.";
	} else if (.@name$ == "Simulated Dark Whisper") {
		if (isbegin_quest(12660)!=2) return "Missing prerequisite: complete Chapter 1 and register with the Wizard Professor at hem_dun01 205,248.";
		if (checkquest(12661,PLAYTIME)==0 || checkquest(12661,PLAYTIME)==1) return "Cooldown: simulation; "+((max(0,pnquestremaining(12661))+59)/60)+" minutes remain (76-hour timer).";
		if (checkquest(12661,PLAYTIME)==2) return "Timer expired: speak to the Wizard Professor to clear it.";
	} else return "Readiness information is unavailable for this instance.";
	if (!getcharid(1)) return "Missing prerequisite: form or join a party; one member is enough.";
	.@iid = instance_id(IM_PARTY);
	if (.@iid) {
		if (instance_live_info(ILI_NAME,.@iid) != .@name$) return "Reserved elsewhere: finish your party's "+instance_live_info(ILI_NAME,.@iid)+" first.";
		if (.@name$ == "Crossroads of Tangled Mana" || .@name$ == "Nyrholt" || .@name$ == "Phantom of Nyrholt") {
			if (!compare(getinstancevar('ch2_roster$,.@iid),","+getcharid(0)+",")) return "Missing prerequisite: you were not on this expedition's original roster.";
		}
	} else if (!is_party_leader()) return "Missing prerequisite: ask your party leader to reserve this instance.";
	return "";
}

function	script	PN_InstanceReadinessDesk	{
	setarray .@names$,"Crossroads of Tangled Mana","Nyrholt","Phantom of Nyrholt","Ghost Palace","Charleston in Distress","Ominous Dark Whisper","Simulated Dark Whisper";
	setarray .@maps$,"veledor","ragsruth","ch2safe4","dali02","verus04","hem_dun01","hem_dun01";
	setarray .@x,140,88,86,43,81,209,205;
	setarray .@y,65,58,146,129,112,248,248;
	setarray .@tags$,"Flame Coins (story)","Flame Coins (story)","Flame Coins (story/daily)","Gray Shards / Thanatos equipment","Charleston Components","Chapter 1 story finale","Chapter 1 repeatable simulation";
	.@pick=select("Crossroads of Tangled Mana:Nyrholt:Phantom of Nyrholt:Ghost Palace:Charleston in Distress:Ominous Dark Whisper:Simulated Dark Whisper:Leave")-1;
	if (.@pick == 7) return;
	mes .@names$[.@pick]+" | Materials: "+.@tags$[.@pick];
	.@reason$=callfunc("PN_InstanceMissing",.@names$[.@pick]);
	if (.@reason$ != "") mes .@reason$;
	else if (instance_id(IM_PARTY)) mes "Reserved: your party's instance is ready for entry checks.";
	else mes "Ready: visit the entrance to reserve and enter.";
	mes "Entrance: "+.@maps$[.@pick]+" "+.@x[.@pick]+","+.@y[.@pick]+". The entrance rechecks your state.";
	navigateto .@maps$[.@pick],.@x[.@pick],.@y[.@pick];
	if (getcharid(1) && select("Check every party member:Done")==1) callfunc "PN_PartyReadiness",.@names$[.@pick];
	return;
}

// Take one synchronous snapshot. Never force attachment to a member already
// using an NPC, never open a dialog on that member, and restore the caller before
// displaying results. Remote map-server sessions cannot be inspected here.
function	script	PN_PartyReadiness	{
	.@name$=getarg(0); .@owner=getcharid(3); .@ownerchar=getcharid(0); .@party=getcharid(1);
	if (!.@party) { mes "Join a party to check its members."; return; }
	.@count=getpartymember(.@party,0,.@names$);
	getpartymember .@party,1,.@chars;
	getpartymember .@party,2,.@accounts;
	for (.@i=0; .@i<.@count; .@i++) {
		.@result$[.@i]="Unavailable: offline or on another map server.";
		if (!isloggedin(.@accounts[.@i],.@chars[.@i])) continue;
		if (.@accounts[.@i]!=.@owner && !attachrid(.@accounts[.@i],0)) {
			.@result$[.@i]="Unavailable: currently using another NPC."; continue;
		}
		if (getcharid(0)!=.@chars[.@i] || getcharid(1)!=.@party) .@result$[.@i]="Unavailable: party membership changed.";
		else {
			.@result$[.@i]=callfunc("PN_InstanceMissing",.@name$);
			if (.@result$[.@i]=="") .@result$[.@i]=(instance_id(IM_PARTY))?"Ready: matching reservation and reentry checks passed.":"Ready to reserve at the entrance.";
		}
		if (.@accounts[.@i]!=.@owner && !attachrid(.@owner,0)) end;
	}
	if (getcharid(3)!=.@owner || getcharid(0)!=.@ownerchar) end;
	mes "[Party readiness: "+.@name$+"]";
	if (!.@count) mes "Party details are unavailable. Try again shortly.";
	for (.@i=0; .@i<.@count; .@i++) mes .@names$[.@i]+": "+.@result$[.@i];
	mes "Checked now. The entrance checks each member again before entry. Unavailable members must check from their own session.";
	return;
}

function	script	PN_GuideOnboarding	{
	if (BaseLevel < 200) {
		mes "Next objective: reach Base Level 200 for Chapter 1. First review your job and skills with the Job Master.";
		mes "Destination: Main Office lobby 46,40. Training and equipment services are upstairs.";
		navigateto "pn_office",46,39;
		return;
	}
	if (!isbegin_quest(18368)) {
		mes "Next objective: accept Lara's invitation at the Small Ash Tree. Chapter 1 begins at Base Level 200.";
		navigateto "prt_fild05",353,252; return;
	}
	if (callfunc("PN_CH1Guide")) return;
	mes "No active Chapter 1 story objective was found. Revisit your last story NPC to finish its conversation. If the quest did not advance, report your last quest title to support.";
	return;
}

function	script	PN_CH1GuideShow	{
	.@q=getarg(0); .@map$=getarg(3); .@x=getarg(4); .@y=getarg(5);
	mes "Chapter 1: "+getarg(1);
	mes "Next objective: "+getarg(2);
	if (.@q==18373) mes "Dragon Scales: "+countitem("Dragon_Scale")+" / 5; missing "+max(0,5-countitem("Dragon_Scale"))+".";
	if (.@q==8964) {
		mes "Broken Petals: "+countitem("Ch1_Broken_Petal")+" / 5; missing "+max(0,5-countitem("Ch1_Broken_Petal"))+".";
		mes (checkquest(8964,HUNTING)==2)?"Shadow Jailer hunt: complete.":"Shadow Jailer hunt: still in progress (10 required).";
	}
	// These destinations are conditional gateways, not ordinary map links.
	if (.@q==18376 && strcharinfo(3)!="ch1_sf01") { .@map$="mu_fild01"; .@x=95; .@y=154; }
	if (.@q==17891 && strcharinfo(3)!="ch1_sf03") { .@map$="hem_fild"; .@x=102; .@y=161; }
	if (.@q==19225 && strcharinfo(3)!="ch1_vrgef2" && strcharinfo(3)!="ch1_vrgef1") { .@map$="ygg_edge"; .@x=211; .@y=211; }
	if (.@q==12666 || .@q==12664) {
		.@reason$=callfunc("PN_InstanceMissing","Ominous Dark Whisper");
		if (.@reason$!="") mes .@reason$;
	}
	mes "Destination: "+.@map$+" "+.@x+","+.@y+".";
	navigateto .@map$,.@x,.@y;
	return 1;
}
