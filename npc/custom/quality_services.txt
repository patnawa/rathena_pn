guild_vs1,50,43,4	script	PN Lab Console	4_M_01,{
	if (!instance_id()) { mes "Use PN Services to create your private lab."; close; }
	// Attached scripts stop on logout. Expire their lock and clean up on re-entry.
	if ('Running && gettimetick(2) >= 'Until) {
		killmonsterall instance_mapname("guild_vs1");
		'Running = 0;
	}
	if ('Running) { mes "A test is already running. Wait up to 75 seconds after an interrupted run."; close; }
	mes "[Private Damage Lab]";
	mes "60-second HP-loss measurement. No rewards. Use identical gear, buffs, element and rotation for comparisons.";
	mes "Target cannot move or attack and is status/knockback immune. This is a controlled dummy, not a full boss encounter.";
	switch(select("Configure and run:Last result:Exit and destroy room")) {
	case 2:
		mes "Last run: " + @PNLabLabel$;
		mes "Damage: " + @PNLabDamage + "; elapsed ms: " + @PNLabMs;
		mes "DPS: " + @PNLabDps;
		close;
	case 3:
		.@id = instance_id();
		if (@PNLabOffice == 2) warp "pn_train",50,35;
		else if (@PNLabOffice == 1) warp "pn_office",100,40;
		else warp "izlude",140,144;
		instance_destroy .@id;
		end;
	}
	mes "Give this run a short label (e.g. Fenrir / Ultra).";
	input .@label$;
	mes "Size:"; .@size = select("Small:Medium:Large")-1;
	mes "Race:"; .@race = select("Formless:Undead:Brute:Plant:Insect:Fish:Demon:Demi-Human:Angel:Dragon")-1;
	mes "Defending element:"; .@element = select("Neutral:Water:Earth:Fire:Wind:Poison:Holy:Dark:Ghost:Undead")-1;
	mes "Element level (1-4):"; input .@elelv,1,4;
	mes "MDEF (0-1000):"; input .@mdef,0,1000;
	mes "MRES (0-1000):"; input .@mres,0,1000;
	mes "Prepare all buffs before continuing. Timing starts after the countdown.";
	mes "Boss-only bonuses are NOT represented by this normal-class dummy.";
	if (select("Start:Cancel")==2) close;
	close2;
	'Running = 1;
	'Until = gettimetick(2)+75;
	.@map$ = instance_mapname("guild_vs1");
	killmonsterall .@map$;
	monster .@map$,50,50,"Damage Lab Target",1002,1;
	.@mob = $@mobid[0];
	setunitdata .@mob,UMOB_MODE,MD_STATUSIMMUNE|MD_KNOCKBACKIMMUNE;
	setunitdata .@mob,UMOB_SIZE,.@size;
	setunitdata .@mob,UMOB_RACE,.@race;
	setunitdata .@mob,UMOB_ELETYPE,.@element;
	setunitdata .@mob,UMOB_ELELEVEL,.@elelv;
	setunitdata .@mob,UMOB_DEF,0;
	setunitdata .@mob,UMOB_MDEF,.@mdef;
	setunitdata .@mob,UMOB_MRES,.@mres;
	setunitdata .@mob,UMOB_FLEE,0;
	setunitdata .@mob,UMOB_MAXHP,2000000000;
	setunitdata .@mob,UMOB_HP,2000000000;
	setunitdata .@mob,UMOB_DMGIMMUNE,1;
	for (.@i=3; .@i>0; .@i--) { dispbottom "Test starts in " + .@i; sleep2 1000; }
	setunitdata .@mob,UMOB_DMGIMMUNE,0;
	.@start = gettimetick(0);
	dispbottom "GO: 60 seconds. Keep equipment and buffs consistent.";
	while (gettimetick(0)-.@start < 60000) {
		sleep2 100;
		if (strcharinfo(3) != .@map$ || !unitexists(.@mob)) { .@invalid=1; break; }
		getunitdata .@mob,.@data;
		.@damage += 2000000000-.@data[UMOB_HP];
		setunitdata .@mob,UMOB_HP,2000000000;
	}
	.@elapsed = gettimetick(0)-.@start;
	if (unitexists(.@mob)) setunitdata .@mob,UMOB_DMGIMMUNE,1;
	killmonsterall .@map$;
	'Running = 0;
	if (.@invalid) { dispbottom "INVALID: left room or target died. No DPS result saved."; end; }
	@PNLabLabel$ = .@label$;
	@PNLabDamage = .@damage;
	@PNLabMs = .@elapsed;
	@PNLabDps = .@damage*1000/max(1,.@elapsed);
	dispbottom "RESULT " + .@label$ + ": damage=" + .@damage + " time_ms=" + .@elapsed + " DPS=" + @PNLabDps;
	logmes "PNLAB label="+.@label$+" damage="+.@damage+" ms="+.@elapsed+" size="+.@size+" race="+.@race+" element="+.@element+" elelv="+.@elelv+" mdef="+.@mdef+" mres="+.@mres;
	end;
OnInstanceInit:
	'Running = 0;
	setmapflag instance_mapname("guild_vs1"),mf_nobranch;
	setmapflag instance_mapname("guild_vs1"),mf_nomemo;
	setmapflag instance_mapname("guild_vs1"),mf_noteleport;
	setmapflag instance_mapname("guild_vs1"),mf_monster_noteleport;
	setmapflag instance_mapname("guild_vs1"),mf_nowarpto;
	setmapflag instance_mapname("guild_vs1"),mf_partylock;
	setmapflag instance_mapname("guild_vs1"),mf_restricted,7;
	setmapflag instance_mapname("guild_vs1"),mf_noexp;
	setmapflag instance_mapname("guild_vs1"),mf_noloot;
	setmapflagnosave instance_mapname("guild_vs1"),"izlude",140,144;
	end;
}
