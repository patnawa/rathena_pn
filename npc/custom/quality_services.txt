guild_vs1,50,43,4	script	PN Lab Console	4_M_01,{
	if (!instance_id()) { mes "Use PN Services to create your private lab."; close; }
	// Attached scripts stop on logout. Expire their lock and clean up on re-entry.
	if ('Running && gettimetick(2) >= 'Until) {
		killmonsterall instance_mapname("guild_vs1");
		'Running = 0;
	}
	if ('Running) { mes "A test is already running. Wait up to 75 seconds after an interrupted run."; close; }
	mes "[Private Damage Lab]";
	mes "60-second HP-loss measurement. No rewards. Use identical gear, buffs, element and rotation for comparisons.";
	mes "Targets cannot move or attack. Both resist knockback; boss targets also resist status effects. Normal targets can receive status effects.";
	switch(select("Configure and run:Load target preset:Saved runs and comparison:Exit and destroy room")) {
	case 2:
		for (.@i=0; .@i<3; .@i++) mes "Preset "+(.@i+1)+": "+((PNLabPresetVersion[.@i]==2)?PNLabPresetName$[.@i]:"Empty");
		.@slot=select("Preset 1:Preset 2:Preset 3:Cancel")-1;
		if (.@slot==3) close;
		if (!callfunc("PN_LabPresetLoad",.@slot)) { mes "This preset is empty or invalid. Configure and save it again."; close; }
		.@loaded=1;
		break;
	case 3:
		callfunc "PN_LabHistory";
		close;
	case 4:
		.@id = instance_id();
		if (@PNLabOffice == 2) warp "pn_train",50,35;
		else if (@PNLabOffice == 1) warp "pn_office",100,40;
		else warp "izlude",140,144;
		instance_destroy .@id;
		end;
	}
	mes "Give this run a short label (e.g. Fenrir / Ultra).";
	input .@label$;
	if (getstrlen(.@label$)<1 || getstrlen(.@label$)>40) { mes "Use a label of 1-40 characters."; close; }
	if (!.@loaded) {
		mes "Combat class:"; @PNLabTarget[0]=select("Normal:Boss")-1;
		mes "Size:"; @PNLabTarget[1]=select("Small:Medium:Large")-1;
		mes "Race:"; @PNLabTarget[2]=select("Formless:Undead:Brute:Plant:Insect:Fish:Demon:Demi-Human:Angel:Dragon")-1;
		mes "Defending element:"; @PNLabTarget[3]=select("Neutral:Water:Earth:Fire:Wind:Poison:Holy:Dark:Ghost:Undead")-1;
		mes "Element level (1-4):"; input @PNLabTarget[4],1,4;
		mes "Physical DEF (0-1000):"; input @PNLabTarget[5],0,1000;
		mes "Physical RES (0-1000):"; input @PNLabTarget[6],0,1000;
		mes "MDEF (0-1000):"; input @PNLabTarget[7],0,1000;
		mes "MRES (0-1000):"; input @PNLabTarget[8],0,1000;
		if (select("Run without saving preset:Save target preset")==2) {
			.@slot=select("Replace preset 1:Replace preset 2:Replace preset 3:Cancel")-1;
			if (.@slot<3) {
				mes "Preset name (1-40 characters):"; input .@preset$;
				if (!callfunc("PN_LabPresetSave",.@slot,.@preset$)) { mes "Invalid preset name or settings."; close; }
			}
		}
	}
	.@setup$=callfunc("PN_LabTargetIdentity");
	if (.@setup$=="") { mes "Invalid target settings. Configure the target again."; close; }
	mes .@setup$;
	mes "Prepare all buffs before continuing. Timing starts after the countdown.";
	mes "Boss-only bonuses apply only to Boss class. This dummy does not reproduce encounter skills or boss phases.";
	if (select("Start:Cancel")==2) close;
	close2;
	'Running = 1;
	'Until = gettimetick(2)+75;
	.@map$ = instance_mapname("guild_vs1");
	killmonsterall .@map$;
	monster .@map$,50,50,"Damage Lab Target",1002,1;
	.@mob = $@mobid[0];
	if (!callfunc("PN_LabApplyTarget",.@mob)) { killmonsterall .@map$; 'Running=0; end; }
	setunitdata .@mob,UMOB_MAXHP,2000000000;
	setunitdata .@mob,UMOB_HP,2000000000;
	setunitdata .@mob,UMOB_DMGIMMUNE,1;

	.@build$=pnlabidentity(2);

// Settings order: class, size, race, element, element level, DEF, RES, MDEF, MRES.
// Character presets contain data only; loading one cannot start or award a run.
function	script	PN_LabTargetValid	{
 if (@PNLabTarget[0]<0 || @PNLabTarget[0]>1 || @PNLabTarget[1]<0 || @PNLabTarget[1]>2
  || @PNLabTarget[2]<0 || @PNLabTarget[2]>9 || @PNLabTarget[3]<0 || @PNLabTarget[3]>9
  || @PNLabTarget[4]<1 || @PNLabTarget[4]>4) return 0;
 for (.@i=5; .@i<9; .@i++) if (@PNLabTarget[.@i]<0 || @PNLabTarget[.@i]>1000) return 0;
 return 1;
}
function	script	PN_LabPresetSave	{
 .@slot=getarg(0); .@name$=getarg(1);
 if (.@slot<0 || .@slot>2 || getstrlen(.@name$)<1 || getstrlen(.@name$)>40 || !callfunc("PN_LabTargetValid")) return 0;
 PNLabPresetVersion[.@slot]=0;
 for (.@i=0; .@i<9; .@i++) PNLabPreset[.@slot*9+.@i]=@PNLabTarget[.@i];
 PNLabPresetName$[.@slot]=.@name$;
 PNLabPresetVersion[.@slot]=2;
 return 1;
}
function	script	PN_LabPresetLoad	{
 .@slot=getarg(0);
 if (.@slot<0 || .@slot>2 || PNLabPresetVersion[.@slot]!=2) return 0;
 for (.@i=0; .@i<9; .@i++) @PNLabTarget[.@i]=PNLabPreset[.@slot*9+.@i];
 return callfunc("PN_LabTargetValid");
}
function	script	PN_LabTargetIdentity	{
 if (!callfunc("PN_LabTargetValid")) return "";
 return "v2/60s/"+((@PNLabTarget[0])?"boss/status-immune":"normal/status-enabled")+"/flee0/pdodge0/knockback-immune:size="+@PNLabTarget[1]
  +" race="+@PNLabTarget[2]+" ele="+@PNLabTarget[3]+" lv="+@PNLabTarget[4]+" def="+@PNLabTarget[5]
  +" res="+@PNLabTarget[6]+" mdef="+@PNLabTarget[7]+" mres="+@PNLabTarget[8];
}
function	script	PN_LabApplyTarget	{
 .@mob=getarg(0);
 if (!callfunc("PN_LabTargetValid") || !unitexists(.@mob)) return 0;
 // The combat engine derives class from mode, not the UMOB_CLASS appearance.
 .@mode=MD_KNOCKBACKIMMUNE;
 if (@PNLabTarget[0]) .@mode|=MD_STATUSIMMUNE|MD_DETECTOR;
 setunitdata .@mob,UMOB_MODE,.@mode;
 setunitdata .@mob,UMOB_SIZE,@PNLabTarget[1];
 setunitdata .@mob,UMOB_RACE,@PNLabTarget[2];
 setunitdata .@mob,UMOB_ELETYPE,@PNLabTarget[3];
 setunitdata .@mob,UMOB_ELELEVEL,@PNLabTarget[4];
 setunitdata .@mob,UMOB_DEF,@PNLabTarget[5];
 setunitdata .@mob,UMOB_RES,@PNLabTarget[6];
 setunitdata .@mob,UMOB_MDEF,@PNLabTarget[7];
 setunitdata .@mob,UMOB_MRES,@PNLabTarget[8];
 setunitdata .@mob,UMOB_FLEE,0;
 setunitdata .@mob,UMOB_PDODGE,0;
 return 1;
}

// Character registry arrays survive normal logout/relog. Only the completion
// path above calls Save; no partial run becomes a saved result.
function	script	PN_LabSave	{
 if (getarg(2)<60000 || getarg(1)<0 || getarg(6)=="") return;
 .@slot=PNLabNext%3;
 PNLabValid[.@slot]=0;
 PNLabName$[.@slot]=getarg(0);
 PNLabDamage[.@slot]=getarg(1);
 PNLabMs[.@slot]=getarg(2);
 PNLabDps[.@slot]=getarg(1)*1000/getarg(2);
 PNLabSetup$[.@slot]=getarg(3);
 PNLabGear$[.@slot]=getarg(4);
 PNLabBuff$[.@slot]=getarg(5);
 PNLabBuild$[.@slot]=getarg(6);
 PNLabVariable[.@slot]=getarg(7);
 PNLabTime[.@slot]=gettimetick(2);
 PNLabValid[.@slot]=1;
 PNLabNext=(.@slot+1)%3;
 return;
}
function	script	PN_LabCompatible	{
 .@a=getarg(0); .@b=getarg(1);
 if (.@a<0 || .@a>2 || .@b<0 || .@b>2) return 0;
 return PNLabValid[.@a] && PNLabValid[.@b] && !PNLabVariable[.@a] && !PNLabVariable[.@b]
  && PNLabSetup$[.@a]==PNLabSetup$[.@b] && PNLabBuff$[.@a]==PNLabBuff$[.@b]
  && PNLabBuild$[.@a]!="" && PNLabBuild$[.@a]==PNLabBuild$[.@b];
}
function	script	PN_LabHistory	{
 mes "[Saved Damage Lab Runs]";
 mes "Last three completed runs, saved on this character. Each run records its exact target class and defenses. Older v1 targets used Guardian class and cannot compare with v2 targets.";
 for (.@i=0; .@i<3; .@i++) {
  if (!PNLabValid[.@i]) continue;
  mes "Run "+(.@i+1)+": "+PNLabName$[.@i]+" | DPS "+PNLabDps[.@i];
  mes "Damage "+PNLabDamage[.@i]+" / "+PNLabMs[.@i]+" ms; saved epoch "+PNLabTime[.@i];
  mes PNLabSetup$[.@i];
  mes "Gear/stats/traits/skills: "+PNLabGear$[.@i];
  mes "Initial buffs: "+PNLabBuff$[.@i]+"; build: "+PNLabBuild$[.@i];
  if (PNLabVariable[.@i]) mes "Buff state changed during this run; excluded from automatic comparison.";
 }
 next;
 .@a=select("Compare run 1:Compare run 2:Compare run 3:Leave")-1;
 if (.@a==3) return;
 if (!PNLabValid[.@a]) { mes "That run slot is empty."; return; }
 .@n=0;
 for (.@i=0; .@i<3; .@i++) {
  if (!PNLabValid[.@i]) continue;
  if (!callfunc("PN_LabCompatible",.@a,.@i)) {
   if (PNLabVariable[.@a]) .@reason$="selected run's buffs changed during the run.";
   else if (PNLabVariable[.@i]) .@reason$="buffs changed during this run.";
   else if (PNLabSetup$[.@a]!=PNLabSetup$[.@i]) .@reason$="target settings differ.";
   else if (PNLabBuff$[.@a]!=PNLabBuff$[.@i]) .@reason$="initial buffs differ.";
   else if (PNLabBuild$[.@a]=="" || PNLabBuild$[.@i]=="") .@reason$="verified server build is unavailable.";
   else .@reason$="server build differs.";
   mes "Run "+(.@i+1)+" excluded: "+.@reason$;
   .@excluded++;
   continue;
  }
  if (PNLabGear$[.@a]!=PNLabGear$[.@i]) {
   mes "A/B candidate run "+(.@i+1)+": DPS "+PNLabDps[.@i]+" vs "+PNLabDps[.@a]+". Target, initial buffs and build match; gear/stat/skill identity differs.";
   .@ab++;
   continue;
  }
  .@dps[.@n]=PNLabDps[.@i]; .@n++;
 }
 mes "Comparison counts: "+.@n+" identical; "+.@ab+" A/B; "+.@excluded+" excluded.";
 if (!.@n) { mes "No comparable stable-buff runs. Keep targets, buffs and server build identical."; return; }
 for (.@i=0; .@i<.@n; .@i++) for (.@j=.@i+1; .@j<.@n; .@j++) {
  if (.@dps[.@i]>.@dps[.@j]) { .@v=.@dps[.@i]; .@dps[.@i]=.@dps[.@j]; .@dps[.@j]=.@v; }
 }
 .@median=.@dps[.@n/2];
 if (.@n==2) .@median=.@dps[0]+(.@dps[1]-.@dps[0])/2;
 mes "Identical setup repeats: "+.@n+"; median DPS "+.@median+"; min/max "+.@dps[0]+" / "+.@dps[.@n-1]+"; spread "+(.@dps[.@n-1]-.@dps[0]);
 mes "Rotation and timing are player-controlled. Matching identities do not prove identical skill use.";
 return;
}
