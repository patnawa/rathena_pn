//= Wave Mode
//===== Description: =========================================
//= [Walkthrough Conversion]
//= Wave Mode Forest and Sky Instances
//- Officially monsters can use their skills
//    (NPC_SUMMONSLAVE, at least). NPC_EMOTION and
//    NPC_EMOTION_ON have been blocked since it disrupts the
//    walk system of the instance on rAthena.
//- Forest and Sky are cyclic survival modes. There is no
//    invented clear condition: they run until 20 monsters
//    escape or the one-hour instance lifetime expires.
//===== Changelogs: ==========================================

// Move an idle Wave Mode unit along its route. Return true only
// when the unit reaches the finish line and must count as escaped.
function	script	F_WaveModeMobIdle	{
	.@game_id = getattachedrid();
	if (!unitexists(.@game_id))
		return false;
	sleep2 1000;
	if (!unitexists(.@game_id))
		return false;

	.@dist_min = 300;
	.@size = getarg(1);
	.@dx = getarg(3);
	getunitdata .@game_id, .@data;

	for ( .@i = 0; .@i < .@size; .@i++ ) {
		.@dist_to_spot[.@i] = distance( .@data[UMOB_X], .@data[UMOB_Y], (getelementofarray( getarg(2),.@i ) + .@dx), getelementofarray( getarg(4),.@i ) );
		if (.@dist_min >= .@dist_to_spot[.@i]) {
			.@dist_min = .@dist_to_spot[.@i];
			.@index = .@i;
		}
	}

	if (.@dist_min <= 1) {
		.@index++;
		if (.@index >= .@size) {
			unitkill .@game_id;
			return true;
		}
	}
	else if (.@index < (.@size -1)) {
		.@total[0] = .@dist_min + getelementofarray( getarg(5), .@index );
		.@total[1] = .@dist_to_spot[.@index + 1] + getelementofarray( getarg(5), .@index+1 );
		if (.@total[0] > .@total[1])
			.@index = .@index + 1;
	}

	if (.@dist_to_spot[.@index] > 14) {
		unitkill .@game_id;
		return false;
	}
	if (!.@data[UMOB_TARGETID]) {
		unitwalk .@game_id, (getelementofarray( getarg(2),.@index ) + .@dx), getelementofarray( getarg(4),.@index );
		sleep2 50;	// for now a delay between unitwalk and mob_setidleevent is needed
	}
	if (unitexists(.@game_id))
		mob_setidleevent .@game_id, getarg(0);
	return false;
}

// Clear stale daily standby quests at their official 04:00 reset.
function	script	F_WaveModeCooldownReady	{
	.@quest = getarg(0);
	.@state = checkquest(.@quest,PLAYTIME);
	if (.@state == 2) {
		erasequest .@quest;
		return true;
	}
	return (.@state == -1);
}

// Each entrant returns to the exact Zonda representative used to enter.
function	script	F_WaveModeReturn	{
	.@map$ = wave_mode_map$;
	.@x = wave_mode_x;
	.@y = wave_mode_y;
	wave_mode_map$ = "";
	wave_mode_x = wave_mode_y = 0;
	WaveMode_ActiveInstance = 0;
	WaveMode_ActiveToken = 0;
	if (.@map$ == "") {
		.@map$ = "prontera";
		.@x = 146;
		.@y = 75;
	}
	warp .@map$, .@x, .@y;
	return;
}

// Return every online player before an instance is destroyed. Offline
// characters are protected by the nosave mapflag below on their next login.
function	script	F_WaveModeReturnAll	{
	.@map$ = getarg(0);
	.@count = getmapunits(BL_PC,.@map$,.@players[0]);
	for (.@i = 0; .@i < .@count; ++.@i) {
		if (attachrid(.@players[.@i]))
			callfunc "F_WaveModeReturn";
	}
	detachrid;
	return;
}

// If an instance disappears while a character is offline, nosave moves the
// character to their save point. Restore the recorded Zonda origin on login.
-	script	WaveModeSafety#pn	-1,{
	end;
OnPCLoginEvent:
	if (WaveMode_ActiveInstance < 1)
		end;
	.@iid = WaveMode_ActiveInstance;
	.@name$ = instance_live_info(ILI_NAME,.@iid);
	// rAthena evaluates both sides of logical operators. Validate the live
	// instance before getinstancevar so a stale ID can never reach that command.
	if (.@name$ != "Wave Mode - Forest" && .@name$ != "Wave Mode - Sky") {
		callfunc "F_WaveModeReturn";
		end;
	}
	if (instance_live_info(ILI_OWNER,.@iid) != getcharid(1)) {
		callfunc "F_WaveModeReturn";
		end;
	}
	if (getinstancevar('run_token,.@iid) == WaveMode_ActiveToken)
		end;
	callfunc "F_WaveModeReturn";
	end;
}
